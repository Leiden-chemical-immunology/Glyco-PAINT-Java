001/*=============================================================================
002 *  Class:        GlycoPaintInstallerWindows.java
003 *  Package:      paint.installer
004 *
005 *  PURPOSE:
006 *    Provides a Windows installer for the Glyco-PAINT suite. The installer
007 *    unpacks the embedded payload (EXE launchers plus the Fiji plugin) and
008 *    installs the user-selected components. If a Fiji installation is found,
009 *    the plugin JAR is installed automatically.
010 *
011 *  DESCRIPTION:
012 *    The installer extracts an embedded ZIP (payload.zip) containing:
013 *      • Viewer.exe
014 *      • Generate Squares.exe
015 *      • Get Omero.exe
016 *      • Create Experiment.exe
017 *      • A Fiji plugin payload in payload/plugin/
018 *
019 *    The user selects which EXE launchers to install and chooses a parent
020 *    installation directory, under which a folder "Glyco-PAINT" is created.
021 *
022 *    Fiji plugin installation uses the following search order:
023 *
024 *      1) Saved Fiji path from preferences (if any)
025 *      2) Auto-detected paths composed from these base directories:
026 *           - %ProgramFiles%
027 *           - %ProgramFiles(x86)%
028 *           - %LOCALAPPDATA%   (usually C:\Users\<user>\AppData\Local)
029 *         combined with these directory names:
030 *           - Fiji
031 *           - Fiji-win64
032 *           - Fiji-win32
033 *           - Fiji-Windows
034 *      3) Ask the user: first show a warning dialog (OK/Cancel); if OK,
035 *         open a directory chooser to select the Fiji folder.
036 *
037 *    If no valid Fiji directory (containing a "plugins" subfolder) is found,
038 *    the plugin payload is copied to <installRoot>\plugin\ for manual install.
039 *
040 *  KEY FEATURES:
041 *    • Installs selected EXE launchers from the payload.
042 *    • Attempts automatic Fiji detection with full logging of attempted paths.
043 *    • Warn-then-choose flow for manual Fiji selection (matches mac behavior).
044 *    • Remembers last install parent and last known Fiji path.
045 *    • Extracts plugin payload and installs/exports it as appropriate.
046 *

087     * Windows Fiji directory names to try under each base path.

091            "Fiji",
092            "Fiji-win64",
093            "Fiji-win32",
094            "Fiji-Windows"
095    };

097    /**
098     * Constructs the list of candidate Fiji directories by combining
099     * these environment variables (if defined) with FIJI_DIR_NAMES:
100     *   - ProgramFiles
101     *   - ProgramFiles(x86)
102     *   - LOCALAPPDATA
103     * <p>
104     * Examples (depending on system):
105     *   C:\Program Files\Fiji
106     *   C:\Program Files (x86)\Fiji-win64
107     *   C:\Users\<user>\AppData\Local\Fiji-Windows
108     */
109    private static String[] buildFijiPaths() {
110        String[] bases = {
111                envPath("ProgramFiles"),
112                envPath("ProgramFiles(x86)"),
113                envPath("LOCALAPPDATA")
114        };
115
116        java.util.List<String> out = new java.util.ArrayList<>();
117        for (String base : bases) {
118            for (String name : FIJI_DIR_NAMES) {
119                out.add(base + File.separator + name);
120            }
121        }
122        return out.toArray(new String[0]);
123    }
124
125    /** All auto-detected candidate Fiji paths (constructed at class load). */
126    private static final String[] FIJI_PATHS = buildFijiPaths();
127
128    /** Helper to read an environment variable, returning empty string if absent. */
129    private static String envPath(String key) {
130        String v = System.getenv(key);
131        return v == null ? "" : v;
132    }

206    /**
207     * Extracts a version string from filenames in payload.zip matching:
208     *   paint-<module>-X.Y[.Z]
209     */
210    private void detectVersion() {
211        try (InputStream inputStream = getClass().getResourceAsStream(PAYLOAD_NAME);
212             ZipInputStream zis = new ZipInputStream(inputStream)) {
213            ZipEntry entry;
214            Pattern pat = Pattern.compile("paint-[a-zA-Z-]+-([0-9]+\\.[0-9]+(\\.[0-9]+)?)");
215            while ((entry = zis.getNextEntry()) != null) {
216                Matcher m = pat.matcher(entry.getName());
217                if (m.find()) {
218                    version = m.group(1);
219                    break;
220                }
221            }
222        } catch (IOException ignored) {}
223    }

231                PREF_NODE,
232                KEY_PARENT_DIR,
233                FileSystemView.getFileSystemView().getDefaultDirectory().getAbsolutePath()
234        );

236        Path parent = Paths.get(defaultParent);

239        JPanel top = new JPanel(new BorderLayout(10, 10));
240        top.setBorder(BorderFactory.createEmptyBorder(10, 10, 0, 10));
241
242        /* Install directory selection */
243        JPanel dirPanel = new JPanel(new BorderLayout(6, 0));
244        JLabel dirLabel = new JLabel("Install location (parent folder):");
245        JTextField dirField = new JTextField(parent.toString());
246        JButton browseButton = new JButton("Browse…");
247
248        dirPanel.add(dirLabel, BorderLayout.WEST);
249        dirPanel.add(dirField, BorderLayout.CENTER);
250        dirPanel.add(browseButton, BorderLayout.EAST);
251        top.add(dirPanel, BorderLayout.NORTH);
252
253        /* Component selection */
254        JPanel compPanel = new JPanel();
255        compPanel.setLayout(new GridLayout(0, 1));
256        compPanel.setBorder(BorderFactory.createTitledBorder("Select components to install"));
257
258        cbViewer     = new JCheckBox("Viewer.exe", true);
259        cbGenerate   = new JCheckBox("Generate Squares.exe", true);
260        cbOmero      = new JCheckBox("Get Omero.exe", true);
261        cbExperiment = new JCheckBox("Create Experiment.exe", true);
262        cbPlugin     = new JCheckBox("Fiji Plugin", true);
263
264        compPanel.add(cbViewer);
265        compPanel.add(cbGenerate);
266        compPanel.add(cbOmero);
267        compPanel.add(cbExperiment);
268        compPanel.add(cbPlugin);
269
270        top.add(compPanel, BorderLayout.CENTER);
271
272        /* Action buttons */
273        JPanel buttons = new JPanel(new FlowLayout(FlowLayout.RIGHT, 8, 0));

275        JButton cancelButton  = new JButton("Cancel");
276        buttons.add(cancelButton);
277        buttons.add(installButton);
278        top.add(buttons, BorderLayout.SOUTH);
279
280        frame.add(top, BorderLayout.NORTH);
281
282        /* Directory chooser */
283        browseButton.addActionListener(e -> {
284            JFileChooser chooser = new JFileChooser();
285            chooser.setDialogTitle("Choose installation parent folder");
286            chooser.setFileSelectionMode(JFileChooser.DIRECTORIES_ONLY);
287            chooser.setCurrentDirectory(new File(dirField.getText()));
288            chooser.setSelectedFile(new File(dirField.getText()));
289            chooser.setAcceptAllFileFilterUsed(false);
290
291            int r = chooser.showOpenDialog(frame);
292            if (r == JFileChooser.APPROVE_OPTION) {
293                File f = chooser.getSelectedFile();
294                dirField.setText(f.getAbsolutePath());
295            }
296        });
297
298        cancelButton.addActionListener(e -> System.exit(0));
299
300        /* Start installation */
301        installButton.addActionListener(e -> {
302            Path chosenParent = Paths.get(dirField.getText()).normalize();
303            installRoot = chosenParent.resolve(PRODUCT_NAME);
304
305            PaintPrefs.putString(PREF_NODE, KEY_PARENT_DIR, chosenParent.toString());
306
307            try {
308                Files.createDirectories(installRoot);
309            } catch (IOException ex) {
310                JOptionPane.showMessageDialog(frame,
311                                              "Cannot create folder:\n" + installRoot + "\n" + ex.getMessage(),
312                                              "Permission error",
313                                              JOptionPane.ERROR_MESSAGE);
314                return;
315            }
316
317            installButton.setEnabled(false);
318            cancelButton.setEnabled(false);
319            browseButton.setEnabled(false);
320            dirField.setEnabled(false);
321
322            Executors.newSingleThreadExecutor().submit(this::runInstaller);
323        });
324
325        frame.setVisible(true);
326    }

332    private void runInstaller() {
333        try {
334            SwingUtilities.invokeLater(() -> progress.setVisible(true));
335            log("Installing " + PRODUCT_NAME + " " + version + " into " + installRoot);
336            log("");
337
338            Path tmpZip    = Files.createTempFile("glyco-paint", ".zip");
339            Path pluginTmp = Files.createTempDirectory("glyco-paint-plugin");
340
341            /* Copy embedded payload to temp */
342            try (InputStream in = getClass().getResourceAsStream(PAYLOAD_NAME)) {
343                if (in == null) throw new IOException("Missing embedded payload.zip");
344                Files.copy(in, tmpZip, StandardCopyOption.REPLACE_EXISTING);
345            }
346
347            /* Extract EXEs and plugin payload */
348            extractZip(tmpZip, installRoot, pluginTmp);
349
350            /* Try to install Fiji plugin (if selected) */
351            boolean pluginInstalled = installFijiPlugin(pluginTmp);
352
353            /* Cleanup temp ZIP */
354            Files.deleteIfExists(tmpZip);
355
356            if (pluginInstalled) {
357                try (java.util.stream.Stream<Path> walk = Files.walk(pluginTmp)) {
358                    walk.sorted((a, b) -> b.compareTo(a))
359                        .forEach(p -> p.toFile().delete());
360                }
361            } else {
362                if (cbPlugin.isSelected()) {
363                    Path manual = installRoot.resolve("plugin");
364                    log("Fiji not found — copying plugin for manual installation: " + manual);
365                    copyDirectory(pluginTmp, manual);
366                }
367            }
368
369            SwingUtilities.invokeLater(() -> {
370                progress.setVisible(false);
371                closeButton.setEnabled(true);
372            });
373            log("");
374            log("Installation complete.");
375
376        } catch (Exception e) {
377            log("Installation failed: " + e.getMessage());
378            SwingUtilities.invokeLater(() -> {
379                progress.setVisible(false);
380                closeButton.setEnabled(true);
381            });
382        }
383    }

389    /**
390     * Attempts to install the Fiji plugin only if the user selected the
391     * "Fiji Plugin" checkbox. Search order:
392     *  1) Saved path
393     *  2) Auto-detected paths (see header for full list)
394     *  3) Ask the user (warning first, then chooser if OK)
395     * <p>
396     * Returns true if installed into a valid Fiji\plugins directory, else false.
397     */
398    private boolean installFijiPlugin(Path pluginSourceRoot) throws IOException {
399        if (!cbPlugin.isSelected()) {
400            log("Fiji plugin skipped by user.");
401            return false;
402        }
403
404        Optional<Path> pluginJar = findPluginJar(pluginSourceRoot);
405        if (!pluginJar.isPresent()) {
406            log("No Fiji plugin JAR found in payload under " + pluginSourceRoot);
407            return false;
408        }
409
410        Path jar = pluginJar.get();
411        log("Found Fiji plugin JAR: " + jar.getFileName());
412
413        /* 1) Saved path */
414        String saved = PaintPrefs.getString(PREF_NODE, KEY_FIJI_DIR, null);
415        if (saved != null) {
416            log("Trying saved Fiji path: " + saved);
417            Path pluginsDir = Paths.get(saved, "plugins");
418            if (Files.isDirectory(pluginsDir)) {
419                log("Saved Fiji path is valid.");
420                installJarIntoFijiDir(jar, pluginsDir);
421                return true;
422            } else {
423                log("Saved Fiji path is NOT valid: " + pluginsDir);
424            }
425        }
426
427        /* 2) Auto-detected list */
428        for (String base : FIJI_PATHS) {
429            if (base == null || base.isEmpty()) continue;
430
431            log("Trying auto-detected Fiji path: " + base);
432            Path pluginsDir = Paths.get(base, "plugins");
433
434            if (Files.isDirectory(pluginsDir)) {
435                log("Found Fiji at: " + base);
436                installJarIntoFijiDir(jar, pluginsDir);
437                PaintPrefs.putString(PREF_NODE, KEY_FIJI_DIR, base);
438                return true;
439            }
440        }
441
442        /* 3) Ask the user — show warning, then chooser only if OK */
443        Path manual = askUserForFijiFolder();
444        if (manual != null) {
445            Path pluginsDir = manual.resolve("plugins");
446            if (Files.isDirectory(pluginsDir)) {
447                installJarIntoFijiDir(jar, pluginsDir);
448                PaintPrefs.putString(PREF_NODE, KEY_FIJI_DIR, manual.toString());
449                return true;
450            }
451        }
452
453        log("No valid Fiji folder. Plugin will be copied for manual installation.");
454        return false;
455    }
456
457    /** Deletes old paint-*.jar and copies the new plugin into Fiji\plugins. */
458    private void installJarIntoFijiDir(Path jar, Path pluginsDir) throws IOException {
459        try (java.util.stream.Stream<Path> stream = Files.list(pluginsDir)) {
460            stream
461                    .filter(p -> p.getFileName().toString().startsWith("paint-") &&
462                            p.getFileName().toString().endsWith(".jar"))
463                    .forEach(p -> { try { Files.delete(p); } catch (IOException ignored) {} });
464        }
465
466        Files.copy(jar, pluginsDir.resolve(jar.getFileName()), StandardCopyOption.REPLACE_EXISTING);
467        log("Installed plugin to: " + pluginsDir);
468    }

474    /**
475     * Extracts the payload, installing only selected EXEs and extracting
476     * plugin payload into a temp directory.
477     */
478    private void extractZip(Path zipFile, Path targetDir, Path pluginTemp) throws IOException {
479        try (ZipInputStream zis = new ZipInputStream(Files.newInputStream(zipFile))) {
480            ZipEntry entry;
481            byte[] buf = new byte[8192];
482            boolean pluginStart = false;
483
484            while ((entry = zis.getNextEntry()) != null) {
485                String name = entry.getName();
486
487                /* Skip macOS fluff if present in ZIP */
488                if (name.startsWith("__MACOSX/") || name.endsWith(".DS_Store")) continue;
489
490                /* Plugin payload */
491                if (name.startsWith("plugin/")) {
492                    if (!pluginStart) {
493                        pluginStart = true;
494                        log("");
495                        log("Extracting Fiji plugin payload...");
496                    }
497                    Path out = pluginTemp.resolve(name.substring("plugin/".length()));
498                    writeZipEntry(zis, buf, entry, out);
499                    continue;
500                }
501
502                /* Respect user selections for EXEs */
503                if (name.equals("Viewer.exe") && !cbViewer.isSelected()) {
504                    continue;
505                }
506                if (name.equals("Generate Squares.exe") && !cbGenerate.isSelected()) {
507                    continue;
508                }
509                if (name.equals("Get Omero.exe") && !cbOmero.isSelected()) {
510                    continue;
511                }
512                if (name.equals("Create Experiment.exe") && !cbExperiment.isSelected()) {
513                    continue;
514                }
515
516                if (name.toLowerCase().endsWith(".exe")) {
517                    log("Installing: " + name);
518                }
519
520                Path out = targetDir.resolve(name);
521                writeZipEntry(zis, buf, entry, out);
522            }
523        }
524    }
525
526    /** Writes a single ZIP entry to disk, creating parent directories as needed. */
527    private void writeZipEntry(ZipInputStream zis, byte[] buf, ZipEntry entry, Path out) throws IOException {
528        if (entry.isDirectory()) {
529            Files.createDirectories(out);
530        } else {
531            Files.createDirectories(out.getParent());
532            try (OutputStream os = Files.newOutputStream(out)) {
533                int n;
534                while ((n = zis.read(buf)) > 0) os.write(buf, 0, n);
535            }
536        }
537    }

539    /** Recursively copies one directory tree into another (REPLACE_EXISTING). */
540    private void copyDirectory(Path src, Path dst) {
541        try (java.util.stream.Stream<Path> walk = Files.walk(src)) {
542            walk.forEach(source -> {
543                Path target = dst.resolve(src.relativize(source));
544                try {
545                    if (Files.isDirectory(source)) {
546                        Files.createDirectories(target);
547                    } else {
548                        Files.copy(source, target, StandardCopyOption.REPLACE_EXISTING);
549                    }
550                } catch (IOException e) {
551                    log("Copy failed: " + source + " -> " + target + ": " + e.getMessage());
552                }
553            });
554        } catch (IOException ignored) {}
555    }

569    /**
570     * Shows a warning dialog explaining Fiji wasn't found. If user presses OK,
571     * opens a folder chooser for the user to select the Fiji directory. If the
572     * user cancels either dialog, returns null.
573     */
574    private Path askUserForFijiFolder() {
575
576        // 1) Warning first (modal)
577        int choice = JOptionPane.showConfirmDialog(
578                frame,
579                "Fiji installation not found.\nPlease select your Fiji folder (contains Fiji.exe).",
580                "Fiji Not Found",
581                JOptionPane.OK_CANCEL_OPTION,
582                JOptionPane.WARNING_MESSAGE
583        );
584
585        // 2. If user presses Cancel -> do not show folder picker
586        if (choice != JOptionPane.OK_OPTION) {
587            log("User cancelled Fiji selection.");
588            return null;
589        }
590
591        final Path[] result = new Path[1];
592
593        // 2) Only open chooser after the warning is dismissed
594        try {
595            SwingUtilities.invokeAndWait(() -> {
596                JFileChooser chooser = new JFileChooser();
597                chooser.setDialogTitle("Select Fiji folder (contains Fiji.exe)");
598                chooser.setFileSelectionMode(JFileChooser.DIRECTORIES_ONLY);
599                chooser.setAcceptAllFileFilterUsed(false);
600
601                int r = chooser.showOpenDialog(frame);
602                if (r == JFileChooser.APPROVE_OPTION) {
603                    result[0] = chooser.getSelectedFile().toPath();
604                } else {
605                    log("User cancelled folder chooser.");
606                }
607            });
608        } catch (Exception ignored) {}
609
610        return result[0];
611    }

617    /**
618     * Prefer paint-fiji-plugin-*.jar; fallback to any .jar under plugin payload.
619     * If multiple candidates exist, choose the lexicographically last name.
620     */
621    private Optional<Path> findPluginJar(Path root) throws IOException {
622
623        // Comparator selecting lexicographically LAST filename
624        Comparator<Path> byNameDescending =
625                (a, b) -> b.getFileName().toString().compareTo(a.getFileName().toString());
626
627        // Preferred: paint-fiji-plugin-*.jar
628        try (java.util.stream.Stream<Path> s = Files.walk(root)) {
629            Optional<Path> preferred =
630                    s.filter(Files::isRegularFile)
631                     .filter(p -> p.getFileName().toString().endsWith(".jar"))
632                     .filter(p -> p.getFileName().toString().startsWith("paint-fiji-plugin-"))
633                     .min(byNameDescending);      // <-- replaces sorted().findFirst()
634
635            if (preferred.isPresent()) {
636                return preferred;
637            }
638        }
639
640        // Fallback: any .jar
641        try (java.util.stream.Stream<Path> s2 = Files.walk(root)) {
642            return s2.filter(Files::isRegularFile)
643                     .filter(p -> p.getFileName().toString().endsWith(".jar"))
644                     .min(byNameDescending);        // <-- also using min()
645        }
646    }
