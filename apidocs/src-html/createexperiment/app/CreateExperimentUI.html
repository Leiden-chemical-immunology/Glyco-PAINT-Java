037package createexperiment.app;
038
039import javax.swing.*;
040import java.awt.*;
041import java.awt.event.ActionEvent;
042import java.io.File;
043import java.io.IOException;
044import java.nio.file.Path;
045import java.util.List;
046import java.util.function.Function;
047import java.util.prefs.Preferences;
048import java.util.regex.Pattern;
049import java.util.regex.PatternSyntaxException;
050
051import static createexperiment.app.ExperimentInfoWriter.exportExperimentInfo;
052
053/**
054 * Fixed and visually balanced CreateExperimentUI.
055 */
056public class CreateExperimentUI {
057
058    private static final String PREF_NODE   = "paint/create-experiment";
059    private static final String KEY_IMAGES  = "lastImagesDir";
060    private static final String KEY_PROJECT = "lastProjectDir";
061
062    private static final String[] DEFAULT_REGEXES = {
063            "",
064            "^\\d-Exp-\\d+-[A-Z]\\d+-\\d+\\.nd2$",
065            "^(?!.*BF).*\\.nd2$"
066    };
067
068    /**
069     * The main method serves as the entry point for the application, initializing
070     * and displaying the Create Experiment GUI.
071     *
072     * @param args the command-line arguments passed to the application
073     */
074    public static void main(String[] args) {
075        SwingUtilities.invokeLater(CreateExperimentUI::createAndShowGUI);
076    }
077
078    /**
079     * Configures and displays a graphical user interface (GUI) for managing regex filters,
080     * selecting input/output directories, and processing files. The GUI consists of multiple
081     * sections, including regex controls, file list display, input/output directory configurations,
082     * and action buttons. This method also uses preferences to remember user choices
083     * (e.g., regex history and last-used directories) and dynamically updates its components
084     * based on user actions.
085     * <p>
086     * Key Features:
087     * - Regex management: Users can add, select, and delete regex patterns.
088     * - Input/output directory selection: Users can specify directories for source files
089     * and output storage.
090     * - File display: Files in the selected input directory are listed and filtered based
091     * on the regex.
092     * - Persisted preferences: User choices (like regex patterns or directories) are saved
093     * and reloaded on subsequent application launches.
094     * - Modular panel layout: Organized interface using labeled sections for regex filtering,
095     * file selection, and processing actions.
096     * <p>
097     * Components:
098     * - Regex Controls: Dropdown allowing users to input regex patterns, supported by history and deletion menus.
099     * - File List: A dynamically updated list of files matching the selected regex in the input directory.
100     * - Input/Output Controls: Buttons and labels for selecting directories and displaying their current paths.
101     * - Action Buttons: Buttons for initiating the file processing operation and closing the application.
102     * <p>
103     * Dialogs:
104     * - Uses JFileChooser dialogs for selecting directories.
105     * <p>
106     * Event Handling:
107     * - Handles user interactions via buttons and combo box events (e.g., regex filtering, file refreshing).
108     * <p>
109     * Constraints:
110     * - Limits regex entry length to a maximum of 100 characters.
111     * - Ensures only valid directories are selected and persistently stores their paths.
112     * <p>
113     * Dependencies:
114     * - javax.swing (for GUI elements like JFrame, JPanel, JComboBox, JList, JButton).
115     * - java.util.prefs.Preferences (for storing user-specific persistent settings).
116     * - java.io.File (for directory and file handling).
117     * <p>
118     * Usage:
119     * Typically called during application initialization to present the user interface.
120     * This method does not return, as it focuses on creating and displaying the application
121     * window with active event handling for user interaction.
122     */
123    private static void createAndShowGUI() {
124        Preferences prefs = Preferences.userRoot().node(PREF_NODE);
125
126        JFrame frame = new JFrame("Create Experiment");
127        frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
128
129        Font smallFont = new JLabel().getFont().deriveFont(Font.PLAIN, 12f);
130        int labelWidth = 100;
131        int buttonWidth = 90;
132        Dimension fieldSize = new Dimension(360, 24);
133        Dimension buttonSize = new Dimension(buttonWidth, 24);
134
135        // === Top panel (directories + regex) ===
136        JPanel top = new JPanel(new GridBagLayout());
137        top.setBorder(BorderFactory.createEmptyBorder(10, 12, 5, 12));
138        GridBagConstraints gbc = new GridBagConstraints();
139        gbc.insets = new Insets(3, 3, 3, 3);
140        gbc.fill = GridBagConstraints.HORIZONTAL;
141
142        Function<String, JLabel> label = t -> {
143            JLabel l = new JLabel(t, SwingConstants.LEFT);
144            l.setFont(smallFont);
145            l.setPreferredSize(new Dimension(labelWidth, 24));
146            return l;
147        };
148
149        final File[] projectDir = {new File(prefs.get(KEY_PROJECT, System.getProperty("user.home")))};
150        final File[] imagesDir  = {new File(prefs.get(KEY_IMAGES, System.getProperty("user.home")))};
151
152        // --- Project Root ---
153        gbc.gridy = 0; gbc.gridx = 0;
154        top.add(label.apply("Project Root:"), gbc);
155        gbc.gridx = 1;
156        JTextField projectField = new JTextField(projectDir[0].getAbsolutePath());
157        projectField.setEditable(false);
158        projectField.setFont(smallFont);
159        projectField.setPreferredSize(fieldSize);
160        top.add(projectField, gbc);
161        gbc.gridx = 2;
162        JButton projectBrowse = new JButton("Browse…");
163        projectBrowse.setFont(smallFont);
164        projectBrowse.setPreferredSize(buttonSize);
165        top.add(projectBrowse, gbc);
166
167        // --- Images Root ---
168        gbc.gridy++;
169        gbc.gridx = 0;
170        top.add(label.apply("Images Root:"), gbc);
171        gbc.gridx = 1;
172        JTextField imagesField = new JTextField(imagesDir[0].getAbsolutePath());
173        imagesField.setEditable(false);
174        imagesField.setFont(smallFont);
175        imagesField.setPreferredSize(fieldSize);
176        top.add(imagesField, gbc);
177        gbc.gridx = 2;
178        JButton imagesBrowse = new JButton("Browse…");
179        imagesBrowse.setFont(smallFont);
180        imagesBrowse.setPreferredSize(buttonSize);
181        top.add(imagesBrowse, gbc);
182
183        // --- Regex ---
184        gbc.gridy++;
185        gbc.gridx = 0;
186        top.add(label.apply("Regex:"), gbc);
187        gbc.gridx = 1;
188        JComboBox<String> regexCombo = new JComboBox<>();
189        regexCombo.setEditable(true);
190        regexCombo.setFont(smallFont);
191        for (String def : DEFAULT_REGEXES) regexCombo.addItem(def);
192        top.add(regexCombo, gbc);
193        gbc.gridx = 2;
194        JButton filterButton = new JButton("Filter");
195        filterButton.setFont(smallFont);
196        filterButton.setPreferredSize(buttonSize);
197        top.add(filterButton, gbc);
198
199        // === File list ===
200        DefaultListModel<File> listModel = new DefaultListModel<>();
201        JList<File> fileList = new JList<>(listModel);
202        fileList.setFont(smallFont);
203        JScrollPane scroll = new JScrollPane(fileList);
204        scroll.setBorder(BorderFactory.createEmptyBorder(5, 12, 5, 12));
205
206        // === Bottom action bar ===
207        JPanel actions = new JPanel(new FlowLayout(FlowLayout.RIGHT, 8, 8));
208        JButton process = new JButton("Process");
209        JButton close = new JButton("Close");
210        process.setFont(smallFont);
211        close.setFont(smallFont);
212        process.setPreferredSize(buttonSize);
213        close.setPreferredSize(buttonSize);
214        actions.add(process);
215        actions.add(close);
216
217        frame.add(top, BorderLayout.NORTH);
218        frame.add(scroll, BorderLayout.CENTER);
219        frame.add(actions, BorderLayout.SOUTH);
220
221        // === Logic ===
222        Runnable refresh = () -> {
223            String regex = ((String) regexCombo.getEditor().getItem()).trim();
224            listModel.clear();
225            if (imagesDir[0] == null) return;
226            Pattern p = null;
227            if (!regex.isEmpty()) {
228                try { p = Pattern.compile(regex); }
229                catch (PatternSyntaxException ex) {
230                    JOptionPane.showMessageDialog(frame, "Invalid regex: " + ex.getDescription());
231                    return;
232                }
233            }
234            File[] files = imagesDir[0].listFiles(f -> f.isFile() && !f.isHidden());
235            if (files == null) return;
236            for (File f : files)
237                if (p == null || p.matcher(f.getName()).matches())
238                    listModel.addElement(f);
239        };
240
241        projectBrowse.addActionListener(e -> {
242            JFileChooser ch = new JFileChooser(projectDir[0]);
243            ch.setDialogTitle("Select Project Directory");
244            ch.setFileSelectionMode(JFileChooser.DIRECTORIES_ONLY);
245            ch.setAcceptAllFileFilterUsed(false); // ✅ hides "All files" filter
246            ch.setMultiSelectionEnabled(false);   // ✅ single directory only
247            if (ch.showOpenDialog(frame) == JFileChooser.APPROVE_OPTION) {
248                projectDir[0] = ch.getSelectedFile();
249                projectField.setText(projectDir[0].getAbsolutePath());
250                prefs.put(KEY_PROJECT, projectDir[0].getAbsolutePath());
251            }
252        });
253
254        imagesBrowse.addActionListener(e -> {
255            JFileChooser ch = new JFileChooser(imagesDir[0]);
256            ch.setDialogTitle("Select Images Directory");
257            ch.setFileSelectionMode(JFileChooser.DIRECTORIES_ONLY);
258            ch.setAcceptAllFileFilterUsed(false);
259            ch.setMultiSelectionEnabled(false);
260            if (ch.showOpenDialog(frame) == JFileChooser.APPROVE_OPTION) {
261                imagesDir[0] = ch.getSelectedFile();
262                imagesField.setText(imagesDir[0].getAbsolutePath());
263                prefs.put(KEY_IMAGES, imagesDir[0].getAbsolutePath());
264                refresh.run();
265            }
266        });
267
268        filterButton.addActionListener(e -> refresh.run());
269        regexCombo.addActionListener(e -> refresh.run());
270
271        process.addActionListener((ActionEvent e) -> {
272            if (projectDir[0] == null || imagesDir[0] == null) {
273                JOptionPane.showMessageDialog(frame, "Select both Project and Images directories.");
274                return;
275            }
276            List<File> selected = fileList.getSelectedValuesList();
277            if (selected.isEmpty()) {
278                JOptionPane.showMessageDialog(frame, "No files selected.");
279                return;
280            }
281            try {
282                String experimentName    = imagesDir[0].getName();
283                Path   experimentDirPath = projectDir[0].toPath().resolve(experimentName);
284
285                Path created = exportExperimentInfo(experimentDirPath, selected);
286                JOptionPane.showMessageDialog(frame,
287                                              "Experiment info written to:\n" + created.toAbsolutePath(),
288                                              "Success", JOptionPane.INFORMATION_MESSAGE);
289            } catch (IOException ex) {
290                JOptionPane.showMessageDialog(frame, "Error: " + ex.getMessage());
291            }
292        });
293
294        close.addActionListener(e -> frame.dispose());
295
296        frame.setSize(560, 520);
297        frame.setLocationRelativeTo(null);
298        frame.setVisible(true);
299
300        refresh.run();
301    }
