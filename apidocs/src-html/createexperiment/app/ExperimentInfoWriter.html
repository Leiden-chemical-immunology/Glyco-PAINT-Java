033package createexperiment.app;
034
035import paint.shared.objects.ExperimentInfo;
036
037import java.io.File;
038import java.io.IOException;
039import java.nio.file.Files;
040import java.nio.file.Path;
041import java.util.ArrayList;
042import java.util.List;
043import java.util.regex.Matcher;
044import java.util.regex.Pattern;
045
046import static paint.shared.constants.PaintFileNames.EXPERIMENT_INFO_CSV;
047import static paint.shared.io.MainIOInterface.writeSpecificExperimentInfoFile;
048
049/**
050 * Provides functionality to generate and write an experiment information CSV file
051 * based on a list of ND2 recording files. Automatically parses and extracts condition
052 * and replicate numbers from filenames, and ensures unique file names to avoid overwriting.
053 */
054public class ExperimentInfoWriter {
055
056    /**
057     * Regular expression pattern for extracting metadata from filenames of experiment recordings.
058     * <p>
059     * Expected filename format:
060     * <pre>
061     * [digits]-Exp-[digits]-[uppercase letter][digits]-[digits].nd2
062     * </pre>
063     * Groups:
064     * <ul>
065     *   <li>Group 1 – Condition number</li>
066     *   <li>Group 2 – Replicate number</li>
067     * </ul>
068     */
069    private static final Pattern FILENAME_PATTERN =
070            Pattern.compile("^\\d+-Exp-(\\d+)-[A-Z]\\d+-(\\d+)\\.nd2$");
071
072    /**
073     * Writes experiment recording metadata into a CSV file located in the specified experiment directory.
074     * <p>
075     * Each ND2 recording file is parsed into an {@link ExperimentInfo} instance containing structured
076     * metadata such as condition and replicate numbers. The resulting table is serialized as
077     * {@code Experiment Info.csv}, with numeric suffixing to avoid overwriting existing files.
078     *
079     * @param experimentDirPath the directory where the experiment CSV will be created
080     * @param recordings        list of ND2 recording files to process
081     * @return the created {@code File} object pointing to the resulting CSV file
082     * @throws IOException if directory creation or file writing fails
083     */
084    public static Path exportExperimentInfo(Path experimentDirPath, List<File> recordings) throws IOException {
085        Files.createDirectories(experimentDirPath);
086
087        // Build ExperimentInfo objects from filenames
088        List<ExperimentInfo> infos = new ArrayList<>();
089
090        for (File rec : recordings) {
091            String name = rec.getName();
092            ExperimentInfo info = new ExperimentInfo();
093            info.setRecordingName(name);
094
095            // Try to parse condition and replicate from filename
096            Matcher m = FILENAME_PATTERN.matcher(name);
097            if (m.matches()) {
098                info.setConditionNumber(Integer.parseInt(m.group(1)));
099                info.setReplicateNumber(Integer.parseInt(m.group(2)));
100            } else {
101                info.setConditionNumber(0);
102                info.setReplicateNumber(0);
103            }
104
105            // Default values
106            info.setProcessFlag(true);
107            info.setProbeName("");
108            info.setProbeType("");
109            info.setCellType("");
110            info.setAdjuvant("");
111            info.setConcentration(0.0);
112            info.setThreshold(0.0);
113
114            infos.add(info);
115        }
116
117        // --- Write table
118        Path csvFilePath = uniqueFile(experimentDirPath, EXPERIMENT_INFO_CSV);
119        writeSpecificExperimentInfoFile(csvFilePath, infos);
120
121        return csvFilePath;
122
123    }
124
125    /**
126     * Ensures the output filename is unique by appending an incremented suffix if needed.
127     *
128     * @param dirPath      directory for the file
129     * @param fileName desired file name
130     * @return a unique path reference that does not overwrite existing files
131     */
132    @SuppressWarnings("SameParameterValue")
133    private static Path uniqueFile(Path dirPath, String fileName) {
134        int    dot  = fileName.lastIndexOf('.');
135        String stem = (dot >= 0) ? fileName.substring(0, dot) : fileName;
136        String ext  = (dot >= 0) ? fileName.substring(dot) : "";
137
138        Path candidate = dirPath.resolve(fileName);
139        int n = 1;
140
141        while (Files.exists(candidate)) {
142            candidate = dirPath.resolve(stem + "-" + n + ext);
143            n++;
144        }
145
146        return candidate;
147    }
148}
