079    private void show() {
080        JFrame frame = new JFrame("Glyco-PAINT – Deliverables Builder");
081        frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
082        frame.setSize(700, 800);
083        frame.setLocationRelativeTo(null);
084        frame.setLayout(new BorderLayout(10, 10));
085
086        JPanel north = new JPanel();
087        north.setBorder(BorderFactory.createEmptyBorder(10, 10, 0, 10));
088        north.setLayout(new BoxLayout(north, BoxLayout.Y_AXIS));
089
090        addSection(north, "macOS Applications");
091        addBoxes(north, 0, 4);
092
093        addSection(north, "Windows Applications");
094        addBoxes(north, 4, 8);
095
096        addSection(north, "Plugin");
097        addBoxes(north, 8, 9);
098
099        frame.add(north, BorderLayout.NORTH);
100
101        log = new JTextArea();
102        log.setEditable(false);
103        log.setFont(new Font(Font.MONOSPACED, Font.PLAIN, 12));
104        frame.add(new JScrollPane(log), BorderLayout.CENTER);
105
106        JPanel south = new JPanel(new FlowLayout(FlowLayout.RIGHT));
107        JButton closeBtn = new JButton("Close");
108        runBtn = new JButton("Generate");
109
110        closeBtn.addActionListener(e -> System.exit(0));
111        runBtn.addActionListener(e -> runSelectedTasks());
112
113        south.add(closeBtn);
114        south.add(runBtn);
115
116        frame.add(south, BorderLayout.SOUTH);
117
118        frame.setVisible(true);
119    }

121    private void addSection(JPanel parent, String title) {
122        JPanel spacer = new JPanel();
123        spacer.setPreferredSize(new Dimension(1, 20));
124        spacer.setOpaque(false);
125        parent.add(spacer);
126
127        JLabel lbl = new JLabel(title);
128        lbl.setFont(lbl.getFont().deriveFont(Font.BOLD, 14f));
129        parent.add(lbl);
130    }
131
132    private void addBoxes(JPanel parent, int from, int to) {
133        JPanel boxPanel = new JPanel();
134        boxPanel.setLayout(new BoxLayout(boxPanel, BoxLayout.Y_AXIS));
135        boxPanel.setBorder(BorderFactory.createEmptyBorder(5, 20, 5, 0));
136
137        for (int i = from; i < to; i++) {
138            Task t = tasks.get(i);
139            JCheckBox box = new JCheckBox(" " + t.label);
140            boxes.put(t, box);
141            boxPanel.add(box);
142        }
143
144        parent.add(boxPanel);
145    }
146
147    private void runSelectedTasks() {
148        runBtn.setEnabled(false);
149
150        new Thread(() -> {
151            try {
152                append("=== Installing shared utils first ===");
153                runSharedUtils();
154
155                append("\n=== Running selected builds ===\n");
156
157                for (Task t : boxes.keySet()) {
158                    if (!boxes.get(t).isSelected()) {
159                        continue;
160                    }
161                    runTask(t);
162                }
163
164                append("\n✅ Finished all selected builds.\n");
165
166            } catch (Exception ex) {
167                append("\n❌ " + ex.getMessage());
168            } finally {
169                SwingUtilities.invokeLater(() -> runBtn.setEnabled(true));
170            }
171        }).start();
172    }
173
174    private void runSharedUtils() throws IOException, InterruptedException {
175        Path utilsDir = PROJECT_ROOT.resolve("paint-shared-utils");
176        append("→ Building paint-shared-utils (installing into local repo)");
177        append("  Dir: " + utilsDir);
178
179        List<String> cmd = Arrays.asList(
180                "mvn", "-q", "-U", "clean", "install", "-DskipTests"
181        );
182
183        ProcessBuilder pb = new ProcessBuilder(cmd);
184        pb.directory(utilsDir.toFile());
185        pb.redirectErrorStream(true);
186
187        Process p = pb.start();
188        try (BufferedReader r =
189                     new BufferedReader(new InputStreamReader(p.getInputStream()))) {
190
191            String line;
192            while ((line = r.readLine()) != null) {
193                append("[paint-shared-utils] " + line);
194            }
195        }
196
197        int exit = p.waitFor();
198        if (exit != 0) {
199            throw new IOException("Failed building paint-shared-utils (exit " + exit + ")");
200        }
201
202        append("✅ Installed paint-shared-utils\n");
203    }
204
205    private void runTask(Task t) throws IOException, InterruptedException {
206        append("→ Building " + t.label);
207        append("  Cmd: " + String.join(" ", t.mavenCmd));
208        append("  Dir: " + t.moduleDir + "\n");
209
210        ProcessBuilder pb = new ProcessBuilder(t.mavenCmd);
211        pb.directory(t.moduleDir.toFile());
212        pb.redirectErrorStream(true);
213
214        Process p = pb.start();
215        try (BufferedReader r = new BufferedReader(new InputStreamReader(p.getInputStream()))) {
216
217            String line;
218            while ((line = r.readLine()) != null) {
219                if (line.contains("sun.misc.Unsafe") || line.contains("HiddenClassDefiner")) {
220                    continue;
221                }
222                append("[" + t.moduleDir.getFileName() + "] " + line);
223            }
224        }
225
226        int exit = p.waitFor();
227        if (exit != 0) {
228            throw new IOException("Failed building " + t.label + " (exit " + exit + ")");
229        }
230
231        append("✅ Built " + t.label + "\n");
232    }
233
234    private void append(String s) {
235        SwingUtilities.invokeLater(() -> {
236            log.append(s + "\n");
237            log.setCaretPosition(log.getDocument().getLength());
238        });
239    }

241    private static class Task {
242        final String label;
243        final Path moduleDir;
244        final List<String> mavenCmd;
245
246        Task(String label, String module, String... cmd) {
247            this.label = label;
248            this.moduleDir = PROJECT_ROOT.resolve(module);
249            this.mavenCmd = Arrays.asList(cmd);
250        }
251    }
