021package convert;
022
023import java.nio.file.Path;
024import java.nio.file.Paths;
025import java.util.*;
026
027public final class ExperimentInfoConverter {
028
029    private final Path inputFile;
030    private final Path outputFile;
031
032    public ExperimentInfoConverter(Path inputDir) {
033        this.inputFile = inputDir.resolve("Experiment Info - Python.csv");
034        this.outputFile = inputDir.resolve("Experiment Info.csv");
035    }
036
037    // ---------------------------------------------------------------
038    // FINAL new header order (column 3)
039    // ---------------------------------------------------------------
040    private static final List<String> HEADER = Arrays.asList(
041            "Experiment Name",
042            "Recording Name",
043            "Condition Number",
044            "Replicate Number",
045            "Probe Name",
046            "Probe Type",
047            "Cell Type",
048            "Adjuvant",
049            "Concentration",
050            "Process Flag",
051            "Threshold"
052    );
053
054    // ---------------------------------------------------------------
055    // NEW COLUMN → OLD POSITION (column 4)
056    // (1-based indexes, because Python export is 1-based)
057    // ---------------------------------------------------------------
058    private static final Map<String,Integer> INDEX_MAP = new LinkedHashMap<>();
059    static {
060        INDEX_MAP.put("Experiment Name",     4);
061        INDEX_MAP.put("Recording Name",      2);
062        INDEX_MAP.put("Condition Number",    5);
063        INDEX_MAP.put("Replicate Number",    6);
064        INDEX_MAP.put("Probe Name",          7);
065        INDEX_MAP.put("Probe Type",          8);
066        INDEX_MAP.put("Cell Type",           9);
067        INDEX_MAP.put("Adjuvant",           10);
068        INDEX_MAP.put("Concentration",      11);
069        INDEX_MAP.put("Process Flag",       13);
070        INDEX_MAP.put("Threshold",          12);
071    }
072
073    public List<Map<String,String>> convert(List<Map<String,String>> src) {
074
075        List<Map<String,String>> out = new ArrayList<>();
076
077        for (Map<String,String> inRow : src) {
078
079            // convert old row to array by index
080            String[] raw = inRow.values().toArray(new String[inRow.size()]);
081
082            Map<String,String> row = new LinkedHashMap<>();
083
084            for (String newCol : HEADER) {
085                int oldPos = INDEX_MAP.get(newCol);
086                int idx = oldPos - 1;  // convert to 0-based
087                String val = (idx < raw.length ? raw[idx] : "");
088                row.put(newCol, val == null ? "" : val.trim());
089            }
090
091            out.add(row);
092        }
093
094        return out;
095    }
096
097    public void run() throws Exception {
098        System.out.println("Reading:  " + inputFile);
099        List<Map<String,String>> src = CsvIO.readSimpleCsv(inputFile);
100
101        System.out.println("Converting Experiment Info...");
102        List<Map<String,String>> out = convert(src);
103
104        System.out.println("Writing:  " + outputFile);
105        CsvIO.writeSimpleCsv(outputFile, HEADER, out);
106
107        System.out.println("✔ Experiment Info conversion complete.");
108    }
109
110    public static void main(String[] args) throws Exception {
111        Path dir = Paths.get("/Users/hans/Downloads/221012 Python Reference - v39");
112        new ExperimentInfoConverter(dir).run();
113    }
114}
