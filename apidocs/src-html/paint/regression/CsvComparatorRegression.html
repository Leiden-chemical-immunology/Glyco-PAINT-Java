057    public static int compareFiles(Path oldCsv, Path newCsv) throws IOException {
058
059        LOGGER.println("📥 Reading baseline file...");
060        List<Map<String, String>> oldRows = readCsv(oldCsv);
061        LOGGER.println("   → " + oldRows.size() + " rows loaded.");
062
063        LOGGER.println("📥 Reading test file...");
064        List<Map<String, String>> newRows = readCsv(newCsv);
065        LOGGER.println("   → " + newRows.size() + " rows loaded.");
066
067        if (oldRows.size() == newRows.size()) {
068            LOGGER.printf("✅ Same number of rows (%d). Ready for strict 1:1 comparison.%n",
069                          oldRows.size());
070        } else {
071            LOGGER.printf("⚠️  Different row counts: baseline=%d, test=%d%n",
072                          oldRows.size(), newRows.size());
073        }
074
075        Map<String, List<Map<String, String>>> oldMulti = toMultiMap(oldRows);
076        Map<String, List<Map<String, String>>> newMulti = toMultiMap(newRows);
077
078        Set<String> allKeys = new TreeSet<>(oldMulti.keySet());
079        allKeys.addAll(newMulti.keySet());
080
081        LOGGER.println("\n🔎 Starting detailed comparison...");
082        LOGGER.println("   → Total unique (Recording Name [+ Square Nr]) keys: " + allKeys.size());
083        LOGGER.println("");
084
085        List<String[]> diffs = new ArrayList<>();
086        int diffCount = 0;
087
088        int processed = 0;
089        int totalKeys = allKeys.size();
090
091        for (String key : allKeys) {
092            List<Map<String, String>> ol = oldMulti.get(key);
093            List<Map<String, String>> nl = newMulti.get(key);
094            if (ol == null) {
095                ol = Collections.emptyList();
096            }
097            if (nl == null) {
098                nl = Collections.emptyList();
099            }
100
101            int max = Math.max(ol.size(), nl.size());
102
103            for (int i = 0; i < max; i++) {
104                Map<String, String> o = (i < ol.size()) ? ol.get(i) : null;
105                Map<String, String> n = (i < nl.size()) ? nl.get(i) : null;
106
107                if (o == null && n != null) {
108                    String rec = safe(n.get(RECORDING_NAME));
109                    String sq = n.containsKey(SQUARE_NUMBER) ? safe(n.get(SQUARE_NUMBER)) : "";
110                    diffs.add(new String[]{rec, sq, String.valueOf(i + 1), "", "", "", "Extra in NEW"});
111                    diffCount++;
112                    continue;
113                }
114                if (o != null && n == null) {
115                    String rec = safe(o.get(RECORDING_NAME));
116                    String sq = o.containsKey(SQUARE_NUMBER) ? safe(o.get(SQUARE_NUMBER)) : "";
117                    diffs.add(new String[]{rec, sq, String.valueOf(i + 1), "", "", "", "Missing in NEW"});
118                    diffCount++;
119                    continue;
120                }
121
122                Set<String> fields = new LinkedHashSet<>();
123                fields.addAll(o.keySet());
124                fields.addAll(n.keySet());
125
126                String rec = safe(o.get(RECORDING_NAME));
127                String sq = o.containsKey(SQUARE_NUMBER) ? safe(o.get(SQUARE_NUMBER)) : "";
128
129                for (String f : fields) {
130                    if (f == null || f.trim().isEmpty()) {
131                        continue;
132                    }
133                    if (RegressionRules.isIgnoredColumn(f, relaxedComparison)) {
134                        continue;
135                    }
136
137                    String ov = RegressionRules.clean(o.get(f));
138                    String nv = RegressionRules.clean(n.get(f));
139
140                    if (RegressionRules.isIgnoreCaseField(f)) {
141                        if (ov.equalsIgnoreCase(nv)) {
142                            continue;
143                        }
144                    }
145
146                    if (RegressionRules.emptyAndZeroEquiv(ov, nv)) {
147                        continue;
148                    }
149
150                    if (RegressionRules.isIgnoreCaseField(f)) {
151                        if (ov.equalsIgnoreCase(nv)) {
152                            continue;
153                        }
154                    }
155
156                    if (RegressionRules.valuesEqual(ov, nv)) {
157                        continue;
158                    }
159
160                    Double od = RegressionRules.parseDouble(ov);
161                    Double nd = RegressionRules.parseDouble(nv);
162
163                    // Skip if one numeric side is missing (strict + relaxed)
164                    if (RegressionRules.numericMissingSkipDifference(od, nd)) {
165                        continue;
166                    }
167
168                    if (od != null && nd != null) {
169                        nd = RegressionRules.correctedValueIfTrackDependent(f, od, nd, o, n);
170                        if (RegressionRules.numericEqualWithTolerance(f, od, nd, relaxedComparison)) {
171                            continue;
172                        }
173                    }
174
175                    String status = (od != null && nd != null)
176                            ? "NUMERIC DIFFERENCE"
177                            : "TEXT DIFFERENCE";
178
179                    diffs.add(new String[]{rec, sq, String.valueOf(i + 1), f, ov, nv, status});
180                    diffCount++;
181                }
182            }
183
184            processed++;
185            if (totalKeys <= 20) {
186                LOGGER.println("   ✓ Compared: " + key);
187            } else if (processed % 100 == 0) {
188                LOGGER.printf("   ...processed %d/%d keys%n", processed, totalKeys);
189            }
190        }
191
192        if (!diffs.isEmpty()) {
193            printGroupedDifferences(diffs);
194        } else {
195            LOGGER.println("\n✅ No differences detected.");
196        }
197
198        return diffCount;
199    }

204    private static void printGroupedDifferences(List<String[]> diffs) {
205        Map<String, Map<String, List<String[]>>> grouped =
206                new LinkedHashMap<>();
207
208        for (String[] row : diffs) {
209            String rec = row[0];
210            String sq = row[1].isEmpty() ? "—" : row[1];
211            Map<String, List<String[]>> perRec = grouped.get(rec);
212            if (perRec == null) {
213                perRec = new LinkedHashMap<>();
214                grouped.put(rec, perRec);
215            }
216            List<String[]> perSq = perRec.get(sq);
217            if (perSq == null) {
218                perSq = new ArrayList<>();
219                perRec.put(sq, perSq);
220            }
221            perSq.add(row);
222        }
223
224        int globalFieldWidth = 2;
225        for (String[] row : diffs) {
226            if (row[3] != null && row[3].length() + 2 > globalFieldWidth) {
227                globalFieldWidth = row[3].length() + 2;
228            }
229        }
230
231        LOGGER.println("\n🔎 Differences grouped by Square");
232        LOGGER.println("───────────────────────────────");
233
234        int total = 0;
235        Set<String> squaresWithDiffs = new TreeSet<>();
236
237        for (Map.Entry<String, Map<String, List<String[]>>> recEntry : grouped.entrySet()) {
238            LOGGER.println("Recording: " + recEntry.getKey());
239            for (Map.Entry<String, List<String[]>> sqEntry : recEntry.getValue().entrySet()) {
240                String sq = sqEntry.getKey();
241                if (!sq.equals("—")) {
242                    squaresWithDiffs.add(sq);
243                }
244
245                LOGGER.println("  ▫ Square " + sq + ":");
246                for (String[] row : sqEntry.getValue()) {
247                    LOGGER.printf(
248                            "     - %-" + globalFieldWidth + "s: '%s' vs '%s' (%s)%n",
249                            row[3], row[4], row[5], row[6]
250                    );
251                    total++;
252                }
253            }
254            LOGGER.println("");
255        }
256
257        LOGGER.printf("%n📊 Total differences listed: %d%n", total);
258        if (!squaresWithDiffs.isEmpty()) {
259            LOGGER.printf("🟧 Squares with at least one difference: %d (%s)%n",
260                          squaresWithDiffs.size(),
261                          join(squaresWithDiffs, ", "));
262        }
263    }

268    @SuppressWarnings("SameParameterValue")
269    private static String join(Set<String> s, String sep) {
270        StringBuilder sb = new StringBuilder();
271        for (String x : s) {
272            if (sb.length() > 0) {
273                sb.append(sep);
274            }
275            sb.append(x);
276        }
277        return sb.toString();
278    }

280    private static String buildKey(Map<String, String> r) {
281        String rec = safe(r.get(RECORDING_NAME));
282        String sq = r.containsKey(SQUARE_NUMBER) ? safe(r.get(SQUARE_NUMBER)) : "";
283        return sq.isEmpty() ? rec : rec + " - " + sq;
284    }

286    private static Map<String, List<Map<String, String>>> toMultiMap(List<Map<String, String>> rows) {
287        Map<String, List<Map<String, String>>> mm = new TreeMap<>();
288
289        for (Map<String, String> r : rows) {
290            String key = buildKey(r);
291            List<Map<String, String>> bucket = mm.get(key);
292            if (bucket == null) {
293                bucket = new ArrayList<>();
294                mm.put(key, bucket);
295            }
296            bucket.add(r);
297        }
298
299        return mm;
300    }

302    private static List<Map<String, String>> readCsv(Path path) throws IOException {
303        List<Map<String, String>> rows = new ArrayList<>();
304        BufferedReader br = Files.newBufferedReader(path);
305        try {
306            String headerLine = br.readLine();
307            if (headerLine == null) {
308                return rows;
309            }
310
311            String[] headers = headerLine.split(",", -1);
312            for (int i = 0; i < headers.length; i++) {
313                headers[i] = headers[i].trim();
314            }
315
316            String line;
317            while ((line = br.readLine()) != null) {
318                if (line.isEmpty()) {
319                    continue;
320                }
321
322                String[] vals = line.split(",", -1);
323                Map<String, String> row = new LinkedHashMap<>();
324
325                for (int i = 0; i < headers.length; i++) {
326                    String h = headers[i];
327                    String v = (i < vals.length ? vals[i] : "");
328                    row.put(h, v == null ? "" : v.trim());
329                }
330                rows.add(row);
331            }
332        } finally {
333            br.close();
334        }
335        return rows;
336    }

338    private static String safe(String s) {
339        return (s == null) ? "" : s;
340    }
