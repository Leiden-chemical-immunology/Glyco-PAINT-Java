001/*=============================================================================
002 *  Class:        GenerateSquaresProcessor.java
003 *  Package:      paint.generatesquares.calc
004 *
005 *  PURPOSE:
006 *    Orchestrates the full square-generation and per-recording analysis pipeline
007 *    for a PAINT experiment. Handles creation of geometric square grids, track
008 *    assignment, attribute calculation, and persistence of results.
009 *
010 *  DESCRIPTION:
011 *    Executes the GENERATE_SQUARES workflow:
012 *      • Loads experiment and recording data.
013 *      • Divides each recording into a regular grid of square regions based on
014 *        {@link paint.shared.config.GenerateSquaresConfig}.
015 *      • Assigns each track to the correct square using spatial filtering.
016 *      • Computes per-square and per-recording metrics using
017 *        {@link paint.generatesquares.calc.CalculateSquareAttributes}.
018 *      • Compiles complete experiment-level squares and tracks tables and writes
019 *        them to disk via {@link paint.shared.io.MainDataInterface}.
020 *
021 *  RESPONSIBILITIES:
022 *    • Generate geometric square grids for each recording.
023 *    • Assign tracks to the appropriate square regions.
024 *    • Compute square-level and recording-level attributes.
025 *    • Aggregate and persist all square, track, and recording tables.
026 *
027 *  USAGE EXAMPLE:
028 *    GenerateSquaresProcessor.generateSquaresForExperiment(
029 *        project,
030 *        "MyExperiment"
031 *    );
032 *
033 *  DEPENDENCIES:
034 *    – paint.shared.config.GenerateSquaresConfig
035 *    – paint.shared.config.paintconfig.PaintConfig
036 *    – paint.shared.objects.{Project, Experiment, Recording, Square, Track}
037 *    – paint.shared.utils.{PaintLogger, SharedSquareUtils}
038 *    – paint.generatesquares.calc.CalculateSquareAttributes
039 *    – paint.shared.io.{MainDataInterface, SquaresTableIO (static), TracksTableIO (static)}
040 *    – tech.tablesaw.api.Table
041 *

086    /**
087     * Processes an experiment to generate square regions for each recording, compute attributes,
088     * and compile data tables for all squares and tracks. The method applies geometric segmentation,
089     * assigns tracks to generated squares, and calculates both square-level and recording-level attributes.
090     * Finally, it writes compiled results to the file system.
091     *
092     * @param project        the project containing configurations and experiment data
093     * @param experimentName the name of the experiment to process
094     */
095    public static void generateSquaresForExperiment(Project project, String experimentName) throws IOException {
096        GenerateSquaresConfig generateSquaresConfig = project.getGenerateSquaresConfig();
097        Experiment            experiment;
098        List<Recording>       recordings;
099
100        LocalDateTime start = LocalDateTime.now();
101        PaintLogger.debugf("Loading Experiment '%s'", experimentName);
102
103        // Load the experiment (without squares, with tracks)
104        experiment = loadExperiment(
105                project.getProjectRootPath(),
106                experimentName,
107                false,   // Don't load Squares
108                true                // But do load Tracks
109        );
110
111        if (experiment == null) {
112            PaintLogger.errorf("Failed to load experiment: %s", experimentName);
113            return;
114        }
115
116        PaintLogger.infof("Starting processing experiment '%s'", experimentName);
117
118        recordings = experiment.getRecordings();
119        for (Recording recording : recordings) {
120
121            if (!recording.isProcessFlagSet()) {
122                continue;
123            }
124
125            // CHECK before starting each recording
126            if (Thread.currentThread().isInterrupted()) {
127                PaintLogger.infof("Cancelled before processing recording %s",
128                                  recording.getRecordingName());
129                return;
130            }
131
132            PaintLogger.infof("   Processing: %s", recording.getRecordingName());
133            PaintLogger.debugf(recording.toString());
134
135            // Create the squares with basic geometric information
136            List<Square> squares = generateSquaresForRecording(recording, generateSquaresConfig);
137            recording.setSquaresOfRecording(squares);
138
139            // Assign the recording tracks to the squares
140            assignTracksToSquares(recording, generateSquaresConfig);
141
142            // CHECK mid-work before calculating attributes
143            if (Thread.currentThread().isInterrupted()) {
144                PaintLogger.infof("Cancelled before attribute calculation for %s",
145                                  recording.getRecordingName());
146                return;
147            }
148
149            // Calculate square-level and recording-level attributes
150            Path experimentPath = project.getProjectRootPath()
151                                         .resolve(experiment.getExperimentName());
152            CalculateSquareAttributes.calculateSquareAttributes(
153                    experimentPath,
154                    recording,
155                    generateSquaresConfig
156            );
157            CalculateSquareAttributes.calculateRecordingAttributes(
158                    recording,
159                    generateSquaresConfig
160            );
161        }
162
163        Duration duration = Duration.between(start, LocalDateTime.now());
164        PaintLogger.infof("Finished processing experiment '%s' in %s",
165                          experimentName, formatDuration(duration));
166        PaintLogger.blankline();
167
168        // CHECK before writing output files
169        if (Thread.currentThread().isInterrupted()) {
170            PaintLogger.infof("Cancelled before writing output for %s", experimentName);
171            return;
172        }
173
174        // Compile all squares and write
175        Table allSquaresTable = compileAllSquares(experiment);
176        Path  experimentPath  = project.getProjectRootPath()
177                                       .resolve(experiment.getExperimentName());
178        writeSquares(experimentPath, allSquaresTable);
179
180        // Update recordings with filter information
181        for (Recording recording : experiment.getRecordings()) {
182            recording.setMinRequiredRSquared(generateSquaresConfig.getMinRequiredRSquared());
183            recording.setMaxAllowableVariability(generateSquaresConfig.getMaxAllowableVariability());
184            recording.setMinRequiredDensityRatio(generateSquaresConfig.getMinRequiredDensityRatio());
185            recording.setNeighbourMode(generateSquaresConfig.getNeighbourMode());
186        }
187
188        // Write recordings
189        writeRecordings(experimentPath, experiment.getRecordings());
190
191        // All tracks
192        writeTracks(experimentPath, compileAllTracks(experiment).sortOn(RECORDING_NAME, TRACK_ID));
193    }

195    /**
196     * Segments a single recording into square regions based on configuration.
197     *
198     * @param recording             the recording for which squares are to be generated
199     * @param generateSquaresConfig the configuration specifying the number of squares and related parameters
200     * @return a list of generated {@link Square} objects
201     */
202    public static List<Square> generateSquaresForRecording(Recording recording,
203            GenerateSquaresConfig generateSquaresConfig) {
204
205        // Total number of squares per recording.
206        int numberOfSquaresInRecording = generateSquaresConfig.getNumberOfSquaresInRecording();
207        int gridSize                   = (int) Math.sqrt(numberOfSquaresInRecording);
208
209        List<Square> squares      = new ArrayList<>();
210        double       squareWidth  = IMAGE_WIDTH  / gridSize;
211        double       squareHeight = IMAGE_HEIGHT / gridSize;
212
213        int squareNumber = 0;
214        for (int rowNumber = 0; rowNumber < gridSize; rowNumber++) {
215            for (int columnNumber = 0; columnNumber < gridSize; columnNumber++) {
216                double X0 = columnNumber * squareWidth;
217                double Y0 = rowNumber    * squareHeight;
218                double X1 = (columnNumber + 1) * squareWidth;
219                double Y1 = (rowNumber    + 1) * squareHeight;
220
221                squares.add(new Square(
222                        recording.getRecordingName() + '-' + squareNumber,
223                        recording.getExperimentName(),
224                        recording.getRecordingName(),
225                        squareNumber,
226                        rowNumber,
227                        columnNumber,
228                        X0,
229                        Y0,
230                        X1,
231                        Y1
232                ));
233
234                squareNumber++;
235            }
236        }
237        return squares;
238    }

240    /**
241     * Assigns tracks to the predefined square regions of a recording.
242     * It processes the tracks table of the recording, assigns each track to the relevant square,
243     * updates the square attributes, and compiles a complete tracks table for the recording.
244     *
245     * @param recording             the {@code Recording} instance containing track and square data.
246     *                              The method modifies this object by assigning tracks to the corresponding squares
247     *                              and updating their track-related attributes.
248     * @param generateSquaresConfig the configuration specifying the number of squares and related parameters
249     */
250    public static void assignTracksToSquares(Recording recording, GenerateSquaresConfig generateSquaresConfig) throws IOException {
251
252        Table tracksOfRecording   = recording.getTracksTable();
253        Table recordingTrackTable = newEmptyTrackTable();
254
255        int numberOfSquaresInRecording = generateSquaresConfig.getNumberOfSquaresInRecording();
256        int gridSize                   = (int) Math.sqrt(numberOfSquaresInRecording);
257
258        int incrementalTrackCount = 0;
259
260        PaintLogger.debugf("Assigning tracks to squares (%d total tracks)",
261                           tracksOfRecording.rowCount());
262
263        // ------------------------------------------------------------------
264        // 🔥 DEBUG CSV SETUP (only if flag enabled)
265        // ------------------------------------------------------------------
266        Path debugCsvPath = null;
267
268        if (debugGenerateSquaresForExperiment) {
269            Path debugDirPath = Paths.get(System.getProperty("user.home")).resolve("Downloads").resolve("Debug");
270            Files.createDirectories(debugDirPath);
271            debugCsvPath = debugDirPath.resolve("all_square_tracks.csv");
272
273            // Write header once
274            Files.write(
275                    debugCsvPath,
276                    "RecordingName,Square,TrackId,X,Y,Duration,MaxSpeed,MedianSpeed,Displacement,Row,Col\n".getBytes(),
277                    StandardOpenOption.CREATE
278            );
279        }
280        // ------------------------------------------------------------------
281
282        // We optimize track assignment by calculating the square index for each track directly.
283        // O(N_tracks) instead of O(N_squares * N_tracks).
284
285        List<Track> allTracks = trackTableToList(tracksOfRecording);
286        Map<Integer, List<Track>> tracksBySquare = new HashMap<>();
287
288        double squareWidth  = IMAGE_WIDTH  / gridSize;
289        double squareHeight = IMAGE_HEIGHT / gridSize;
290
291        for (Track track : allTracks) {
292            double tx = track.getTrackXLocation();
293            double ty = track.getTrackYLocation();
294
295            int col = (int) (tx / squareWidth);
296            int row = (int) (ty / squareHeight);
297
298            // Boundary handling (ensure it doesn't exceed grid dimensions)
299            if (col >= gridSize) {
300                col = gridSize - 1;
301            }
302            if (row >= gridSize) {
303                row = gridSize - 1;
304            }
305            if (col < 0) {
306                col = 0;
307            }
308            if (row < 0) {
309                row = 0;
310            }
311
312            int squareIndex = row * gridSize + col;
313            tracksBySquare.computeIfAbsent(squareIndex, k -> new ArrayList<>()).add(track);
314        }
315
316        int labelNumber = 0;
317        for (Square square : recording.getSquaresOfRecording()) {
318            int squareIndex = square.getSquareNumber();
319            List<Track> tracksInSquare = tracksBySquare.getOrDefault(squareIndex, Collections.emptyList());
320
321            incrementalTrackCount += tracksInSquare.size();
322
323            // Update the fields on each Track
324            for (Track track : tracksInSquare) {
325                track.setSquareNumber(squareIndex);
326                track.setLabelNumber(labelNumber);
327            }
328
329            Table updatedSquareTracks = trackListToTable(tracksInSquare);
330            recordingTrackTable.append(updatedSquareTracks);
331
332            // Update the square
333            square.setTracksList(tracksInSquare);
334            square.setTracksTable(updatedSquareTracks);
335            square.setNumberOfTracks(tracksInSquare.size());
336
337            // ------------------------------------------------------------------
338            // 🔥 DEBUG CSV APPEND (only when flag enabled)
339            // ------------------------------------------------------------------
340            if (debugGenerateSquaresForExperiment && debugCsvPath != null) {
341                StringBuilder sb = new StringBuilder();
342                for (Track track : tracksInSquare) {
343                    sb.append(String.format(
344                            "%s,%d,%s,%.3f,%.3f,%.3f,%.3f,%.3f,%.3f,%d,%d%n",
345                            square.getRecordingName(),
346                            squareIndex,
347                            track.getTrackId(),
348                            track.getTrackXLocation(),
349                            track.getTrackYLocation(),
350                            track.getTrackDuration(),
351                            track.getTrackMaxSpeed(),
352                            track.getTrackMedianSpeed(),
353                            track.getTrackDisplacement(),
354                            square.getRowNumber(),
355                            square.getColNumber()
356                    ));
357                }
358                Files.write(debugCsvPath, sb.toString().getBytes(), StandardOpenOption.APPEND);
359            }
360            // ------------------------------------------------------------------
361
362            labelNumber++;
363        }
364
365        recording.setTracksTable(recordingTrackTable);
366
367        PaintLogger.debugf("assignTracksToSquare - number of tracks assigned: %d; in recording: %d",
368                           incrementalTrackCount,
369                           tracksOfRecording.rowCount());
370
371        PaintLogger.debugf("✅ Total %d tracks assigned to %d squares.",
372                           recordingTrackTable.rowCount(),
373                           recording.getSquaresOfRecording().size());
374    }

377    /**
378     * Compiles all square data from the recordings in the specified experiment into a single table.
379     * The method iterates through each recording in the experiment, retrieves its square data,
380     * and appends it to a cumulative table. If a recording does not have square data available,
381     * an error is logged.
382     *
383     * @param experiment the experiment containing recordings whose square data is to be combined
384     * @return a {@code Table} containing the aggregated square data from all recordings in the experiment,
385     *         or an empty table if no square data exists
386     */
387    private static Table compileAllSquares(Experiment experiment) {
388        Table          allSquaresTable  = newEmptySquareTable();
389
390        for (Recording recording : experiment.getRecordings()) {
391            Table table = squareListToTable(recording.getSquaresOfRecording());
392            if (table != null) {
393                appendSquareTableInPlace(allSquaresTable, table);
394            } else {
395                PaintLogger.errorf("compileAllSquares - squares table does not exist for '%s'",
396                                   recording.getRecordingName());
397            }
398        }
399        return allSquaresTable;
400    }

402    /**
403     * Compiles all track data from the recordings in the specified experiment into a single table.
404     * The method iterates through each recording in the experiment, retrieves its track data,
405     * and appends it to an aggregate table. If a recording does not have track data available,
406     * an error is logged.
407     *
408     * @param experiment the experiment containing recordings whose track data is to be combined
409     * @return a {@code Table} containing the aggregated track data from all recordings in the experiment,
410     *         or an empty table if no track data exists
411     */
412    private static Table compileAllTracks(Experiment experiment) {
413        Table allTracksTable = newEmptyTrackTable();
414
415        for (Recording recording : experiment.getRecordings()) {
416            PaintLogger.debugf("Compiling tracks for experiment '%s' - recording '%s'",
417                               experiment.getExperimentName(),
418                               recording.getRecordingName());
419
420            Table table = recording.getTracksTable();
421            if (table != null) {
422                appendTrackTableInPlace(allTracksTable, table);
423            } else {
424                PaintLogger.errorf("compileAllTracks - tracks table does not exist for '%s'",
425                                   recording.getRecordingName());
426            }
427        }
428        PaintLogger.debugf("");
429        return allTracksTable;
430    }
