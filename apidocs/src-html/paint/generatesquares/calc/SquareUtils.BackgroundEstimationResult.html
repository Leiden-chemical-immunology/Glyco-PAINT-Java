046import paint.shared.objects.Square;

048import java.util.ArrayList;
049import java.util.Collections;
050import java.util.Comparator;
051import java.util.List;
052import java.util.stream.Collectors;

054public class SquareUtils {
055
056    /**
057     * Private constructor to prevent instantiation.
058     */
059    private SquareUtils() {
060    }
061
062    /**
063     * Estimates the background density from a list of squares. This is done
064     * by iteratively excluding squares whose track counts exceed a dynamically
065     * computed threshold:
066     * <p>
067     *      threshold = mean + 2 * standardDeviation
068     * <p>
069     * The mean is recalculated each iteration until:
070     *   • It stabilizes (relative change < EPSILON), or
071     *   • A maximum number of iterations is reached.
072     *
073     * @param squares List of squares containing track counts; must not be null or empty.
074     * @return BackgroundEstimationResult containing the estimated mean and the squares
075     *         considered part of the background.
076     */
077    public static BackgroundEstimationResult calculateBackgroundDensity(List<Square> squares) {
078        if (squares == null || squares.isEmpty()) {
079            return new BackgroundEstimationResult(Double.NaN, Collections.emptyList());
080        }
081
082        double mean = squares.stream()
083                             .mapToDouble(Square::getNumberOfTracks)
084                             .average()
085                             .orElse(Double.NaN);
086
087        if (Double.isNaN(mean) || mean == 0) {
088            return new BackgroundEstimationResult(mean, Collections.emptyList());
089        }
090
091        final double EPSILON  = 0.01;
092        final int    MAX_ITER = 10;
093
094        double prevMean;
095        List<Square> current = new ArrayList<>(squares);
096
097        for (int iter = 0; iter < MAX_ITER; iter++) {
098            prevMean = mean;
099            final double meanForLambda = mean;
100
101            double std = Math.sqrt(
102                    current.stream()
103                           .mapToDouble(square -> Math.pow(square.getNumberOfTracks() - meanForLambda, 2))
104                           .average()
105                           .orElse(0)
106            );
107
108            final double threshold = meanForLambda + 2 * std;
109
110            List<Square> filtered = new ArrayList<>();
111            for (Square square : current) {
112                if (square.getNumberOfTracks() <= threshold) {
113                    filtered.add(square);
114                }
115            }
116
117            if (filtered.isEmpty()) {
118                break;
119            }
120
121            mean = filtered.stream()
122                           .mapToDouble(Square::getNumberOfTracks)
123                           .average()
124                           .orElse(mean);
125
126            current = filtered;
127
128            // Stop when convergence is reached
129            if (Math.abs(mean - prevMean) / prevMean < EPSILON) {
130                break;
131            }
132        }
133
134        return new BackgroundEstimationResult(mean, current);
135    }
136
137    /**
138     * Holds the results of a background estimation: the mean background track count
139     * and the list of squares that were categorized as background.
140     */
141    public static class BackgroundEstimationResult {
142
143        private final double       numberOfTracksInBackgroundSquare;
144        private final List<Square> backgroundSquares;
145        private final int          numberOfBackgroundSquares;
146
147        /**
148         * Constructs an immutable result object.
149         *
150         * @param numberOfTracksInBackgroundSquare    estimated background track mean
151         * @param backgroundSquares squares identified as belonging to the background
152         */
153        public BackgroundEstimationResult(double numberOfTracksInBackgroundSquare, List<Square> backgroundSquares) {
154            this.numberOfTracksInBackgroundSquare = numberOfTracksInBackgroundSquare;
155            this.backgroundSquares                = backgroundSquares;
156            this.numberOfBackgroundSquares        = backgroundSquares.size();
157        }
158
159        /** @return mean background track count */
160        public double getNumberOfTracksInBackgroundSquare() {
161            return numberOfTracksInBackgroundSquare;
162        }
163
164        /** @return list of background squares */
165        public List<Square> getBackgroundSquares() {
166            return backgroundSquares;
167        }
168
169        /** @return the number of background squares */
170        public int getNumberOfBackgroundSquares() {
171            return numberOfBackgroundSquares;
172        }
173    }
174
175    /**
176     * Computes the average number of tracks among the smallest non-zero track counts
177     * in the recording. The number of squares to average over is configurable.
178     *
179     * @param squaresOfRecording      list of squares (must not be null)
180     * @param nrOfAverageCountSquares number of smallest non-zero values to average
181     * @return average of selected small non-zero track counts, or 0.0 if none found
182     */
183    public static double calcAverageTrackCountInBackgroundSquares(
184            List<Square> squaresOfRecording,
185            int          nrOfAverageCountSquares
186    ) {
187
188        List<Integer> trackCounts =
189                squaresOfRecording.stream()
190                                  .map(Square::getNumberOfTracks)
191                                  .sorted(Comparator.reverseOrder())
192                                  .collect(Collectors.toList());
193
194        double total = 0.0;
195        int    n     = 0;
196
197        // Walk backwards (smallest values first)
198        for (int i = trackCounts.size() - 1; i >= 0; i--) {
199            int value = trackCounts.get(i);
200            if (value > 0) {
201                total += value;
202                n++;
203                if (n >= nrOfAverageCountSquares) {
204                    break;
205                }
206            }
207        }
208
209        return n == 0 ? 0.0 : total / n;
210    }
