026    private static final int   MARGIN_LEFT   = 70;
027    private static final int   MARGIN_RIGHT  = 40;
028    private static final int   MARGIN_TOP    = 40;
029    private static final int   MARGIN_BOTTOM = 60;
030    private static final Color COLOR_DATA    = new Color(30, 100, 200);
031    private static final Color COLOR_FIT     = Color.RED;
032    private static final Color COLOR_AXES    = Color.GRAY;
033    private static final Color COLOR_SUCCESS = new Color(0, 128, 0);

040    /**
041     * Renders a Tau plot with the given data points, the fitted curve (if available),
042     * axes, labels, and annotation text.
043     *
044     * @param x         the x-coordinates of the data points
045     * @param y         the y-coordinates of the data points
046     * @param result    the Tau-fit result, including Tau and R²
047     * @param fitFailed true if the exponential fit failed
048     * @param width     image width in pixels
049     * @param height    image height in pixels
050     * @return          a BufferedImage containing the rendered plot
051     */
052    public static BufferedImage renderTauPlot(double[] x,
053            double[]                        y,
054            CalculateTau.CalculateTauResult result,
055            boolean                         fitFailed,
056            int                             width,
057            int                             height) {
058
059        // --- Initialize blank image and 2D context ---
060        BufferedImage img = new BufferedImage(width, height, BufferedImage.TYPE_INT_ARGB);
061        Graphics2D g2 = img.createGraphics();
062        g2.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
063
064        // --- Draw background ---
065        g2.setColor(Color.WHITE);
066        g2.fillRect(0, 0, width, height);
067
068        // --- Define margins and plotting area ---
069        int x0 = MARGIN_LEFT;
070        int y0 = height - MARGIN_BOTTOM;
071
072        // --- Determine axis ranges from data ---
073        double minX = Arrays.stream(x).min().orElse(0);
074        double maxX = Arrays.stream(x).max().orElse(1);
075        double minY = Arrays.stream(y).min().orElse(0);
076        double maxY = Arrays.stream(y).max().orElse(1);
077
078        // Add 5% padding on both axes so points stay within the frame
079        double padX = (maxX - minX) * 0.05;
080        double padY = (maxY - minY) * 0.05;
081        if (padX == 0) padX = 1.0;
082        if (padY == 0) padY = 1.0;
083
084        minX -= padX;
085        maxX += padX;
086        minY -= padY;
087        maxY += padY;
088
089        // --- Scaling factors ---
090        double xScale = (width - MARGIN_LEFT - MARGIN_RIGHT) / (maxX - minX);
091        double yScale = (height - MARGIN_TOP - MARGIN_BOTTOM) / (maxY - minY);
092
093        // --- Axes ---
094        g2.setColor(COLOR_AXES);
095        g2.drawLine(x0, y0, width - MARGIN_RIGHT, y0);
096        g2.drawLine(x0, y0, x0, MARGIN_TOP);
097
098        // --- Data points ---
099        g2.setColor(COLOR_DATA);
100        for (int i = 0; i < x.length; i++) {
101            int px = (int) (MARGIN_LEFT + (x[i] - minX) * xScale);
102            int py = (int) (y0 - (y[i] - minY) * yScale);
103            g2.fillOval(px - 3, py - 3, 6, 6);
104        }
105
106        // --- Fitted exponential curve ---
107        if (!fitFailed && result != null && Double.isFinite(result.getTau())) {
108            double tau = result.getTau();
109            double t = (tau > 0) ? 1000.0 / tau : Double.NaN;
110
111            double m = Arrays.stream(y).max().orElse(1);
112            double b = Arrays.stream(y).min().orElse(0);
113            g2.setColor(COLOR_FIT);
114
115            int    steps = 200;
116            double prevX = minX;
117            double prevY = m * Math.exp(-t * prevX) + b;
118            for (int i = 1; i <= steps; i++) {
119
120                double cx = minX + (maxX - minX) * i / steps;
121                double cy = m * Math.exp(-t * cx) + b;
122                int    x1 = (int) (MARGIN_LEFT + (prevX - minX) * xScale);
123                int    y1 = (int) (y0 - (prevY - minY) * yScale);
124                int    x2 = (int) (MARGIN_LEFT + (cx - minX) * xScale);
125                int    y2 = (int) (y0 - (cy - minY) * yScale);
126
127                g2.drawLine(x1, y1, x2, y2);
128                prevX = cx;
129                prevY = cy;
130            }
131        }
132
133        // --- Axis labels ---
134        g2.setColor(Color.BLACK);
135        g2.setFont(FONT_LABEL);
136        g2.drawString("Duration", width / 2 - 30, height - 20);
137
138        g2.rotate(-Math.PI / 2);
139        g2.drawString("Frequency", -height / 2 - 30, 20);
140        g2.rotate(Math.PI / 2);
141
142        // --- Fit annotation ---
143        g2.setColor(fitFailed ? Color.RED.darker() : COLOR_SUCCESS);
144        String msg = fitFailed
145                ? "Fit failed"
146                : String.format("Tau = %.1f ms, R² = %.3f", result.getTau(), result.getRSquared());
147        g2.drawString(msg, MARGIN_LEFT + 10, MARGIN_TOP + 20);
148
149        g2.dispose();
150        return img;
151    }

153    /**
154     * Exports a background-track histogram for each recording as a PNG file:
155     * <p>
156     *     Output/Background Plots/<recording>.png
157     * <p>
158     * @param experiment      the experiment containing the recordings
159     * @param experimentPath  the experiment root path
160     * @throws IOException    if image output fails
161     */
162    public static void exportBackgroundHistogramsToPngs(Experiment experiment, Path experimentPath) throws IOException {        Path outputDir = experimentPath.resolve("Output").resolve(BACKGROUND_PLOTS);
163        Files.createDirectories(outputDir);
164
165        for (Recording recording : experiment.getRecordings()) {
166            List<Square> squares = recording.getSquaresOfRecording();
167            if (squares == null || squares.isEmpty()) {
168                PaintLogger.debugf("Recording '%s': no squares%n", recording.getRecordingName());
169                continue;
170            }
171
172            SquareUtils.BackgroundEstimationResult backgroundResult          = calculateBackgroundDensity(squares);
173            Set<Square>                            backgroundSet             = new HashSet<>(backgroundResult.getBackgroundSquares());
174            double                                 backgroundTracksPerSquare = backgroundResult.getNumberOfTracksInBackgroundSquare();
175
176            int totalSquares          = squares.size();
177            int nBackground           = backgroundSet.size();
178            int totalTracks           = squares.stream().mapToInt(Square::getNumberOfTracks).sum();
179            int backgroundTracksTotal = backgroundSet.stream().mapToInt(Square::getNumberOfTracks).sum();
180
181            int maxTracks = squares.stream().mapToInt(Square::getNumberOfTracks).max().orElse(0);
182            int binSize   = Math.max(1, maxTracks / 20);
183            int binCount  = (maxTracks / binSize) + 1;
184
185            int[] allBins = new int[binCount];
186            int[] bgBins  = new int[binCount];
187
188            for (Square sq : squares) {
189                int n   = sq.getNumberOfTracks();
190                int bin = Math.min(n / binSize, binCount - 1);
191                allBins[bin]++;
192                if (backgroundSet.contains(sq)) {
193                    bgBins[bin]++;
194                }
195            }
196
197            int plotWidth  = 900;
198            int plotHeight = 600;
199            BufferedImage img = new BufferedImage(plotWidth, plotHeight, BufferedImage.TYPE_INT_ARGB);
200            Graphics2D g2 = img.createGraphics();
201
202            // Draw histogram into the graphics context
203            drawHistogram(
204                    g2,
205                    plotWidth,
206                    plotHeight,
207                    allBins,
208                    bgBins,
209                    binSize,
210                    recording.getRecordingName(),
211                    totalSquares,
212                    totalTracks,
213                    nBackground,
214                    backgroundTracksTotal,
215                    backgroundTracksPerSquare
216            );
217            g2.dispose();
218
219            Path outFile = outputDir.resolve(recording.getRecordingName() + ".png");
220            ImageIO.write(img, "png", outFile.toFile());
221            PaintLogger.debugf("Wrote background histogram to %s%n", outFile);
222        }
223    }

229    /**
230     * Saves a Tau-fit plot (successful or failed) for the given square.
231     * Output is stored as:
232     *     Output/Tau-Fitting Plots/Success/<...>.png
233     *     Output/Tau-Fitting Plots/Failed/<...>.png
234     * <p>
235     * The output directory tree is created automatically.
236     * <p>
237     * @param tracks         the tracks used in Tau fitting
238     * @param tauResult      the Tau fitting result (may indicate failure)
239     * @param experimentPath the root path of the experiment
240     * @param recordingName  the recording name
241     * @param squareIndex    the square index (used in file naming)
242     */
243    public static void saveTauFitPlot(
244            List<Track> tracks,
245            CalculateTau.CalculateTauResult tauResult,
246            Path experimentPath,
247            String recordingName,
248            int squareIndex
249    ) {
250        boolean fitFailed =
251                (tauResult == null) ||
252                        (tauResult.getStatus() != CalculateTau.CalculateTauResult.Status.TAU_SUCCESS) ||
253                        !Double.isFinite(tauResult.getTau()) ||
254                        !Double.isFinite(tauResult.getRSquared());
255
256        // Build frequency distribution: duration → count
257        Map<Double, Integer> freqMap = new TreeMap<>();
258        for (Track track : tracks) {
259            double d = track.getTrackDuration();
260            freqMap.put(d, freqMap.getOrDefault(d, 0) + 1);
261        }
262
263        double[] x = new double[freqMap.size()];
264        double[] y = new double[freqMap.size()];
265        int idx = 0;
266        for (Map.Entry<Double, Integer> e : freqMap.entrySet()) {
267            x[idx] = e.getKey();
268            y[idx] = e.getValue();
269            idx++;
270        }
271
272        // Render plot
273        BufferedImage img = renderTauPlot(x, y, tauResult, fitFailed, 900, 600);
274
275        // Determine output directory
276        Path rootDir   = experimentPath.resolve("Output").resolve(TAU_FITTING_PLOTS);
277        Path targetDir = rootDir.resolve(fitFailed ? "Failed" : "Success");
278
279        try {
280            Files.createDirectories(targetDir);
281            Path plotPath = targetDir.resolve(
282                    String.format("%s_square_%03d.png", recordingName, squareIndex)
283            );
284            ImageIO.write(img, "png", plotPath.toFile());
285        } catch (Exception e) {
286            // Log failure without throwing
287            PaintLogger.errorf(
288                    "Failed to save Tau-fit plot for '%s' square %03d: %s",
289                    recordingName,
290                    squareIndex,
291                    e.getMessage()
292            );
293        }
294    }

300    /**
301     * Draws a histogram into the given Graphics2D context. The histogram displays:
302     * <p>
303     *   • Track-count distribution across all squares
304     *   • Track-count distribution among "background" squares
305     *   • Summary statistics rendered as text in the upper-right corner
306     * <p>
307     * All layout calculations are pixel-accurate to ensure consistent rendering
308     * regardless of image size.
309     */
310    private static void drawHistogram(
311            Graphics2D g2,
312            int        w,
313            int        h,
314            int[]      allBins,
315            int[]      bgBins,
316            int        binSize,
317            String     title,
318            int        totalSquares,
319            int        totalTracks,
320            int        nBackground,
321            int        backgroundTracksTotal,
322            double     avgTracksInBackground
323    ) {
324        g2.setRenderingHint(
325                RenderingHints.KEY_ANTIALIASING,
326                RenderingHints.VALUE_ANTIALIAS_ON
327        );
328        g2.setColor(Color.WHITE);
329        g2.fillRect(0, 0, w, h);
330
331        int marginLeft   = 70;
332        int marginBottom = 50;
333        int marginTop    = 50;
334        int marginRight  = 40;
335
336        int binCount    = allBins.length;
337        int maxBinCount = Arrays.stream(allBins).max().orElse(1);
338
339        double xStep  = (double)(w - marginLeft - marginRight) / binCount;
340        double yScale = (double)(h - marginTop - marginBottom) / maxBinCount;
341
342        // Axes
343        g2.setColor(Color.GRAY);
344        int x0 = marginLeft;
345        int y0 = h - marginBottom;
346        g2.drawLine(x0, y0, w - marginRight, y0);
347        g2.drawLine(x0, y0, x0, marginTop);
348
349        // Tick marks
350        g2.setColor(Color.BLACK);
351        int nTicks = 5;
352        for (int i = 0; i <= nTicks; i++) {
353            int y = y0 - (i * (h - marginTop - marginBottom) / nTicks);
354            int value = (int)Math.round(i * (double)maxBinCount / nTicks);
355            g2.drawLine(x0 - 5, y, x0, y);
356            g2.drawString(String.valueOf(value), x0 - 45, y + 5);
357        }
358
359        // Bars
360        for (int i = 0; i < binCount; i++) {
361            int total      = allBins[i];
362            int background = bgBins[i];
363            int x          = (int)(x0 + i * xStep);
364            int barW       = (int)(xStep * 0.8);
365
366            if (background > 0) {
367                int bgHeight = (int)(background * yScale);
368                int bgY      = y0 - bgHeight;
369                g2.setColor(new Color(100, 160, 255, 180));
370                g2.fillRect(x, bgY, barW, bgHeight);
371            }
372
373            int fgHeight = (int)((total - background) * yScale);
374            if (fgHeight > 0) {
375                int fgY = y0 - fgHeight - (int)(background * yScale);
376                g2.setColor(new Color(180, 180, 180));
377                g2.fillRect(x, fgY, barW, fgHeight);
378            }
379
380            g2.setColor(Color.BLACK);
381            g2.drawRect(x, y0 - (int)(total * yScale), barW, (int)(total * yScale));
382        }
383
384        // Labels
385        g2.setFont(new Font("SansSerif", Font.PLAIN, 12));
386        g2.drawString(
387                "Track count bins (bin size ≈ " + binSize + ")",
388                w / 2 - 90,
389                h - 15
390        );
391        g2.drawString("Number of squares", 10, marginTop - 10);
392
393        g2.setFont(new Font("SansSerif", Font.BOLD, 16));
394        g2.drawString(
395                "Track Count Histogram – " + title,
396                w / 2 - 150,
397                marginTop - 20
398        );
399
400        // Right-side summary panel
401        g2.setFont(new Font("SansSerif", Font.BOLD, 14));
402        g2.setColor(new Color(0, 70, 180));
403
404        int textX = w - 400;
405        int textY = marginTop + 20;
406        int lineSpacing = 20;
407
408        String[] labels = new String[]{
409                "Number of squares in recording",
410                "Number of tracks in recording",
411                "Number of background squares",
412                "Number of tracks in background",
413                "Average number of tracks in the background"
414        };
415        String[] values = new String[]{
416                String.valueOf(totalSquares),
417                String.valueOf(totalTracks),
418                String.valueOf(nBackground),
419                String.valueOf(backgroundTracksTotal),
420                String.format("%.3f", avgTracksInBackground)
421        };
422
423        FontMetrics fm = g2.getFontMetrics();
424        int maxLabelW = 0;
425        for (String label : labels) {
426            maxLabelW = Math.max(maxLabelW, fm.stringWidth(label));
427        }
428
429        // Column positions
430        int colonPad = 10;                  // space between label end and colon
431        int valuePad = 10;                  // space between colon and value
432        int colonX = textX + maxLabelW + colonPad;
433        int valueX = colonX + fm.stringWidth(":") + valuePad;
434
435        for (int i = 0; i < labels.length; i++) {
436            int y = textY + i * lineSpacing;
437            g2.drawString(labels[i], textX, y);
438            g2.drawString(":", colonX, y);
439            g2.drawString(values[i], valueX, y);
440        }
441    }
