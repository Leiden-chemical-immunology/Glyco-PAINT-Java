074    /**
075     * Calculates attributes for every square in a recording, including Tau,
076     * density, variability, density ratio, and various kinematic statistics.
077     * Visibility filtering is performed based on configuration parameters,
078     * followed by label number assignment for visible squares.
079     *
080     * @param experimentPath        the experiment path (used for plot output)
081     * @param recording             the recording containing track and square data
082     * @param generateSquaresConfig configuration parameters for the calculation
083     */
084    public static void calculateSquareAttributes(Path experimentPath,
085            Recording recording,
086            GenerateSquaresConfig generateSquaresConfig) {
087        double                     minRequiredRSquared              = generateSquaresConfig.getMinRequiredRSquared();
088        int                        minNumberOfTracksToCalculate     = generateSquaresConfig.getMinTracksToCalculate();
089        int                        minNumberOfTracksToCalculateTau  = generateSquaresConfig.getMinTracksToCalculateTau();
090        double                     maxAllowableVariability          = generateSquaresConfig.getMaxAllowableVariability();
091        double                     minRequiredDensityRatio          = generateSquaresConfig.getMinRequiredDensityRatio();
092        String                     neighbourMode                    = generateSquaresConfig.getNeighbourMode();
093        int                        numberOfSquaresInRecording       = generateSquaresConfig.getNumberOfSquaresInRecording();
094        double                     squareArea                       = calculateSquareArea(numberOfSquaresInRecording);    // Here we look at the single square
095        double                     concentration                    = recording.getConcentration();
096        List<Square>               squaresOfRecording               = recording.getSquaresOfRecording();
097        boolean                    showTauFittingPlots              = PaintConfig.getBoolean(GENERATE_SQUARES, TAU_FITTING_PLOTS, false);
098        BackgroundEstimationResult bgResult                         = calculateBackgroundDensity(squaresOfRecording);
099        CalcParams                 params                           = new CalcParams(recording, generateSquaresConfig, bgResult);
100
101        PaintLogger.debugf("Estimated Background track count = %.2f, number of background squares = %d%n",
102                           params.numberOfTracksInBackgroundSquare, bgResult.getNumberOfBackgroundSquares());
103
104        for (Square square : squaresOfRecording) {
105            calculateAttributesForSquare(square, experimentPath, recording, generateSquaresConfig, params);
106        }
107
108        // --- Apply visibility filters ---
109        applyVisibilityFilter(squaresOfRecording,
110                              params.minRequiredDensityRatio,
111                              params.maxAllowableVariability,
112                              params.minRequiredRSquared,
113                              params.neighbourMode);
114
115        // --- Assign label numbers to visible squares ---
116        int labelNumber = 0;
117        for (Square sq : squaresOfRecording) {
118            if (sq.isVisible()) {
119                sq.setLabelNumber(labelNumber++);
120            }
121        }
122    }

124    /**
125     * Container for calculation parameters to reduce argument counts.
126     */
127    private static class CalcParams {
128        final double  minRequiredRSquared;
129        final int     minNumberOfTracksToCalculate;
130        final int     minNumberOfTracksToCalculateTau;
131        final double  maxAllowableVariability;
132        final double  minRequiredDensityRatio;
133        final String  neighbourMode;
134        final int     numberOfSquaresInRecording;
135        final double  squareArea;
136        final double  concentration;
137        final boolean showTauFittingPlots;
138        final double  numberOfTracksInBackgroundSquare;
139        final double  backgroundTracksOri;
140
141        CalcParams(Recording recording, GenerateSquaresConfig config, BackgroundEstimationResult bgResult) {
142            this.minRequiredRSquared              = config.getMinRequiredRSquared();
143            this.minNumberOfTracksToCalculate     = config.getMinTracksToCalculate();
144            this.minNumberOfTracksToCalculateTau  = config.getMinTracksToCalculateTau();
145            this.maxAllowableVariability          = config.getMaxAllowableVariability();
146            this.minRequiredDensityRatio          = config.getMinRequiredDensityRatio();
147            this.neighbourMode                    = config.getNeighbourMode();
148            this.numberOfSquaresInRecording       = config.getNumberOfSquaresInRecording();
149            this.squareArea                       = calculateSquareArea(numberOfSquaresInRecording);
150            this.concentration                    = recording.getConcentration();
151            this.showTauFittingPlots              = PaintConfig.getBoolean(GENERATE_SQUARES, TAU_FITTING_PLOTS, false);
152            this.numberOfTracksInBackgroundSquare = bgResult.getNumberOfTracksInBackgroundSquare();
153            this.backgroundTracksOri              = calcAverageTrackCountInBackgroundSquares(recording.getSquaresOfRecording(),
154                                                                                            (int) (0.1 * numberOfSquaresInRecording));
155        }
156    }

158    private static void calculateAttributesForSquare(Square square, Path experimentPath, Recording recording,
159                                                     GenerateSquaresConfig config, CalcParams params) {
160
161        List<Track> tracksInSquare = square.getTracks();
162        if (tracksInSquare == null || tracksInSquare.isEmpty()) {
163            return;
164        }
165
166        Table table = square.getTracksTable();
167        if (table.rowCount() == 0) {
168            return;
169        }
170
171        // Do not calculate attributes if not enough tracks
172        if (tracksInSquare.size() < params.minNumberOfTracksToCalculate) {
173            square.resetCalculatedAttributes();
174            return;
175        }
176
177        int squareNumber = square.getSquareNumber();
178
179        // --- Tau fitting ---
180        if (tracksInSquare.size() >= params.minNumberOfTracksToCalculateTau) {
181            CalculateTau.CalculateTauResult results = calculateTau(tracksInSquare, params.minRequiredRSquared);
182
183            if (params.showTauFittingPlots) {
184                saveTauFitPlot(tracksInSquare, results, experimentPath,
185                               recording.getRecordingName(), squareNumber);
186            }
187
188            if (results.getStatus() == CalculateTau.CalculateTauResult.Status.TAU_SUCCESS) {
189                square.setTau(round(results.getTau(), 0));
190                square.setRSquared(round(results.getRSquared(), 3));
191            } else {
192                square.setTau(Double.NaN);
193                square.setRSquared(Double.NaN);
194            }
195        } else {
196            square.setTau(Double.NaN);
197            square.setRSquared(Double.NaN);
198        }
199
200        // --- Variability, density, kinematic metrics ---
201
202        // Note that double values are written with 3 decimals in Squares
203
204        final int PRECISION_L  = 2;
205        final int PRECISION_M1 = 3;
206        final int PRECISION_M2 = 4;
207        final int PRECISION_H  = 5;
208
209        square.setVariability(round(calculateVariability(table, squareNumber, params.numberOfSquaresInRecording, 10),       PRECISION_M1));
210        square.setDensity(round(calculateDensity(tracksInSquare.size(), params.squareArea, RECORDING_DURATION, params.concentration), PRECISION_H));
211        square.setDensityRatio(round(calculateDensityRatio(tracksInSquare.size(), params.numberOfTracksInBackgroundSquare),           PRECISION_M1));
212        square.setDensityRatioOri(round(calculateDensityRatio(tracksInSquare.size(), params.backgroundTracksOri),                     PRECISION_M1));
213
214        square.setMedianDiffusionCoefficient(round(table.doubleColumn(DIFFUSION_COEFFICIENT).median(),        PRECISION_M2));
215        square.setMedianDiffusionCoefficientExt(round(table.doubleColumn(DIFFUSION_COEFFICIENT_EXT).median(), PRECISION_M1));
216
217        square.setMedianDisplacement(round(table.doubleColumn(TRACK_DISPLACEMENT).median(), PRECISION_L));
218        square.setMaxDisplacement(round(table.doubleColumn(TRACK_DISPLACEMENT).max(),       PRECISION_L));
219        square.setTotalDisplacement(round(table.doubleColumn(TRACK_DISPLACEMENT).sum(),     PRECISION_L));
220
221        square.setMedianMaxSpeed(round(table.doubleColumn(TRACK_MAX_SPEED).median(),        PRECISION_L));
222        square.setMaxMaxSpeed(round(table.doubleColumn(TRACK_MAX_SPEED).max(),              PRECISION_L));
223
224        square.setMedianMedianSpeed(round(table.doubleColumn(TRACK_MEDIAN_SPEED).median(),  PRECISION_L));
225        square.setMaxMedianSpeed(round(table.doubleColumn(TRACK_MEDIAN_SPEED).max(),        PRECISION_L));
226
227        square.setMaxTrackDuration(round(table.doubleColumn(TRACK_DURATION).max(),          PRECISION_L));
228        square.setTotalTrackDuration(round(table.doubleColumn(TRACK_DURATION).sum(),        PRECISION_L));
229        square.setMedianTrackDuration(round(table.doubleColumn(TRACK_DURATION).median(),    PRECISION_L));
230    }

232    /**
233     * Computes recording-level attributes, including Tau, R², background statistics,
234     * and density estimates based on selected (visible) squares.
235     *
236     * @param recording             the recording being processed
237     * @param generateSquaresConfig configuration parameters
238     */
239    public static void calculateRecordingAttributes(Recording recording,
240            GenerateSquaresConfig generateSquaresConfig) {
241
242        double minRequiredRSquared = generateSquaresConfig.getMinRequiredRSquared();
243
244        List<Square>               squaresOfRecording   = recording.getSquaresOfRecording();
245        BackgroundEstimationResult bgResult             = calculateBackgroundDensity(squaresOfRecording);
246        double                     meanBackgroundTracks = bgResult.getNumberOfTracksInBackgroundSquare();
247        int                        backgroundTracks     = bgResult.getBackgroundSquares()
248                                                                  .stream()
249                                                                  .mapToInt(Square::getNumberOfTracks)
250                                                                  .sum();
251
252        recording.setNumberOfSquaresInBackground(bgResult.getBackgroundSquares().size());
253        recording.setNumberOfTracksInBackground(backgroundTracks);
254        recording.setAverageTracksInBackGround(round(meanBackgroundTracks, 3));
255
256        List<Track> selectedTracks = getTracksFromSelectedSquares(squaresOfRecording);
257
258        if (selectedTracks.isEmpty()) {
259            recording.setTau(Double.NaN);
260            recording.setRSquared(Double.NaN);
261            recording.setDensity(0.0);
262            return;
263        }
264
265        // --- Tau for the whole recording ---
266        CalculateTau.CalculateTauResult tauResult = calculateTau(selectedTracks, minRequiredRSquared);
267
268        if (tauResult.getStatus() == CalculateTau.CalculateTauResult.Status.TAU_SUCCESS) {
269            recording.setTau(round(tauResult.getTau(), 0));
270            recording.setRSquared(round(tauResult.getRSquared(), 3));
271        } else {
272            recording.setTau(Double.NaN);
273            recording.setRSquared(Double.NaN);
274        }
275
276        // --- Recording density metrics ---
277        int totalTracks = selectedTracks.size();
278
279        int    nSelectedSquares = getNumberOfSelectedSquares(recording);
280        double area             = calculateSquareArea(generateSquaresConfig.getNumberOfSquaresInRecording()) * nSelectedSquares;
281        double concentration    = recording.getConcentration();
282
283        recording.setDensity(round(calculateDensity(totalTracks, area, RECORDING_DURATION, concentration), 3));
284    }

286    /**
287     * Returns the density ratio between a square and the background track density.
288     */
289    public static double calculateDensityRatio(int numberOfTracksInSquare,
290            double numberOfTracksInBackgroundSquare) {
291        return (numberOfTracksInBackgroundSquare == 0)
292                ? 0
293                : numberOfTracksInSquare / numberOfTracksInBackgroundSquare;
294    }

296    /**
297     * Computes spatial variability of track positions within a square.
298     * Variability is defined as the coefficient of variation (σ / μ) over
299     * grid-cell track counts within a `granularity × granularity` subdivision
300     * of the square.
301     */
302    public static double calculateVariability(Table tracks,
303            int squareNumber,
304            int numberOfSquaresInRecording,
305            int granularity) {
306
307        int[][] matrix    = new int[granularity][granularity];
308        int     dimension = (int) Math.sqrt(numberOfSquaresInRecording);
309        double  width     = IMAGE_WIDTH / dimension;
310        double  height    = IMAGE_WIDTH / dimension;
311
312        // Access the columns once
313        DoubleColumn xCol = tracks.doubleColumn(TRACK_X_LOCATION);
314        DoubleColumn yCol = tracks.doubleColumn(TRACK_Y_LOCATION);
315
316        for (int i = 0; i < tracks.rowCount(); i++) {
317            double x = xCol.get(i);  // The x-coordinate of the track
318            double y = yCol.get(i);  // The y-coordinate of the track
319
320            int[] indices = getIndices(x, y, width, height, squareNumber, dimension, granularity);
321            int   xi      = indices[0];
322            int   yi      = indices[1];
323
324            if (xi >= 0 && xi < granularity && yi >= 0 && yi < granularity) {
325                matrix[yi][xi]++;
326            }
327        }
328
329        // Flatten matrix into an 1D array for stats
330        int totalCells = granularity * granularity;
331        double[] values = new double[totalCells];
332
333        int idx = 0;
334        for (int r = 0; r < granularity; r++) {
335            for (int c = 0; c < granularity; c++) {
336                values[idx++] = matrix[r][c];
337            }
338        }
339
340        double mean = mean(values);
341        if (mean == 0) {
342            return 0.0;
343        }
344
345        double std = std(values, mean);
346        return std / mean;
347    }
348
349    /**
350     * Computes the mean of an array of doubles.
351     */
352    private static double mean(double[] values) {
353        double sum = 0.0;
354        for (double value : values) {
355            sum += value;
356        }
357        return sum / values.length;
358    }
359
360    /**
361     * Computes the population standard deviation of an array of doubles.
362     */
363    private static double std(double[] values, double mean) {
364        double sumSq = 0.0;
365        for (double value : values) {
366            double diff = value - mean;
367            sumSq += diff * diff;
368        }
369        return Math.sqrt(sumSq / values.length);
370    }

372    /**
373     * Calculates the grid indices for a point within a specific square of a recording.
374     * The method determines the location of the point in a finer grid inside the square,
375     * based on the specified granularity.
376     *
377     * @param x1                   the x-coordinate of the point in the global coordinate system
378     * @param y1                   the y-coordinate of the point in the global coordinate system
379     * @param width                the width of each square in the grid
380     * @param height               the height of each square in the grid
381     * @param squareSequenceNumber the sequence number of the square in the grid
382     * @param numberOfSquaresInRow the total number of squares in a single row of the grid
383     * @param granularity          the number of subdivisions along one axis within a square
384     * @return an array of two integers, where the first value is the x-index (column index)
385     * and the second value is the y-index (row index) of the point in the finer grid
386     */
387    private static int[] getIndices(double x1,
388            double y1,
389            double width,
390            double height,
391            int    squareSequenceNumber,
392            int    numberOfSquaresInRow,
393            int    granularity) {
394
395        double x0 = (squareSequenceNumber % numberOfSquaresInRow) * width;
396        double y0 = (squareSequenceNumber / numberOfSquaresInRow) * height;     // Integer division is intended
397
398        int xi = (int) (((x1 - x0) / width)  * granularity);
399        int yi = (int) (((y1 - y0) / height) * granularity);
400
401        return new int[]{xi, yi};
402    }
