018 *    • Singleton-based access to application-wide configuration.
019 *    • Section-based organization of settings.
020 *    • Automatic default value population via {@link DefaultConfigLoader}.
021 *    • Fault-tolerant loading with automatic backup of invalid files.
022 *    • Case-insensitive key retrieval.

067    private PaintConfig(Path path) {

077    /** Initialize for a given project root (creates file if missing). */
078    public static void initialise(Path projectPath) {
079        Path configPath = projectPath.resolve(PAINT_CONFIGURATION_JSON);
080        if (INSTANCE == null) {
081            synchronized (PaintConfig.class) {
082                if (INSTANCE == null) {
083                    INSTANCE = new PaintConfig(configPath);
084                    INSTANCE.store.ensureLoaded(() -> DefaultConfigLoader.loadDefaults(INSTANCE.store));
085                }
086            }
087        } else {
088            Path expected = projectPath.resolve(PAINT_CONFIGURATION_JSON);
089            if (!INSTANCE.path.equals(expected)) {
090                PaintLogger.warnf("PaintConfig already initialised at %s (attempted reinit with %s)\n",
091                                  INSTANCE.path, expected);
092            }
093        }
094    }
095
096    /** Force reinitialisation to a new project root. */
097    public static void reinitialise(Path projectPath) {
098        synchronized (PaintConfig.class) {
099            INSTANCE = null;
100            initialise(projectPath);
101            PaintLogger.debugf("PaintConfig reinitialised at %s",
102                               projectPath.resolve(PAINT_CONFIGURATION_JSON));
103        }
104    }
105
106    /** Access singleton (lazy-inits to user-home if needed). */
107    public static PaintConfig instance() {
108        if (INSTANCE == null) {
109            Path defaultPath = Paths.get(System.getProperty("user.home"), PAINT_CONFIGURATION_JSON);
110            initialise(defaultPath);
111        }
112        return INSTANCE;
113    }

135    public String getStringValue(String section, String key, String def) {
136        JsonObject sec = getSection(section);
137        if (sec != null) {
138            String real = jsonCase.findKeyIgnoreCase(sec, key);
139            if (real != null && sec.get(real).isJsonPrimitive()) {
140                return sec.getAsJsonPrimitive(real).getAsString();
141            }
142        }
143        PaintLogger.warnf("No value for '%s' found, default '%s' applied", key, def);
144        setStringValue(section, key, def, true);
145        return def;
146    }
147
148    public int getIntValue(String section, String key, int def) {
149        JsonObject sec = getSection(section);
150        if (sec != null) {
151            String real = jsonCase.findKeyIgnoreCase(sec, key);
152            if (real != null && sec.get(real).isJsonPrimitive()) {
153                try {
154                    return sec.getAsJsonPrimitive(real).getAsInt();
155                } catch (Exception ignored) {
156                    PaintLogger.warnf("Invalid '%s', default %d applied", real, def);
157                    setIntValue(section, key, def, true);
158                }
159            }
160        }
161        PaintLogger.warnf("No value for '%s', default %d applied", key, def);
162        setIntValue(section, key, def, true);
163        return def;
164    }
165
166    public double getDoubleValue(String section, String key, double def) {
167        JsonObject sec = getSection(section);
168        if (sec != null) {
169            String real = jsonCase.findKeyIgnoreCase(sec, key);
170            if (real != null && sec.get(real).isJsonPrimitive()) {
171                try {
172                    return sec.getAsJsonPrimitive(real).getAsDouble();
173                } catch (Exception ignored) {
174                    PaintLogger.warnf("Invalid '%s', default %.2f applied", real, def);
175                    setDoubleValue(section, key, def, true);
176                }
177            }
178        }
179        PaintLogger.warnf("No value for '%s', default %.2f applied", key, def);
180        setDoubleValue(section, key, def, true);
181        return def;
182    }
183
184    public boolean getBooleanValue(String section, String key, boolean def) {
185        JsonObject sec = getSection(section);
186        if (sec != null) {
187            String real = jsonCase.findKeyIgnoreCase(sec, key);
188            if (real != null && sec.get(real).isJsonPrimitive()) {
189                try {
190                    return sec.getAsJsonPrimitive(real).getAsBoolean();
191                } catch (Exception ignored) { /* fallthrough */ }
192            }
193        }
194        PaintLogger.warnf("No value for '%s', default %b applied", key, def);
195        setBooleanValue(section, key, def, true);
196        return def;
197    }
198
199    public boolean getBooleanValueNoWarning(String section, String key, boolean def) {
200        JsonObject sec = getSection(section);
201        if (sec != null) {
202            String real = jsonCase.findKeyIgnoreCase(sec, key);
203            if (real != null && sec.get(real).isJsonPrimitive()) {
204                try {
205                    return sec.getAsJsonPrimitive(real).getAsBoolean();
206                } catch (Exception ignored) { /* fallthrough */ }
207            }
208        }
209        setBooleanValue(section, key, def, true);
210        return def;
211    }

213    public void setStringValue(String section, String key, String value, boolean autoSave) {
214        JsonObject sec = store.getOrCreateSection(section);
215        sec.addProperty(key, value);
216        if (autoSave) {
217            save();
218        }
219    }

244
245    @Override public String toString() {
246        return GSON.toJson(store.root());
247    }
248
249    public void removeValue(String section, String key, boolean autoSave) {
250        JsonObject sec = getSection(section);
251        if (sec != null) {
252            String real = jsonCase.findKeyIgnoreCase(sec, key);
253            if (real != null) {
254                sec.remove(real);
255                if (autoSave) save();
256            }
257        }
258    }
259
260    public void removeSectionValue(String section, boolean autoSave) {
261        store.removeSection(section);
262        if (autoSave) {
263            save();
264        }
265    }
266
267    public static void remove (String s, String k) {
268        instance().removeValue (s, k, true);
269    }
270
271    public static void removeSection(String s) {
272        instance().removeSectionValue(s, true);
273    }
