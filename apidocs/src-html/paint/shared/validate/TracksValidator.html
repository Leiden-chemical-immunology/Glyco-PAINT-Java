033package paint.shared.validate;
034
035import paint.shared.objects.Track;
036import tech.tablesaw.api.ColumnType;
037
038import java.util.ArrayList;
039import java.util.List;
040
041/**
042 * Validates the schema of {@code tracks.csv} by checking header correctness
043 * and column data types according to {@link Track.Column}.
044 *
045 * <p>This version no longer depends on TrackSchema; it uses the embedded
046 * schema directly inside the Track class.</p>
047 */
048public final class TracksValidator extends AbstractFileValidator {
049
050    // ───────────────────────────────────────────────────────────────────────────────
051    // HEADER VALIDATION
052    // ───────────────────────────────────────────────────────────────────────────────
053
054    /**
055     * Validates that the actual CSV headers match the expected headers defined in
056     * {@link Track.Column}.
057     *
058     * @param actualHeader the list of headers found in the CSV file
059     * @param result       the {@link ValidationResult} to record any mismatches
060     */
061    @Override
062    protected void validateHeader(List<String> actualHeader, ValidationResult result) {
063
064        // Build expected header from Track.Column enum
065        List<String> expected = new ArrayList<>();
066        for (Track.Column col : Track.Column.values()) {
067            expected.add(col.header);
068        }
069
070        headersMatch(expected, actualHeader, result);
071    }
072
073    // ───────────────────────────────────────────────────────────────────────────────
074    // TYPE VALIDATION
075    // ───────────────────────────────────────────────────────────────────────────────
076
077    /**
078     * Returns the expected {@link ColumnType} for each column in the tracks CSV,
079     * as defined in {@link Track.Column}.
080     *
081     * @return an array of expected column types
082     */
083    @Override
084    protected ColumnType[] getExpectedTypes() {
085
086        Track.Column[] cols = Track.Column.values();
087        ColumnType[]  types = new ColumnType[cols.length];
088
089        for (int i = 0; i < cols.length; i++) {
090            types[i] = cols[i].type;
091        }
092
093        return types;
094    }
095}
