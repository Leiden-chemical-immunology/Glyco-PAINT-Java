056/**
057 * Represents all metadata and associated data objects for a single experiment
058 * recording in the PAINT analysis framework.
059 *
060 * <p>
061 * This class contains experiment-level descriptive metadata as well as
062 * references to the square and track objects produced during analysis.
063 * </p>
064 *
065 * <p>
066 * The schema for ExperimentInfo is embedded directly in {@link Column}, which
067 * binds human-readable CSV headers to their corresponding Tablesaw types.
068 * Table I/O classes use this enum for all schema operations.
069 * </p>
070 */
071public class ExperimentInfo {
072
073    //=========================================================================
074    //  CORE FIELDS
075    //=========================================================================
076
077    private String  experimentName;
078    private String  recordingName;
079    private int     conditionNumber;
080    private int     replicateNumber;
081    private String  probeName;
082    private String  probeType;
083    private String  cellType;
084    private String  adjuvant;
085    private double  concentration;
086    private boolean processFlag;
087    private double  threshold;
088
089    /** Associated objects for downstream analysis. */
090    private final List<Square> squares = new ArrayList<>();
091    private final List<Track>  tracks  = new ArrayList<>();
092
093    /**
094     * Default empty constructor.
095     */
096    public ExperimentInfo() {
097    }
098
099    //=========================================================================
100    //  CONSTRUCTORS
101    //=========================================================================
102
103    /**
104     * Constructs an ExperimentInfo instance from a row of key-value pairs
105     * (typically parsed from experiment_info.csv).
106     *
107     * @param row the map of header → text value
108     */
109    public ExperimentInfo(Map<String, String> row) {
110        try {
111            this.experimentName  = row.get(EXPERIMENT_NAME);
112            this.recordingName   = row.get(RECORDING_NAME);
113            this.conditionNumber = parseInt(row.get(CONDITION_NUMBER));
114            this.replicateNumber = parseInt(row.get(REPLICATE_NUMBER));
115            this.probeName       = row.get(PROBE_NAME);
116            this.probeType       = row.get(PROBE_TYPE);
117            this.cellType        = row.get(CELL_TYPE);
118            this.adjuvant        = row.get(ADJUVANT);
119            this.concentration   = parseDouble(row.get(CONCENTRATION));
120            this.processFlag     = isBooleanTrue(row.get(PROCESS_FLAG));
121            this.threshold       = parseDouble(row.get(THRESHOLD));
122        }
123        catch (Exception e) {
124            PaintLogger.errorf("Problem parsing Experiment Info");
125            PaintLogger.errorf(row.toString());
126
127            StringWriter sw = new StringWriter();
128            e.printStackTrace(new PrintWriter(sw));
129            PaintLogger.errorf("Exception details:\n" + sw);
130        }
131    }
132
133    /**
134     * @return the name of the experiment.
135     */
136    public String getExperimentName() {
137        return experimentName;
138    }

291
292    /**
293     * Returns a human-readable summary of this ExperimentInfo, useful
294     * for diagnostic logs and console output.
295     */
296    @Override
297    public String toString() {
298
299        return "\tExperiment Info\n" +
300                String.format("\t  Experiment Name    : %s%n", experimentName) +
301                String.format("\t  Recording Name     : %s%n", recordingName) +
302                String.format("\t  Condition Nr       : %d%n", conditionNumber) +
303                String.format("\t  Replicate Nr       : %d%n", replicateNumber) +
304                String.format("\t  Probe Name         : %s%n", probeName) +
305                String.format("\t  Probe Type         : %s%n", probeType) +
306                String.format("\t  Cell Type          : %s%n", cellType) +
307                String.format("\t  Adjuvant           : %s%n", adjuvant) +
308                String.format("\t  Concentration      : %.2f%n", concentration) +
309                String.format("\t  Threshold          : %.2f%n", threshold) +
310                String.format("\t  Track Count        : %d%n", tracks.size()) +
311                String.format("\t  Square Count       : %d%n", squares.size());
312    }

314    //=========================================================================
315    // ENUM
316    //=========================================================================
317
318    public enum Column {
319
320        EXPERIMENT_NAME(  "Experiment Name",  ColumnType.STRING),
321        RECORDING_NAME(   "Recording Name",   ColumnType.STRING),
322        CONDITION_NUMBER( "Condition Number", ColumnType.INTEGER),
323        REPLICATE_NUMBER( "Replicate Number", ColumnType.INTEGER),
324        PROBE_NAME(       "Probe Name",       ColumnType.STRING),
325        PROBE_TYPE(       "Probe Type",       ColumnType.STRING),
326        CELL_TYPE(        "Cell Type",        ColumnType.STRING),
327        ADJUVANT(         "Adjuvant",         ColumnType.STRING),
328        CONCENTRATION(    "Concentration",    ColumnType.DOUBLE),
329        PROCESS_FLAG(     "Process Flag",     ColumnType.BOOLEAN),
330        THRESHOLD(        "Threshold",        ColumnType.DOUBLE);
331
332        /**
333         * The CSV header for this column.
334         */
335        public final String header;
336
337        /**
338         * The Tablesaw type associated with this column.
339         */
340        public final ColumnType type;
341
342        Column(String header, ColumnType type) {
343            this.header = header;
344            this.type = type;
345        }
346    }
