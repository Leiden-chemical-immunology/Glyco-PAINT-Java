034package paint.shared.objects;
035
036import java.util.ArrayList;
037import java.util.List;
038
039/**
040 * Represents a scientific or research experiment containing
041 * a name and a collection of {@link Recording} instances.
042 *
043 * <p>Provides constructors for initialization and utility methods
044 * for managing the set of associated recordings.</p>
045 */
046public class Experiment {
047
048    // ───────────────────────────────────────────────────────────────────────────────
049    // FIELDS
050    // ───────────────────────────────────────────────────────────────────────────────
051
052    /**
053     * The name identifying this experiment.
054     */
055    private final String experimentName;
056
057    /**
058     * The collection of recordings associated with this experiment.
059     */
060    private final ArrayList<Recording> recordings;
061
062    // ───────────────────────────────────────────────────────────────────────────────
063    // CONSTRUCTORS
064    // ───────────────────────────────────────────────────────────────────────────────
065
066    /**
067     * Constructs an {@code Experiment} with a specified name.
068     *
069     * @param experimentName the name of the experiment
070     */
071    public Experiment(String experimentName) {
072        this.experimentName = experimentName;
073        this.recordings     = new ArrayList<>();
074    }
075
076    // ───────────────────────────────────────────────────────────────────────────────
077    // ACCESSORS & MUTATORS
078    // ───────────────────────────────────────────────────────────────────────────────
079
080    /**
081     * @return the name of this experiment.
082     */
083    public String getExperimentName() {
084        return experimentName;
085    }
086
087    /**
088     * Adds a recording to this experiment.
089     *
090     * @param recording the {@link Recording} to add
091     */
092    public void   addRecording(Recording recording) {
093        this.recordings.add(recording);
094    }
095
096    /**
097     * @return the list of {@link Recording} objects in this experiment.
098     */
099    public List<Recording> getRecordings() {
100        return recordings;
101    }
102
103    // ───────────────────────────────────────────────────────────────────────────────
104    // STRING REPRESENTATION
105    // ───────────────────────────────────────────────────────────────────────────────
106
107    /**
108     * Returns a formatted string summary of the experiment and its recordings.
109     *
110     * @return formatted string representation of the experiment
111     */
112    @Override
113    public String toString() {
114        StringBuilder sb = new StringBuilder();
115        sb.append("\n\n");
116        sb.append("----------------------------------------------------------------------\n");
117        sb.append("Experiment: ").append(experimentName).append("\n");
118        sb.append("----------------------------------------------------------------------\n");
119        sb.append("\n");
120        sb.append(String.format("%nExperiment %s has %d recordings%n", experimentName, recordings.size()));
121        for (Recording recording : recordings) {
122            sb.append(String.format("\t%s%n", recording.getRecordingName()));
123        }
124        return sb.toString();
125    }
126}
