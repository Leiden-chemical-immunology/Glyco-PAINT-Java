001/*=============================================================================
002 *  Class:        Square.java
003 *  Package:      paint.shared.objects
004 *
005 *  PURPOSE:
006 *    Represents a spatial analysis region (square) within a PAINT recording.
007 *    A square models:
008 *
009 *       • Its grid location (row/column)
010 *       • Pixel coordinates (x0,y0,x1,y1)
011 *       • Associated track statistics
012 *       • Visibility and exclusion flags
013 *       • A list of Track objects and (optional) a Tablesaw track table
014 *
015 *  DESCRIPTION:
016 *    Each square corresponds to a rectangular region extracted from a
017 *    recording frame. It forms the basis for spatial analysis across:
018 *
019 *       • Single-molecule densities
020 *       • Variability and diffusion metrics
021 *       • Track-based temporal properties
022 *
023 *    This version embeds the schema through the {@link Column} enum,
024 *    replacing the old SquareSchema class entirely. All table I/O classes now
025 *    rely on Square.Column.values() for headers and column types.
026 *
027 *  KEY FEATURES:
028 *    • Full embedded schema enum with CSV headers + Tablesaw types
029 *    • Rich metadata including geometry, track metrics, and flags
030 *    • Utility constructors for fully-specified and grid-generated squares
031 *    • Integration with PAINT’s track pipeline via a Track list and table
032 *    • Clean, formatted debugging/diagnostic output via toString()
033 *
034 *  AUTHOR:
035 *    Hans Bakker
036 *
037 *  MODULE:
038 *    paint-shared-utils
039 *
040 *  UPDATED:
041 *    2025-12-31
042 *
043 *  COPYRIGHT:

126    /**
127     * Creates a fully-specified {@code Square} with geometric coordinates.
128     *
129     * @param uniqueKey      a unique identifier for this square (typically exp§rec§num)
130     * @param experimentName name of the experiment containing this square
131     * @param recordingName  name of the recording containing this square
132     * @param squareNumber   index of this square in the recording (0..N-1)
133     * @param rowNumber      row index in the grid
134     * @param colNumber      column index in the grid
135     * @param x0             top-left x coordinate in µm
136     * @param y0             top-left y coordinate in µm
137     * @param x1             bottom-right x coordinate in µm
138     * @param y1             bottom-right y coordinate in µm
139     */
140    public Square(String uniqueKey,
141            String experimentName,
142            String recordingName,
143            int squareNumber,
144            int rowNumber,
145            int colNumber,
146            double x0, double y0,
147            double x1, double y1) {
148
149        initialiseDoublesToNaN(this);
150        this.uniqueKey      = uniqueKey;
151        this.experimentName = experimentName;
152        this.recordingName  = recordingName;
153        this.squareNumber   = squareNumber;
154        this.rowNumber      = rowNumber;
155        this.colNumber      = colNumber;
156        this.x0             = round(x0, 2);
157        this.y0             = round(y0, 2);
158        this.x1             = round(x1, 2);
159        this.y1             = round(y1, 2);
160    }

162    /**
163     * Computes the theoretical square area for a given grid size.
164     *
165     * @param n total number of squares in the recording grid
166     * @return the area of a single square in µm²
167     */
168    public static double calculateSquareArea(int n) {
169        return IMAGE_WIDTH * IMAGE_HEIGHT / n;
170    }

711    @Override
712    public String toString() {
713        StringBuilder sb = new StringBuilder();
714
715        sb.append("\n----------------------------------------------------------------------\n")
716          .append("Square ").append(squareNumber)
717          .append(" (Experiment: ").append(experimentName != null ? experimentName : "N/A").append(")\n")
718          .append(" (Recording: ").append(recordingName != null ? recordingName : "N/A").append(")\n")
719          .append("----------------------------------------------------------------------\n");
720
721        sb.append(String.format("Row,Col Number                 : %d,%d%n", rowNumber, colNumber));
722        sb.append(String.format("Coordinates [x0,y0]-[x1,y1]    : [%.2f, %.2f] - [%.2f, %.2f]%n", x0, y0, x1, y1));
723        sb.append(String.format("Unique Key                     : %s%n", uniqueKey != null ? uniqueKey : "N/A"));
724        sb.append(String.format("Label Number                   : %d%n", labelNumber));
725        sb.append(String.format("Cell ID                        : %d%n", cellId));
726
727        sb.append(String.format("Visible                        : %b%n", visible));
728        sb.append(String.format("Square Manually Excluded       : %b%n", squareManuallyExcluded));
729        sb.append(String.format("Image Excluded                 : %b%n", imageExcluded));
730
731        sb.append(String.format("Number of Tracks               : %d%n", numberOfTracks));
732        sb.append(String.format("Variability                    : %.4f%n", variability));
733        sb.append(String.format("Density                        : %.4f%n", density));
734        sb.append(String.format("Density Ratio                  : %.4f%n", densityRatio));
735        sb.append(String.format("Tau                            : %.4f%n", tau));
736        sb.append(String.format("R²                             : %.4f%n", rSquared));
737
738        sb.append(String.format("Median Diffusion Coefficient   : %.4f%n", medianDiffusionCoefficient));
739        sb.append(String.format("Median Diffusion CoefficientExt: %.4f%n", medianDiffusionCoefficientExt));
740        sb.append(String.format("Median Displacement            : %.4f%n", medianDisplacement));
741        sb.append(String.format("Max Displacement               : %.4f%n", maxDisplacement));
742        sb.append(String.format("Total Displacement             : %.4f%n", totalDisplacement));
743
744        sb.append(String.format("Median Max Speed               : %.4f%n", medianMaxSpeed));
745        sb.append(String.format("Max Max Speed                  : %.4f%n", maxMaxSpeed));
746        sb.append(String.format("Median Median Speed            : %.4f%n", medianMedianSpeed));
747        sb.append(String.format("Max Median Speed               : %.4f%n", maxMedianSpeed));
748
749        sb.append(String.format("Max Track Duration             : %.4f%n", maxTrackDuration));
750        sb.append(String.format("Total Track Duration           : %.4f%n", totalTrackDuration));
751        sb.append(String.format("Median Track Duration          : %.4f%n", medianTrackDuration));
752
753        if (tracks != null) {
754            sb.append(String.format("Tracks attached                : %d%n", tracks.size()));
755        }
756        if (tracksTable != null) {
757            sb.append("Tracks table available\n");
758        }
759
760        return sb.toString();
761    }
