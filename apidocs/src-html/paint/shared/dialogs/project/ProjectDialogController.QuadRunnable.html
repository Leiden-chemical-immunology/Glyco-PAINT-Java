050/**
051 * Controller for ProjectDialog. It reacts to UI events coming from the panels,
052 * controls enabling/disabling of buttons, performs validations, and manages the

058    private final JDialog             dialog;
059    private final PaintConfig         paintConfig;

122    public void init() {
123
124        // Browsing
125        paths.onBrowseProject(this::handleBrowseProject);
126        paths.onBrowseImages(this::handleBrowseImages);
127
128        // Text changes → OK may update
129        paths.onRootsChanged(this::updateOk);
130        experiments.onSelectionChanged(this::updateOk);
131
132        // Parameter changes (TrackMate only)
133        if (params != null) {
134            params.onParamsChanged(this::updateOk);
135        }
136
137        // Sweep toggle
138        bottom.onVerboseToggle(this::onVerboseToggle);
139        bottom.onSweepToggle(this::onSweepToggle);
140
141        // OK/Cancel
142        bottom.onOk(this::handleOk);
143        bottom.onCancel(this::handleCancel);
144
145        // Initial state
146        updateOk();
147    }

149    // ----------------------------------------------------------------------------------------------------
150    //  OK
151    // ----------------------------------------------------------------------------------------------------
152
153    private void handleOk() {
154
155        if (mode == DialogMode.TRACKMATE_BATCH) {
156            final String img = paths.imagesRootText().trim();
157            if (!new File(img).isDirectory()) {
158                JOptionPane.showMessageDialog(
159                        dialog,
160                        "The Images Root directory does not exist. Please select a valid directory.",
161                        "Invalid Images Root",
162                        JOptionPane.WARNING_MESSAGE
163                );
164                return;
165            }
166        }
167
168        PaintLogger.debugf("ProjectDialogController.handleOK - project path: %s", paths.imagesRootText());
169        PaintLogger.debugf("ProjectDialogController.handleOK - images  path: %s", paths.projectRootText());
170        PaintLogger.debugf("ProjectDialogController.handleOK - experiments : %s", experiments.selectedExperimentNames());
171
172        PaintPrefs.putString("Path", "Project Root", paths.projectRootText());
173        PaintPrefs.putString("Path", "Images Root", paths.imagesRootText());
174
175        // Persist experiment selection if requested (clear section, then write only selected)
176        if (bottom.isSaveExperimentsSelected()) {
177            PaintConfig.removeSection("Experiments"); // clears entire section + autosaves
178            for (String name : experiments.selectedExperimentNames()) {
179                PaintConfig.setBoolean("Experiments", name, true);
180            }
181        }
182
183        clearCancelled.run();
184        workerStarted = true;
185
186        Runnable uiDisable = () -> {
187            setInputsEnabled(false);
188            bottom.showRunning();
189        };
190
191        Runnable uiEnable = () -> {
192            setInputsEnabled(true);
193            bottom.resetOk(mode == DialogMode.VIEWER);
194        };
195
196        Runnable onSuccess = () -> {
197            if (mode == DialogMode.TRACKMATE_SINGLE || mode == DialogMode.VIEWER) {
198                // In SINGLE mode, OK just opens the modeless TrackMateSingleDialog.
199                // TrackMate runs later from that dialog, so don't claim success here.
200                bottom.resetOk(false); // or bottom.resetOk(validToRun()); if you prefer
201                return;
202            }
203
204            PaintLogger.blankline();
205            PaintLogger.infof("Operation completed successfully.");
206            bottom.showCompleted(mode == DialogMode.VIEWER);
207        };
208
209        Runnable onFailure = () -> {
210            JOptionPane.showMessageDialog(
211                    dialog,
212                    "Operation finished with errors. Check the log.",
213                    "Warning",
214                    JOptionPane.WARNING_MESSAGE
215            );
216        };
217
218        startWorker.run(uiDisable, uiEnable, onSuccess, onFailure);
219    }
220
221    // ----------------------------------------------------------------------------------------------------
222    //  CANCEL
223    // ----------------------------------------------------------------------------------------------------
224
225    private void handleCancel() {
226
227        setCancelled.run();
228        Thread t = getWorker.get();
229
230        bottom.showStopping();
231
232        // -----------------------------------------------------------------------------------------------
233        // CASE 1 — Worker still running → perform real cancellation
234        // -----------------------------------------------------------------------------------------------
235        if (t != null && t.isAlive()) {
236            PaintLogger.infof("Cancellation requested — waiting for worker to finish...");
237            new Thread(() -> handleWorkerShutdown(t), "ForceShutdownWatcher").start();
238            return;
239        }
240
241        // -----------------------------------------------------------------------------------------------
242        // CASE 2 — Cancel BEFORE worker started (invalid dialog state)
243        // Re-enable full UI so user can fix input
244        // -----------------------------------------------------------------------------------------------
245        if (!workerStarted) {
246            PaintLogger.infof("Cancellation before start — closing dialog.");
247            clearCancelled.run();
248            SwingUtilities.invokeLater(dialog::dispose);
249            return;
250        }
251
252        // -----------------------------------------------------------------------------------------------
253        // CASE 3 — Worker started and is DONE → treat select Cancel as dialog close
254        // -----------------------------------------------------------------------------------------------
255        PaintLogger.infof("Cancel after completion — closing dialog.");
256        paint.shared.utils.PaintConsoleWindow.closeIfVisible();
257        SwingUtilities.invokeLater(dialog::dispose);
258        clearCancelled.run();
259    }
260
261    // ----------------------------------------------------------------------------------------------------
262    //  Worker Shutdown
263    // ----------------------------------------------------------------------------------------------------
264
265    private void handleWorkerShutdown(Thread thread) {
266        try {
267            thread.join();
268        } catch (InterruptedException ignored) {}
269        SwingUtilities.invokeLater(() -> finishWorkerShutdown(thread));
270    }
271
272    private void finishWorkerShutdown(Thread thread) {
273
274        if (thread.isAlive()) {
275            PaintLogger.errorf("Worker thread did not stop — forcing JVM halt.");
276            Runtime.getRuntime().halt(0);
277            return;
278        }
279
280        PaintLogger.infof("Worker thread terminated cleanly.");
281
282        workerStarted = false;
283        bottom.resetOk(true);
284        clearCancelled.run();
285
286        enableFullUI();
287    }

289    // ----------------------------------------------------------------------------------------------------
290    //  Sweep Toggle
291    // ----------------------------------------------------------------------------------------------------
292
293    private void onSweepToggle(boolean selected) {
294
295        if (selected) {
296            final Path root = getProjectPath.get();
297            final Path sweepFile = root.resolve("Paint Sweep Configuration.json");
298
299            if (!java.nio.file.Files.exists(sweepFile)) {
300
301                int res = JOptionPane.showConfirmDialog(
302                        dialog,
303                        "The file \"Paint Sweep Configuration.json\" does not exist in the project root.\n\n" +
304                                "Do you want to create it now with default sweep settings?",
305                        "Sweep Configuration Missing",
306                        JOptionPane.YES_NO_OPTION,
307                        JOptionPane.WARNING_MESSAGE
308                );
309
310                if (res == JOptionPane.YES_OPTION) {
311                    try {
312                        paintConfig.setSweepDefaults(root);
313
314                        JOptionPane.showMessageDialog(
315                                dialog,
316                                "Sweep configuration file has been created:\n" +
317                                        sweepFile.toAbsolutePath() +
318                                        "\nYou should edit that file to enable the desired sweep options.",
319                                "Sweep File Created",
320                                JOptionPane.INFORMATION_MESSAGE
321                        );
322                    } catch (Exception ex) {
323                        JOptionPane.showMessageDialog(
324                                dialog,
325                                "Failed to create sweep configuration:\n" + ex.getMessage(),
326                                "Error",
327                                JOptionPane.ERROR_MESSAGE
328                        );
329                    }
330                } else {
331                    bottom.setSweepSelected(false);
332                }
333            }
334        }
335
336        updateOk();
337    }

340    }

342    // ----------------------------------------------------------------------------------------------------
343    //  Browse handlers
344    // ----------------------------------------------------------------------------------------------------
345
346    private void handleBrowseProject(File dir) {
347        updateProjectRoot(dir.toPath());
348    }
349
350    private void handleBrowseImages(File ignored) {
351        updateOk();
352    }
353
354    private void updateProjectRoot(Path newRoot) {
355        setProjectPath.accept(newRoot);
356        bottom.updateOkEnabled(validToRun());
357    }
358
359    // ----------------------------------------------------------------------------------------------------
360    //  Utility
361    // ----------------------------------------------------------------------------------------------------
362
363    private void updateOk() {
364        bottom.updateOkEnabled(validToRun());
365    }
366
367    private boolean validToRun() {
368        if (mode == DialogMode.VIEWER) {
369            return paths.isProjectRootValid();
370        }
371        return paths.isProjectRootValid() && experiments.anySelected();
372    }
373
374    /**
375     * Enables/disables all interactive UI elements EXCEPT the Cancel button.
376     */
377    private void setInputsEnabled(boolean enabled) {
378
379        // Disable Project + Images root fields
380        paths.setEnabled(enabled, mode);
381
382        // Disable TrackMate / Squares parameters
383        if (params != null) {
384            params.setEnabled(enabled);
385        }
386
387        experiments.setEnabled(enabled);
388
389        bottom.setEnabled(enabled);
390
391        if (!enabled) {
392            bottom.keepCancelEnabled();
393        }
394    }
395
396    /**
397     * Restores full dialog UI after cancellation-before-start.
398     */
399    private void enableFullUI() {
400        enableAllUiFromDialog.run();          // Re-enable all fields/buttons
401        bottom.resetOk(validToRun());         // Restore OK button state
402    }

404    /**
405     * Functional interface used to launch worker logic with four callbacks.
406     */
407    @FunctionalInterface
408    public interface QuadRunnable {
409        @SuppressWarnings("unused")
410        void run(Runnable a, Runnable b, Runnable c, Runnable d);
411    }
