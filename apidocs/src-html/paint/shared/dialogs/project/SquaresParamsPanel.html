036import paint.shared.config.paintconfig.PaintConfig;

038import javax.swing.*;
039import javax.swing.border.TitledBorder;
040import java.awt.*;
041
042import static paint.shared.constants.PaintStringConstants.NUMBER_OF_SQUARES_IN_RECORDING;
043import static paint.shared.constants.PaintStringConstants.MIN_REQUIRED_R_SQUARED;
044import static paint.shared.constants.PaintStringConstants.MIN_REQUIRED_DENSITY_RATIO;
045import static paint.shared.constants.PaintStringConstants.MAX_ALLOWABLE_VARIABILITY;
046import static paint.shared.constants.PaintStringConstants.GENERATE_SQUARES;
047import static paint.shared.constants.PaintStringConstants.TRACKMATE;
048import static paint.shared.constants.PaintStringConstants.RUN_GENERATE_SQUARES_AFTER;

050import static paint.shared.dialogs.ProjectDialog.DialogMode;

052/**
053 * Panel that displays all the configuration parameters required for the "Generate Squares" step.
054 * In TRACKMATE mode, a checkbox allows configuring whether to run Generate Squares automatically
055 * after TrackMate processing. The panel allows changes, exposes a callback for any parameter
056 * modification, and supports persistence of these settings back to PaintConfig.
057 */
058public class SquaresParamsPanel {
059
060    private final JPanel            panel = new JPanel(new GridBagLayout());
061    private       JCheckBox         runAfterTrackMate;
062    private final JComboBox<String> gridSizeCombo;
063    private final JTextField        minRSqField;
064    private final JTextField        minDensityField;
065    private final JTextField        maxVariabilityField;
066
067    private final JLabel            gridSizeLabel;
068    private final JLabel            minRSqLabel;
069    private final JLabel            minDensityLabel;
070    private final JLabel            maxVariabilityLabel;
071
072    private final Color             normalLabelColor;
073    private final Color             disabledLabelColor = Color.GRAY;
074
075    // Callback triggered whenever any user-editable parameter changes.
076    private Runnable onChange = () -> {
077    };
078
079    /**
080     * Constructs the panel and initializes fields based on PaintConfig and dialog mode.
081     */
082    public SquaresParamsPanel(DialogMode mode) {
083
084        panel.setBorder(new TitledBorder("Generate Squares Parameters"));
085        final GridBagConstraints pg = new GridBagConstraints();
086        pg.insets  = new Insets(5,5,5,5);
087        pg.anchor  = GridBagConstraints.WEST;
088        pg.fill    = GridBagConstraints.NONE;
089
090        // Load defaults from configuration
091        int nrSquares   = PaintConfig.getInt(   GENERATE_SQUARES, NUMBER_OF_SQUARES_IN_RECORDING, 400);
092        double minRSq   = PaintConfig.getDouble(GENERATE_SQUARES, MIN_REQUIRED_R_SQUARED,         0.1);
093        double minDens  = PaintConfig.getDouble(GENERATE_SQUARES, MIN_REQUIRED_DENSITY_RATIO,     2.0);
094        double maxVar   = PaintConfig.getDouble(GENERATE_SQUARES, MAX_ALLOWABLE_VARIABILITY,      10.0);
095
096        int row = 0;
097
098        // TRACKMATE mode: checkbox to run Generate Squares after TrackMate completes
099        if (mode == DialogMode.TRACKMATE_BATCH) {
100            runAfterTrackMate = new JCheckBox(
101                    "Run Generate Squares after TrackMate",
102                    PaintConfig.getBoolean(TRACKMATE, RUN_GENERATE_SQUARES_AFTER, true)
103            );
104            pg.gridx = 0; pg.gridy = row; pg.gridwidth = 2;
105            panel.add(runAfterTrackMate, pg);
106            row++;
107            pg.gridwidth = 1;
108
109            runAfterTrackMate.addActionListener(e -> {
110                setSquaresEnabled(runAfterTrackMate.isSelected());
111                onChange.run();
112            });
113        }
114
115        final Dimension labelSize = new Dimension(220, 20);
116        final Dimension fieldSize = new Dimension(80, 24);
117
118        pg.gridx = 0;
119        pg.gridy = row;
120        gridSizeLabel = label(panel, NUMBER_OF_SQUARES_IN_RECORDING, labelSize, pg);
121        pg.gridx = 1;
122        gridSizeCombo = new JComboBox<>(new String[]{"5x5", "10x10", "15x15", "20x20", "25x25", "30x30", "35x35", "40x40"});
123        int n = (int) Math.sqrt(nrSquares);
124        gridSizeCombo.setSelectedItem(n + "x" + n);
125        panel.add(gridSizeCombo, pg);
126        row++;
127
128        // Min R²
129        pg.gridx = 0;
130        pg.gridy = row;
131        minRSqLabel = label(panel, "Min Required R²", labelSize, pg);
132        pg.gridx = 1;
133        minRSqField = text(String.valueOf(minRSq), fieldSize);
134        panel.add(minRSqField, pg);
135        row++;
136
137        // Min Density Ratio
138        pg.gridx = 0;
139        pg.gridy = row;
140        minDensityLabel = label(panel, MIN_REQUIRED_DENSITY_RATIO, labelSize, pg);
141        pg.gridx = 1;
142        minDensityField = text(String.valueOf(minDens), fieldSize);
143        panel.add(minDensityField, pg);
144        row++;
145
146        // Max Variability
147        pg.gridx = 0;
148        pg.gridy = row;
149        maxVariabilityLabel = label(panel, MAX_ALLOWABLE_VARIABILITY, labelSize, pg);
150        pg.gridx = 1;
151        maxVariabilityField = text(String.valueOf(maxVar), fieldSize);
152        panel.add(maxVariabilityField, pg);
153
154        normalLabelColor = gridSizeLabel.getForeground();
155
156        // Change listeners for all controls (REPLACED WITH METHOD REFERENCES)
157        gridSizeCombo.addActionListener(this::handleChange);
158        minRSqField.getDocument().addDocumentListener((SimpleDocumentListener) this::handleChange);
159        minDensityField.getDocument().addDocumentListener((SimpleDocumentListener) this::handleChange);
160        maxVariabilityField.getDocument().addDocumentListener((SimpleDocumentListener) this::handleChange);
161
162        // Initial enable state for controls in TRACKMATE mode
163        if (mode == DialogMode.TRACKMATE_BATCH) {
164            setSquaresEnabled(runAfterTrackMate.isSelected());
165        }
166    }
167
168    /**
169     * Returns the Swing component associated with this panel.
170     */
171    public JPanel component() {
172        return panel;
173    }
174
175    /**
176     * Registers a callback that fires whenever any parameter changes.
177     * If null is passed, it resets to a no-op callback.
178     */
179    public void onParamsChanged(Runnable runnable) {
180        this.onChange = (runnable != null ? runnable : () -> {
181        });
182    }
183
184    /**
185     * Enables or disables the panel. When in TRACKMATE mode, the Generate Squares fields
186     * respect both the main enabled flag and the state of the "Run after TrackMate" checkbox.
187     */
188    public void setEnabled(boolean enabled) {
189
190        if (runAfterTrackMate != null) {
191            runAfterTrackMate.setEnabled(enabled);
192        }
193
194        boolean squaresEnabled = enabled && (runAfterTrackMate == null || runAfterTrackMate.isSelected());
195
196        setSquaresEnabled(squaresEnabled);
197
198        gridSizeLabel.setForeground(squaresEnabled ? normalLabelColor : disabledLabelColor);
199        minRSqLabel.setForeground(squaresEnabled ? normalLabelColor : disabledLabelColor);
200        minDensityLabel.setForeground(squaresEnabled ? normalLabelColor : disabledLabelColor);
201        maxVariabilityLabel.setForeground(squaresEnabled ? normalLabelColor : disabledLabelColor);
202    }
203
204    /**
205     * Persists the current field values to PaintConfig and saves the configuration.
206     */
207    public void persistTo(DialogMode mode) {
208        if (gridSizeCombo != null) {
209            String selectedItem = (String) gridSizeCombo.getSelectedItem();
210            if (selectedItem != null && selectedItem.contains("x")) {
211                int side = Integer.parseInt(selectedItem.split("x")[0].trim());
212                PaintConfig.setInt(GENERATE_SQUARES, NUMBER_OF_SQUARES_IN_RECORDING, side * side);
213            }
214        }
215        if (minRSqField != null) {
216            PaintConfig.setDouble(GENERATE_SQUARES, MIN_REQUIRED_R_SQUARED,
217                                  parseDouble(minRSqField.getText(), 0.1));
218        }
219        if (minDensityField != null) {
220            PaintConfig.setDouble(GENERATE_SQUARES, MIN_REQUIRED_DENSITY_RATIO,
221                                  parseDouble(minDensityField.getText(), 2.0));
222        }
223        if (maxVariabilityField != null) {
224            PaintConfig.setDouble(GENERATE_SQUARES, MAX_ALLOWABLE_VARIABILITY ,
225                                  parseDouble(maxVariabilityField.getText(), 10.0));
226        }
227        if (mode == DialogMode.TRACKMATE_BATCH && runAfterTrackMate != null) {
228            PaintConfig.setBoolean(TRACKMATE, RUN_GENERATE_SQUARES_AFTER, runAfterTrackMate.isSelected());
229        }
230        PaintConfig.instance().save();
231    }
232
233    /**
234     * Adds a fixed-size label into the GridBag container.
235     */
236    private static JLabel label(JPanel jPanel, String text, Dimension size, GridBagConstraints pg) {
237        JLabel jLabel = new JLabel(text);
238        jLabel.setPreferredSize(size);
239        jPanel.add(jLabel, pg);
240        return jLabel;
241    }
242
243    /**
244     * Builds a fixed-size text field initialized to a given value.
245     */
246    private static JTextField text(String v, Dimension size) {
247        JTextField jTextField = new JTextField(v);
248        jTextField.setColumns(8);
249        jTextField.setPreferredSize(size);
250        return jTextField;
251    }
252
253    /**
254     * Enables or disables all the Generate Squares fields (grid size, R², density, variability).
255     */
256    public void setSquaresEnabled(boolean enabled) {
257        gridSizeCombo.setEnabled(enabled);
258        minRSqField.setEnabled(enabled);
259        minDensityField.setEnabled(enabled);
260        maxVariabilityField.setEnabled(enabled);
261
262        gridSizeLabel.setForeground(enabled ? normalLabelColor : disabledLabelColor);
263        minRSqLabel.setForeground(enabled ? normalLabelColor : disabledLabelColor);
264        minDensityLabel.setForeground(enabled ? normalLabelColor : disabledLabelColor);
265        maxVariabilityLabel.setForeground(enabled ? normalLabelColor : disabledLabelColor);
266    }
267
268    /**
269     * Parses a double from string with fallback to a default on error.
270     */
271    private static double parseDouble(String doubleString, double def) {
272        try {
273            return Double.parseDouble(doubleString.trim());
274        } catch (Exception e) {
275            return def;
276        }
277    }

285        onChange.run();
286    }
287
288    @SuppressWarnings("unused")
289    private void handleChange(javax.swing.event.DocumentEvent e) {
290        onChange.run();
291    }
292
293    public boolean isRunAfterTrackMateSelected() {
294        return runAfterTrackMate != null && runAfterTrackMate.isSelected();
295    }
