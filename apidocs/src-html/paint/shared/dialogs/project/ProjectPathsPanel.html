028package paint.shared.dialogs.project;
029
030import javax.swing.*;
031import javax.swing.border.EmptyBorder;
032import javax.swing.event.DocumentEvent;
033import java.awt.*;
034import java.awt.event.ActionEvent;
035import java.io.File;
036import java.nio.file.Path;
037import java.util.function.Consumer;
038
039import static paint.shared.dialogs.ProjectDialog.DialogMode;
040
041/**
042 * Panel allowing users to select the Project Root and Images Root directories.
043 * <p>
044 * The panel exposes browse buttons, text fields, and callback hooks. It notifies
045 * listeners when the text changes or when browse actions complete.
046 */
047public class ProjectPathsPanel {
048
049    private final JPanel     panel = new JPanel(new GridBagLayout());  // The root UI panel containing the layout.
050    private final JTextField projectRootField;                         // Text field for the project root directory.
051    private final JTextField imagesRootField;                          // Text field for the images root directory
052    private final JButton    browseProjectBtn;                         // Browse button for project root.
053    private final JButton    browseImagesBtn;                          // Browse button for images root.
054
055    /**
056     * Runnable invoked when either root changes.
057     */
058    private Runnable rootsChanged = () -> {
059    };
060
061    /**
062     * Consumer invoked when a project directory is chosen via the browse dialog.
063     */
064    private Consumer<File> onProjectChosen = f -> {
065    };
066
067    /**
068     * Consumer invoked when an images directory is chosen via the browse dialog.
069     */
070    private Consumer<File> onImagesChosen = f -> {
071    };
072
073    /**
074     * Constructs the UI panel, initializes fields, and wires callbacks.
075     *
076     * @param mode           dialog mode (affects Images Root field behavior)
077     * @param initialProject initial project directory to display
078     */
079    public ProjectPathsPanel(DialogMode mode, Path initialProject) {
080        panel.setBorder(new EmptyBorder(10, 10, 10, 10));
081        final GridBagConstraints gbc = new GridBagConstraints();
082        gbc.insets  = new Insets(5, 5, 5, 5);
083        gbc.anchor  = GridBagConstraints.WEST;
084        gbc.fill    = GridBagConstraints.HORIZONTAL;
085        gbc.weightx = 1.0;
086
087        final Dimension labelSize = new Dimension(200, 20);
088        int row = 0;
089
090        // Project Root label
091        JLabel lblProject = new JLabel("Project Root");
092        lblProject.setPreferredSize(labelSize);
093
094        gbc.gridx   = 0;
095        gbc.gridy   = row;
096        gbc.weightx = 0;
097        gbc.fill    = GridBagConstraints.NONE;
098        panel.add(lblProject, gbc);
099
100        // Text field for project root
101        projectRootField = new JTextField(initialProject != null ? initialProject.toString() : System.getProperty("user.home"), 32);
102        gbc.gridx        = 1;
103        gbc.weightx      = 1.0;
104        gbc.fill         = GridBagConstraints.HORIZONTAL;
105        panel.add(projectRootField, gbc);
106
107        // Browse project button
108        browseProjectBtn = new JButton("Browse...");
109        gbc.gridx        = 2;
110        gbc.weightx      = 0;
111        gbc.fill         = GridBagConstraints.NONE;
112        panel.add(browseProjectBtn, gbc);
113
114        row++;
115
116        // Images Root label
117        JLabel lblImages = new JLabel("Images Root");
118        lblImages.setPreferredSize(labelSize);
119
120        gbc.gridx   = 0;
121        gbc.gridy   = row;
122        gbc.weightx = 0;
123        gbc.fill    = GridBagConstraints.NONE;
124        panel.add(lblImages, gbc);
125
126        // Load saved Images Root
127        String savedImagesRoot = paint.shared.utils.PaintPrefs.getString(
128                "Path",
129                "Images Root",
130                System.getProperty("user.home")
131        );
132
133        // Text field for images root
134        imagesRootField = new JTextField(savedImagesRoot, 32);
135        gbc.gridx       = 1;
136        gbc.weightx     = 1.0;
137        gbc.fill        = GridBagConstraints.HORIZONTAL;
138        panel.add(imagesRootField, gbc);
139
140        // Browse images button
141        browseImagesBtn = new JButton("Browse...");
142        gbc.gridx       = 2;
143        gbc.weightx     = 0;
144        gbc.fill        = GridBagConstraints.NONE;
145        panel.add(browseImagesBtn, gbc);
146
147        // Browse action for project root
148        browseProjectBtn.addActionListener(this::browseProjectRoot);
149
150        // Browse action for images root
151        browseImagesBtn.addActionListener(this::browseImagesRoot);
152
153        // Notify when text fields change
154        projectRootField.getDocument().addDocumentListener((SimpleDocumentListener) this::onTextChanged);
155        imagesRootField.getDocument().addDocumentListener((SimpleDocumentListener) this::onTextChanged);
156
157        // GENERATE_SQUARES mode: disable Images Root controls
158        if (mode == DialogMode.GENERATE_SQUARES) {
159            imagesRootField.setEnabled(false);
160            imagesRootField.setBackground(UIManager.getColor("Panel.background"));
161            imagesRootField.setForeground(Color.GRAY);
162            imagesRootField.setFocusable(false);
163            browseImagesBtn.setEnabled(false);
164        }
165    }
166
167    /**
168     * @return the root Swing component
169     */
170    public JPanel component() {
171        return panel;
172    }
173
174    /**
175     * @return the project root path as text
176     */
177    public String projectRootText() {
178        return projectRootField.getText().trim();
179    }
180
181    /**
182     * @return the images root path as text
183     */
184    public String imagesRootText() {
185        return imagesRootField.getText().trim();
186    }
187
188    /**
189     * @return true if the project root path points to a directory
190     */
191    public boolean isProjectRootValid() {
192        return new File(projectRootText()).isDirectory();
193    }
194
195    /**
196     * Enables or disables UI fields depending on dialog mode.
197     */
198    public void setEnabled(boolean enabled, DialogMode mode) {
199        // Project root always follows enabled state
200        projectRootField.setEnabled(enabled);
201        browseProjectBtn.setEnabled(enabled);
202
203        if (mode == DialogMode.GENERATE_SQUARES) {
204            // Only this mode disables images root
205            imagesRootField.setEnabled(false);
206            browseImagesBtn.setEnabled(false);
207        } else {
208            // TRACKMATE and VIEWER
209            imagesRootField.setEnabled(enabled);
210            browseImagesBtn.setEnabled(enabled);
211        }
212    }
213
214    /**
215     * Update the project root text field when changed externally.
216     */
217    public void onProjectRootChanged(Path newRoot) {
218        if (newRoot != null) {
219            projectRootField.setText(newRoot.toString());
220        }
221    }
222
223    /**
224     * Assign callback for general root changes.
225     */
226    public void onRootsChanged(Runnable r) {
227        this.rootsChanged = (r != null ? r : () -> {
228        });
229    }
230
231    /**
232     * Assign callback specifically for project directory browsing.
233     */
234    public void onBrowseProject(Consumer<File> c) {
235        this.onProjectChosen = (c != null ? c : f -> {
236        });
237    }
238
239    /**
240     * Assign callback specifically for images directory browsing.
241     */
242    public void onBrowseImages(Consumer<File> c) {
243        this.onImagesChosen = (c != null ? c : f -> {
244        });
245    }
246
247    /**
248     * Invoked when text changes in either root field; triggers the rootsChanged callback.
249     */
250    @SuppressWarnings("unused")
251    private void onTextChanged(DocumentEvent e) {
252        rootsChanged.run();
253    }
254
255    /**
256     * Returns the value if it is a valid directory; otherwise returns the user's home directory.
257     */
258    private static String valueOrHome(String s) {
259        if (s == null || s.trim().isEmpty()) {
260            return System.getProperty("user.home");
261        }
262        File f = new File(s.trim());
263        return f.isDirectory() ? f.getAbsolutePath() : System.getProperty("user.home");
264    }
265
266    /** Handles the Browse action for the Project Root. */
267    @SuppressWarnings("unused")
268    private void browseProjectRoot(ActionEvent e) {
269        File dir = DirectoryChooser.chooseDirectory(panel, "Project Root", valueOrHome(projectRootField.getText()));
270        if (dir != null && dir.isDirectory()) {
271            projectRootField.setText(dir.getAbsolutePath());
272            onProjectChosen.accept(dir);
273        }
274    }
275
276    /** Handles the Browse action for the Images Root. */
277    @SuppressWarnings("unused")
278    private void browseImagesRoot(ActionEvent e) {
279        File dir = DirectoryChooser.chooseDirectory(panel, "Images Root", valueOrHome(imagesRootField.getText()));
280        if (dir != null && dir.isDirectory()) {
281            imagesRootField.setText(dir.getAbsolutePath());
282            onImagesChosen.accept(dir);
283        }
284    }
