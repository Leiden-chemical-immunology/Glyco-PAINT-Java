026package paint.shared.constants;
027
028/**
029 * Defines geometric constants used for spatial calculations and image
030 * rendering within the PAINT application.
031 */
032public final class PaintGeometry {
033
034    private PaintGeometry() {
035        // Prevent instantiation
036    }
037
038    /** Pixel width in µm (specified by Nikon). */
039    public static final double PIXEL_WIDTH          = 0.1603251;
040    /** Pixel height in µm (specified by Nikon). */
041    public static final double PIXEL_HEIGHT         = 0.1603251;
042    /** Number of pixels in image width (specified by Nikon). */
043    public static final int    NUMBER_PIXELS_WIDTH  = 512;
044    /** Number of pixels in image height (specified by Nikon). */
045    public static final int    NUMBER_PIXELS_HEIGHT = 512;
046    /** Total image width in µm. */
047    public static final double IMAGE_WIDTH          = PIXEL_WIDTH * NUMBER_PIXELS_WIDTH;
048    /** Total image height in µm. */
049    public static final double IMAGE_HEIGHT         = PIXEL_HEIGHT * NUMBER_PIXELS_HEIGHT;
050}
