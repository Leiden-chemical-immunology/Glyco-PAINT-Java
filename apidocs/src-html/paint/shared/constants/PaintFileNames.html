027package paint.shared.constants;
028
029/**
030 * Centralizes the naming conventions for all primary data files used
031 * within the PAINT application.
032 */
033public final class PaintFileNames {
034
035    private PaintFileNames() {
036        // Prevent instantiation
037    }
038
039    /** Standard name for the recordings metadata CSV file. */
040    public static final String RECORDINGS_CSV                 = "Recordings.csv";
041    /** Standard name for the tracks data CSV file. */
042    public static final String TRACKS_CSV                     = "Tracks.csv";
043    /** Standard name for the squares analysis results CSV file. */
044    public static final String SQUARES_CSV                    = "Squares.csv";
045    /** Standard name for the experiment-level info metadata CSV file. */
046    public static final String EXPERIMENT_INFO_CSV            = "Experiment Info.csv";
047    /** Default filename for the primary project configuration JSON. */
048    public static final String PAINT_CONFIGURATION_JSON       = "Paint Configuration.json";
049    /** Default filename for the parameter sweep configuration JSON. */
050    public static final String PAINT_SWEEP_CONFIGURATION_JSON = "Paint Sweep Configuration.json";
051}
