048package paint.shared.utils;
049
050import com.dd.plist.NSDictionary;
051import com.dd.plist.NSObject;
052import com.dd.plist.PropertyListParser;
053
054import java.io.File;
055
056/**
057 * Provides static, thread-safe access to application preferences for Glyco-PAINT.
058 * <p>
059 * Preferences are stored in {@code ~/Library/Preferences/Glyco-PAINT.plist} using
060 * Apple’s Property List (plist) format. The class handles reading, writing, and
061 * automatic persistence of key-value pairs for various data types.
062 * </p>
063 * <ul>
064 *   <li>Thread-safe access (synchronized methods)</li>
065 *   <li>Automatic saving of preferences after updates</li>
066 *   <li>Reload capability for external modifications</li>
067 *   <li>Fallback to default values if keys are missing</li>
068 * </ul>
069 */
070public final class PaintPrefs {
071
072    /**
073     * Private constructor to prevent instantiation.
074     */
075    private PaintPrefs() {
076        // Deliberately empty
077    }
078
079    // ───────────────────────────────────────────────────────────────────────────────
080    // CONSTANTS AND STATE
081    // ───────────────────────────────────────────────────────────────────────────────
082
083    private static final File PREF_FILE = new File(
084            System.getProperty("user.home"),
085            "Library/Preferences/Glyco-PAINT.plist"
086    );
087
088    private static NSDictionary plist = new NSDictionary();
089
090    static {
091        try {
092            if (PREF_FILE.exists()) {
093                plist = (NSDictionary) PropertyListParser.parse(PREF_FILE);
094            }
095        } catch (Exception e) {
096            PaintLogger.errorf("Could not load plist: " + e.getMessage());
097            plist = new NSDictionary();
098        }
099    }
100
101
102    // ───────────────────────────────────────────────────────────────────────────────
103    // GET METHODS
104    // ───────────────────────────────────────────────────────────────────────────────
105
106    /**
107     * Retrieves a boolean preference value or writes the given default if not present.
108     *
109     * @param section      section name (namespace)
110     * @param key          preference key
111     * @param defaultValue default to return and store if missing
112     * @return stored boolean or default
113     */
114    public static synchronized boolean getBoolean(
115            String section,
116            String key,
117            boolean defaultValue) {
118        NSDictionary sectionDict = (NSDictionary) plist.objectForKey(section);
119        if (sectionDict == null) {
120            sectionDict = new NSDictionary();
121            plist.put(section, sectionDict);
122        }
123        NSObject obj = sectionDict.objectForKey(key);
124        if (obj == null) {
125            sectionDict.put(key, defaultValue);
126            save();
127            return defaultValue;
128        }
129        return Boolean.parseBoolean(obj.toString());
130    }
131
132    /**
133     * Retrieves a string preference value or writes the given default if missing.
134     *
135     * @param section      section name (namespace)
136     * @param key          preference key
137     * @param defaultValue default to return and store if missing
138     * @return stored string or default
139     */
140    public static synchronized String getString(
141            String section,
142            String key,
143            String defaultValue) {
144        NSDictionary sectionDict = (NSDictionary) plist.objectForKey(section);
145        if (sectionDict == null) {
146            sectionDict = new NSDictionary();
147            plist.put(section, sectionDict);
148        }
149        NSObject obj = sectionDict.objectForKey(key);
150        if (obj == null) {
151            sectionDict.put(key, defaultValue);
152            save();
153            return defaultValue;
154        }
155        return obj.toString();
156    }
157
158    // ───────────────────────────────────────────────────────────────────────────────
159    // PUT METHODS (SECTIONED)
160    // ───────────────────────────────────────────────────────────────────────────────
161
162    /**
163     * Stores a boolean value within a named section and persists the change.
164     */
165    public static synchronized void putBoolean(
166            String section,
167            String key,
168            boolean value) {
169        NSDictionary sectionDict = (NSDictionary) plist.objectForKey(section);
170        if (sectionDict == null) {
171            sectionDict = new NSDictionary();
172            plist.put(section, sectionDict);
173        }
174        sectionDict.put(key, value);
175        save();
176    }
177
178    /**
179     * Stores a string value within a named section and persists the change.
180     */
181    public static synchronized void putString(
182            String section,
183            String key,
184            String value) {
185        NSDictionary sectionDict = (NSDictionary) plist.objectForKey(section);
186        if (sectionDict == null) {
187            sectionDict = new NSDictionary();
188            plist.put(section, sectionDict);
189        }
190        sectionDict.put(key, value);
191        save();
192    }
193
194
195    // ───────────────────────────────────────────────────────────────────────────────
196    // FILE MANAGEMENT
197    // ───────────────────────────────────────────────────────────────────────────────
198
199    /**
200     * Saves the current preferences to the plist file.
201     * Logs an error via {@link PaintLogger} if saving fails.
202     */
203    private static synchronized void save() {
204        try {
205            PropertyListParser.saveAsXML(plist, PREF_FILE);
206        } catch (Exception e) {
207            PaintLogger.errorf("Could not save plist: %s", e.getMessage());
208        }
209    }
210}
