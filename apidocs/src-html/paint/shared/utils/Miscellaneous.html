037package paint.shared.utils;
038
039import java.lang.reflect.Field;
040import java.time.Duration;
041
042/**
043 * Static utility class containing miscellaneous helper methods used across PAINT modules.
044 * <p>
045 * Provides methods for time formatting, numeric rounding, and
046 * robust boolean string parsing.
047 */
048public final class Miscellaneous {
049
050    /**
051     * Prevents instantiation of this static utility class.
052     */
053    private Miscellaneous() {
054        // Prevent instantiation
055    }
056
057    // ───────────────────────────────────────────────────────────────────────────────
058    // MAIN (DEMONSTRATION)
059    // ───────────────────────────────────────────────────────────────────────────────
060
061    /**
062     * Demonstration entry point for manual testing.
063     *
064     * @param args command-line arguments (unused)
065     */
066    public static void main(String[] args) {
067        int[] testValues = {0, 5, 59, 60, 61, 65, 3599, 3600, 3605, 3665, 7322};
068        for (int seconds : testValues) {
069            System.out.printf("%5d seconds → %s%n", seconds, formatDuration(seconds));
070        }
071    }
072
073    // ───────────────────────────────────────────────────────────────────────────────
074    // DURATION FORMATTING UTILITIES
075    // ───────────────────────────────────────────────────────────────────────────────
076
077    /**
078     * Formats a duration given in seconds into a human-readable string such as
079     * "1 hour 2 minutes 5 seconds".
080     *
081     * @param totalSeconds duration in seconds
082     * @return formatted human-readable duration string
083     */
084    public static String formatDuration(int totalSeconds) {
085        int hours   = totalSeconds / 3600;
086        int minutes = (totalSeconds % 3600) / 60;
087        int secs    = totalSeconds % 60;
088
089        StringBuilder sb = new StringBuilder();
090        if (hours > 0) {
091            sb.append(hours).append(" hour").append(hours == 1 ? "" : "s").append(" ");
092        }
093        if (minutes > 0) {
094            sb.append(minutes).append(" minute").append(minutes == 1 ? "" : "s").append(" ");
095        }
096        if (secs > 0 || sb.length() == 0) {
097            sb.append(secs).append(" second").append(secs == 1 ? "" : "s");
098        }
099
100        return sb.toString().trim();
101    }
102
103    /**
104     * Converts a {@link Duration} object into a human-readable string.
105     *
106     * @param duration duration instance to format
107     * @return formatted string representation of the duration
108     */
109    public static String formatDuration(Duration duration) {
110        int totalSeconds = (int) duration.getSeconds();
111        return formatDuration(totalSeconds);
112    }
113
114    // ───────────────────────────────────────────────────────────────────────────────
115    // ERROR HANDLING UTILITIES
116    // ───────────────────────────────────────────────────────────────────────────────
117
118    /**
119     * Extracts a concise user-friendly message from a {@link Throwable}.
120     * If no colon is present in the exception message, returns the raw text.
121     *
122     * @param t exception or error object
123     * @return trimmed message string or an empty string if {@code t} is null
124     */
125    public static String friendlyMessage(Throwable t) {
126        if (t == null) {
127            return "";
128        }
129        String m     = t.toString();
130        int    colon = m.lastIndexOf(':');
131        return (colon != -1) ? m.substring(colon + 1).trim() : m;
132    }
133
134    // ───────────────────────────────────────────────────────────────────────────────
135    // NUMERIC UTILITIES
136    // ───────────────────────────────────────────────────────────────────────────────
137
138    /**
139     * Rounds a floating-point number to a specified number of decimal places.
140     *
141     * @param value    numeric value to round
142     * @param decimals number of decimal places
143     * @return rounded numeric value
144     */
145    public static double round(double value, int decimals) {
146        if (decimals < 0) {
147            return value;
148        } else {
149            double scale = Math.pow(10, decimals);
150            return Math.round(value * scale) / scale;
151        }
152    }
153
154    public static void initialiseDoublesToNaN(Object target) {
155        for (Field field : target.getClass().getDeclaredFields()) {
156            if (field.getType() == double.class) {
157                try {
158                    field.setAccessible(true);
159                    field.setDouble(target, Double.NaN);
160                } catch (IllegalAccessException e) {
161                    throw new RuntimeException(e);
162                }
163            }
164        }
165    }
166}
