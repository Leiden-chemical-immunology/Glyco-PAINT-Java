066    /**
067     * Filters track coordinates to include only those within the specified square.
068     * <p>
069     * This method handles inclusive or exclusive boundaries depending on whether
070     * the square is in the last column or row. It operates on a Tablesaw table
071     * containing at least TRACK_X_LOCATION and TRACK_Y_LOCATION columns.
072     * </p>
073     *
074     * @param tracks     table of track data with X/Y coordinate columns
075     * @param square     the square region defining coordinate boundaries
076     * @param lastRowCol index of the last row/column in the grid (for boundary handling)
077     * @return new {@link Table} containing only the tracks located within the specified square
078     */
079    public static Table filterTracksInSquare(Table tracks, Square square, int lastRowCol) {
080        double x0 = square.getX0();
081        double y0 = square.getY0();
082        double x1 = square.getX1();
083        double y1 = square.getY1();
084
085        // See if the square sits at the right or bottom border of the grid.
086        boolean isLastCol = square.getColNumber() == lastRowCol;
087        boolean isLastRow = square.getRowNumber() == lastRowCol;
088
089        // Ensures that left < right and top < bottom even if coordinates were swapped.
090        double left   = Math.min(x0, x1);
091        double right  = Math.max(x0, x1);
092        double top    = Math.min(y0, y1);
093        double bottom = Math.max(y0, y1);
094
095        // Get the x,y coordinates of the tracks
096        DoubleColumn x = tracks.doubleColumn(TRACK_X_LOCATION);
097        DoubleColumn y = tracks.doubleColumn(TRACK_Y_LOCATION);
098
099        // For normal squares → take left ≤ x < right
100            // For rightmost column → take left ≤ x ≤ right (inclusive on both sides)
101        Selection selX = isLastCol
102                ? x.isGreaterThanOrEqualTo(left).and(x.isLessThanOrEqualTo(right))
103                : x.isGreaterThanOrEqualTo(left).and(x.isLessThan(right));
104
105        // For normal squares → top ≤ y < bottom
106                // For bottom squares → top ≤ y ≤ bottom
107        Selection selY = isLastRow
108                ? y.isGreaterThanOrEqualTo(top).and(y.isLessThanOrEqualTo(bottom))
109                : y.isGreaterThanOrEqualTo(top).and(y.isLessThan(bottom));
110
111        return tracks.where(selX.and(selY));
112    }

118    /**
119     * Applies the full visibility filter to squares belonging to a specific recording.
120     * <p>
121     * The method performs a two-stage filtering process:
122     *
123     * <h3>1. Numeric filter</h3>
124     * A square becomes visible if all of the following are true:
125     * <ul>
126     *     <li>its R² value is not NaN and ≥ {@code minRequiredRSquared}</li>
127     *     <li>its density ratio ≥ {@code minRequiredDensityRatio}</li>
128     *     <li>its variability ≤ {@code maxAllowableVariability}</li>
129     * </ul>
130     * If {@code recordingName} is non-null, only squares whose
131     * {@code getRecordingName()} matches the given name are evaluated;
132     * others are skipped.
133     *
134     * <h3>2. Neighbour filter</h3>
135     * Applied only when {@code neighbourMode} is not {@code "Free"}.
136     * A square that passed the numeric filter will remain visible only if
137     * it has at least one other <em>visible</em> neighbouring square.
138     * Neighbourhood rules:
139     * <ul>
140     *   <li><b>"Relaxed"</b>: any of the 8 surrounding squares (edge or corner) counts as neighbouring</li>
141     *   <li><b>"Strict"</b>: only 4-connected neighbours (up, down, left, right) count</li>
142     * </ul>
143     * If {@code neighbourMode} is neither {@code "Free"}, {@code "Relaxed"},
144     * nor {@code "Strict"}, an {@link IllegalArgumentException} is thrown.
145     *
146     * <h3>Final step</h3>
147     * Any square not selected during the neighbour filter is set to invisible.
148     *
149     * @param squares                    the squares to filter
150     * @param recordingName              if non-null, only squares from this recording are evaluated;
151     *                                   if null, all squares are processed
152     * @param minRequiredDensityRatio    minimum density ratio required to be visible
153     * @param maxAllowableVariability    maximum variability allowed to be visible
154     * @param minRequiredRSquared        minimum R² required to be visible
155     * @param neighbourMode              neighbour rule: {@code "Free"}, {@code "Relaxed"}, or {@code "Strict"}
156     */
157    public static void applyVisibilityFilterOnRecording(
158            List<Square> squares,
159            String       recordingName,
160            double       minRequiredDensityRatio,
161            double       maxAllowableVariability,
162            double       minRequiredRSquared,
163            String       neighbourMode) {
164
165        if (squares == null || squares.isEmpty()) {
166            return;
167        }
168
169        // Build the subset we will actually process.
170        List<Square> target = new ArrayList<>();
171        if (recordingName == null) {
172            target.addAll(squares);
173        } else {
174            for (Square s : squares) {
175                if (recordingName.equals(s.getRecordingName())) {
176                    target.add(s);
177                }
178            }
179        }
180
181        if (target.isEmpty()) {
182            return;
183        }
184
185        int visibleBasic = 0;
186
187        // Pass 1 — Numeric filter (ONLY on target)
188        for (Square square : target) {
189            boolean passes =
190                    !Double.isNaN(square.getRSquared()) &&
191                            square.getDensityRatio() >= minRequiredDensityRatio &&
192                            square.getVariability()  <= maxAllowableVariability &&
193                            square.getRSquared()     >= minRequiredRSquared;
194
195            square.setVisible(passes);
196            if (passes) {
197                visibleBasic++;
198            }
199        }
200
201        // Pass 2 — Neighbour-based refinement (ONLY on target)
202        if ("Free".equalsIgnoreCase(neighbourMode)) {
203            return;
204        }
205
206        boolean relaxed = "Relaxed".equalsIgnoreCase(neighbourMode);
207        boolean strict  = "Strict".equalsIgnoreCase(neighbourMode);
208        if (!relaxed && !strict) {
209            throw new IllegalArgumentException("Invalid neighbourMode: " + neighbourMode);
210        }
211
212        Set<Square> keep      = new HashSet<>();
213        int         keptCount = 0;
214
215        for (Square square : target) {
216            if (!square.isVisible()) {
217                continue;
218            }
219
220            boolean hasNeighbour = false;
221            int rowNumber = square.getRowNumber();
222            int colNumber = square.getColNumber();
223
224            for (Square other : target) {
225                if (other == square || !other.isVisible()) {
226                    continue;
227                }
228
229                int differenceInRow    = Math.abs(other.getRowNumber() - rowNumber);
230                int differenceInColumn = Math.abs(other.getColNumber() - colNumber);
231
232                if (relaxed) {
233                    if (differenceInRow <= 1 && differenceInColumn <= 1) {
234                        hasNeighbour = true;
235                        break;
236                    }
237                } else { // strict
238                    if ((differenceInRow == 1 && differenceInColumn == 0) || (differenceInRow == 0 && differenceInColumn == 1)) {
239                        hasNeighbour = true;
240                        break;
241                    }
242                }
243            }
244
245            if (hasNeighbour) {
246                keep.add(square);
247                keptCount++;
248            }
249        }
250
251        // Apply neighbour filtering result (ONLY on target)
252        for (Square square : target) {
253            if (!keep.contains(square)) {
254                square.setVisible(false);
255            }
256        }
257
258        PaintLogger.debugf("NeighbourMode [%s] neighbour-filtered: %d / %d retained",
259                           neighbourMode, keptCount, visibleBasic);
260    }

263    /**
264     * Applies the visibility filter to all squares in a recording using the given
265     * numeric thresholds and neighbour-mode rules.
266     * <p>
267     * This is a convenience method that delegates to
268     * {@link #applyVisibilityFilterOnRecording(List, String, double, double, double, String)}
269     * without restricting to a specific recording name (i.e., it applies the filter
270     * to all squares in the list).
271     *
272     * <h3>Filtering steps</h3>
273     * <ol>
274     *   <li><b>Numeric filter:</b>
275     *       A square becomes visible if:
276     *       <ul>
277     *         <li>its R² is not NaN and ≥ {@code minRSquared}</li>
278     *         <li>its density ratio ≥ {@code minDensityRatio}</li>
279     *         <li>its variability ≤ {@code maxVariability}</li>
280     *       </ul>
281     *   </li>
282     *   <li><b>Neighbour filter:</b>
283     *       Only applied when {@code neighbourMode} is not {@code "Free"}.
284     *       <ul>
285     *         <li>{@code "Relaxed"} → any touching neighbour (8-connected)</li>
286     *         <li>{@code "Strict"} → edge-adjacent neighbour only (4-connected)</li>
287     *       </ul>
288     *   </li>
289     * </ol>
290     *
291     * @param squares        the list of squares to filter
292     * @param minDensityRatio minimum required density ratio
293     * @param maxVariability   maximum allowable variability
294     * @param minRSquared      minimum required R²
295     * @param neighbourMode    neighbour filtering mode:
296     *                         {@code "Free"}, {@code "Relaxed"}, or {@code "Strict"}
297     */
298    public static void applyVisibilityFilter(
299            List<Square> squares,
300            double       minDensityRatio,
301            double       maxVariability,
302            double       minRSquared,
303            String       neighbourMode) {
304        applyVisibilityFilterOnRecording(squares, null, minDensityRatio, maxVariability,  minRSquared, neighbourMode);
305    }

308    /**
309     * Retrieves a list of tracks from all selected squares in the provided recording.
310     * If a square is selected and contains tracks, those tracks are added to the result.
311     *
312     * @param squares The squares to be checked.
313     * @return A list of tracks from all selected squares in the recording.
314     */
315    public static List<Track> getTracksFromSelectedSquares(List<Square> squares) {
316        List<Track> selectedTracks = new ArrayList<>();
317        for (Square square : squares) {
318            if (square.isVisible() && square.getTracks() != null) {
319                selectedTracks.addAll(square.getTracks());
320            }
321        }
322        return selectedTracks;
323    }

325    /**
326     * Counts the number of squares marked as selected in the provided recording.
327     *
328     * @param recording The recording object containing a collection of squares.
329     * @return The count of squares that are marked as selected.
330     */
331    public static int getNumberOfSelectedSquares(Recording recording) {
332        int count = 0;
333        for (Square square : recording.getSquaresOfRecording()) {
334            if (square.isVisible()) {
335                count++;
336            }
337        }
338        return count;
339    }

341    /**
342     * Calculates the density of tracks in a specified area over time and considering a concentration factor.
343     * <p>
344     * The density is computed as the number of tracks divided by the area,
345     * then divided by the time, and finally divided by the concentration.
346     * If any of the area, time, or concentration values are non-positive, an IllegalArgumentException is thrown.
347     *
348     * @param nrTracks      The total number of tracks to calculate density for.
349     * @param area          The area over which the density is calculated. Must be positive.
350     * @param time          The time period over which the density is calculated. Must be positive.
351     * @param concentration The concentration factor to apply to the density calculation. Must be positive.
352     * @return The calculated density as a double.
353     * @throws IllegalArgumentException if area, time, or concentration is less than or equal to zero.
354     */
355    public static double calculateDensity(int nrTracks, double area, double time, double concentration) {
356        if (area <= 0 || time <= 0 || concentration <= 0) {
357            throw new IllegalArgumentException("Area, time, and concentration must be positive");
358        }
359
360        double density  = nrTracks / area;
361        density        /= time;
362        density        /= concentration;
363
364        return density;
365    }
