028package paint.shared.io.internal;
029
030import paint.shared.objects.Recording;
031
032import tech.tablesaw.api.ColumnType;
033import tech.tablesaw.api.Row;
034import tech.tablesaw.api.Table;
035
036import java.util.ArrayList;
037import java.util.List;
038
039import static paint.shared.constants.PaintStringConstants.*;
040
041public class RecordingsTableIO extends BaseTableIO {
042
043    // =====================================================================
044    //  INTERNAL HELPERS (extract schema from Recording.Column)
045    // =====================================================================
046
047    private String[] getColumnHeaders() {
048        Recording.Column[] cols = Recording.Column.values();
049        String[] headers = new String[cols.length];
050        for (int i = 0; i < cols.length; i++) {
051            headers[i] = cols[i].header;
052        }
053        return headers;
054    }
055
056    private ColumnType[] getColumnTypes() {
057        Recording.Column[] cols = Recording.Column.values();
058        ColumnType[] types = new ColumnType[cols.length];
059        for (int i = 0; i < cols.length; i++) {
060            types[i] = cols[i].type;
061        }
062        return types;
063    }
064
065    // =====================================================================
066    //  TABLE CREATION
067    // =====================================================================
068
069    /**
070     * @return a new empty {@link Table} with the Recordings schema.
071     */
072    public Table emptyTable() {
073        return newEmptyTable(
074                "Recordings",
075                getColumnHeaders(),
076                getColumnTypes()
077        );
078    }
079
080    // =====================================================================
081    //  ENTITY → TABLE CONVERSION
082    // =====================================================================
083
084    /**
085     * Converts a list of {@link Recording} entities into a Tablesaw {@link Table}.
086     *
087     * @param recordings list of recording objects to convert
088     * @return a table containing the recording data
089     */
090    public Table toTable(List<Recording> recordings) {
091        Table table = emptyTable();
092
093        for (Recording recording : recordings) {
094            Row row = table.appendRow();
095
096            row.setString(  EXPERIMENT_NAME,                 recording.getExperimentName());
097            row.setString(  RECORDING_NAME,                  recording.getRecordingName());
098            row.setInt(     CONDITION_NUMBER,                recording.getConditionNumber());
099            row.setInt(     REPLICATE_NUMBER,                recording.getReplicateNumber());
100            row.setString(  PROBE_NAME,                      recording.getProbeName());
101            row.setString(  PROBE_TYPE,                      recording.getProbeType());
102            row.setString(  CELL_TYPE,                       recording.getCellType());
103            row.setString(  ADJUVANT,                        recording.getAdjuvant());
104            row.setDouble(  CONCENTRATION,                   recording.getConcentration());
105            row.setBoolean( PROCESS_FLAG,                    recording.isProcessFlagSet());
106            row.setDouble(  THRESHOLD,                       recording.getThreshold());
107            row.setInt(     NUMBER_OF_SPOTS,                 recording.getNumberOfSpots());
108            row.setInt(     NUMBER_OF_TRACKS,                recording.getNumberOfTracks());
109            row.setInt(     NUMBER_OF_SQUARES_IN_BACKGROUND, recording.getNumberOfSquaresInBackground());
110            row.setInt(     NUMBER_OF_TRACKS_IN_BACKGROUND,  recording.getNumberOfTracksInBackground());
111            row.setDouble(  AVERAGE_TRACKS_IN_BACKGROUND,    recording.getAverageTracksInBackGround());
112            row.setInt(     NUMBER_OF_SPOTS_IN_ALL_TRACKS,   recording.getNumberOfSpotsInAllTracks());
113            row.setInt(     NUMBER_OF_FRAMES,                recording.getNumberOfFrames());
114            row.setDouble(  RUN_TIME,                        recording.getRunTime());
115            row.setDateTime(TIME_STAMP,                      recording.getTimeStamp());
116            row.setBoolean( EXCLUDE,                         recording.isExcluded());
117            row.setDouble(  TAU,                             recording.getTau());
118            row.setDouble(  R_SQUARED,                       recording.getRSquared());
119            row.setDouble(  DENSITY,                         recording.getDensity());
120            row.setDouble(  MIN_REQUIRED_DENSITY_RATIO,      recording.getMinRequiredDensityRatio());
121            row.setDouble(  MIN_REQUIRED_R_SQUARED,          recording.getMinRequiredRSquared());
122            row.setDouble(  MAX_ALLOWABLE_VARIABILITY,       recording.getMaxAllowableVariability());
123            row.setString(  NEIGHBOUR_MODE,                  recording.getNeighbourMode());
124        }
125
126        return table;
127    }
128
129    // =====================================================================
130    //  TABLE → ENTITY CONVERSION
131    // =====================================================================
132
133    /**
134     * Converts a Tablesaw {@link Table} into a list of {@link Recording} entities.
135     *
136     * @param table the table to convert
137     * @return a list of recording objects
138     */
139    public List<Recording> toEntities(Table table) {
140        List<Recording> recordings = new ArrayList<>();
141
142        for (Row tablesawRow : table) {
143            Recording recording = new Recording();
144
145            recording.setExperimentName(              tablesawRow.getString(   EXPERIMENT_NAME));
146            recording.setRecordingName(               tablesawRow.getString(   RECORDING_NAME));
147            recording.setConditionNumber(             tablesawRow.getInt(      CONDITION_NUMBER));
148            recording.setReplicateNumber(             tablesawRow.getInt(      REPLICATE_NUMBER));
149            recording.setProbeName(                   tablesawRow.getString(   PROBE_NAME));
150            recording.setProbeType(                   tablesawRow.getString(   PROBE_TYPE));
151            recording.setCellType(                    tablesawRow.getString(   CELL_TYPE));
152            recording.setAdjuvant(                    tablesawRow.getString(   ADJUVANT));
153            recording.setConcentration(               tablesawRow.getDouble(   CONCENTRATION));
154            recording.setProcessFlag(                 tablesawRow.getBoolean(  PROCESS_FLAG));
155            recording.setThreshold(                   tablesawRow.getDouble(   THRESHOLD));
156            recording.setNumberOfSpots(               tablesawRow.getInt(      NUMBER_OF_SPOTS));
157            recording.setNumberOfTracks(              tablesawRow.getInt(      NUMBER_OF_TRACKS));
158            recording.setNumberOfSquaresInBackground( tablesawRow.getInt(      NUMBER_OF_SQUARES_IN_BACKGROUND));
159            recording.setNumberOfTracksInBackground(  tablesawRow.getInt(      NUMBER_OF_TRACKS_IN_BACKGROUND));
160            recording.setAverageTracksInBackGround(   tablesawRow.getDouble(   AVERAGE_TRACKS_IN_BACKGROUND));
161            recording.setNumberOfSpotsInAllTracks(    tablesawRow.getInt(      NUMBER_OF_SPOTS_IN_ALL_TRACKS));
162            recording.setNumberOfFrames(              tablesawRow.getInt(      NUMBER_OF_FRAMES));
163            recording.setRunTime(                     tablesawRow.getDouble(   RUN_TIME));
164            recording.setTimeStamp(                   tablesawRow.getDateTime( TIME_STAMP));
165            recording.setExcluded(                    tablesawRow.getBoolean(  EXCLUDE));
166            recording.setTau(                         tablesawRow.getDouble(   TAU));
167            recording.setRSquared(                    tablesawRow.getDouble(   R_SQUARED));
168            recording.setDensity(                     tablesawRow.getDouble(   DENSITY));
169            recording.setMinRequiredDensityRatio(     tablesawRow.getDouble(   MIN_REQUIRED_DENSITY_RATIO));
170            recording.setMinRequiredRSquared(         tablesawRow.getDouble(   MIN_REQUIRED_R_SQUARED));
171            recording.setMaxAllowableVariability(     tablesawRow.getDouble(   MAX_ALLOWABLE_VARIABILITY));
172            recording.setNeighbourMode(               tablesawRow.getString(   NEIGHBOUR_MODE));
173
174            recordings.add(recording);
175        }
176
177        return recordings;
178    }
179}
