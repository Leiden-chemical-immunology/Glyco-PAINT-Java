036package paint.shared.io.internal;
037
038import static paint.shared.constants.PaintStringConstants.*;
039
040import paint.shared.io.MainIOInterface;
041import paint.shared.objects.Track;
042
043import tech.tablesaw.api.ColumnType;
044import tech.tablesaw.api.Row;
045import tech.tablesaw.api.Table;
046import tech.tablesaw.columns.Column;
047
048import java.util.ArrayList;
049import java.util.List;
050
051/**
052 * Internal schema-validated table I/O implementation for {@link Track}.
053 *
054 * <p>This class supports CSV reading, strict schema enforcement, entity-row
055 * conversion, and table append operations. External callers must use
056 * {@link MainIOInterface}.</p>
057 */
058public class TracksTableIO extends BaseTableIO {
059
060    // =========================================================================
061    //  INTERNAL HELPERS (schema extracted from Track.Column)
062    // =========================================================================
063
064    private String[] getColumnHeaders() {
065        Track.Column[] cols = Track.Column.values();
066        String[] headers = new String[cols.length];
067        for (int i = 0; i < cols.length; i++) {
068            headers[i] = cols[i].header;
069        }
070        return headers;
071    }
072
073    private ColumnType[] getColumnTypes() {
074        Track.Column[] cols = Track.Column.values();
075        ColumnType[] types = new ColumnType[cols.length];
076        for (int i = 0; i < cols.length; i++) {
077            types[i] = cols[i].type;
078        }
079        return types;
080    }
081
082    // =====================================================================
083    //  TABLE CREATION
084    // =====================================================================
085
086    /**
087     * @return a new empty {@link Table} with the Tracks schema.
088     */
089    public Table emptyTable() {
090        return newEmptyTable(
091                "Tracks",
092                getColumnHeaders(),
093                getColumnTypes()
094        );
095    }
096
097    // =====================================================================
098    //  ENTITY → TABLE CONVERSION
099    // =====================================================================
100
101    /**
102     * Converts a list of {@link Track} entities into a Tablesaw {@link Table}.
103     *
104     * @param tracks list of track objects to convert
105     * @return a table containing the track data
106     */
107    public Table toTable(List<Track> tracks) {
108        Table table = emptyTable();
109
110        for (Track track : tracks) {
111            Row tablesawRow = table.appendRow();
112
113            tablesawRow.setString( UNIQUE_KEY,                track.getUniqueKey());
114            tablesawRow.setString( EXPERIMENT_NAME,           track.getExperimentName());
115            tablesawRow.setString( RECORDING_NAME,            track.getRecordingName());
116            tablesawRow.setInt(    TRACK_ID,                  track.getTrackId());
117            tablesawRow.setInt(    NUMBER_OF_SPOTS,           track.getNumberOfSpots());
118            tablesawRow.setInt(    NUMBER_OF_GAPS,            track.getNumberOfGaps());
119            tablesawRow.setInt(    LONGEST_GAP,               track.getLongestGap());
120            tablesawRow.setDouble( TRACK_DURATION,            track.getTrackDuration());
121            tablesawRow.setDouble( TRACK_X_LOCATION,          track.getTrackXLocation());
122            tablesawRow.setDouble( TRACK_Y_LOCATION,          track.getTrackYLocation());
123            tablesawRow.setDouble( TRACK_DISPLACEMENT,        track.getTrackDisplacement());
124            tablesawRow.setDouble( TRACK_MAX_SPEED,           track.getTrackMaxSpeed());
125            tablesawRow.setDouble( TRACK_MEDIAN_SPEED,        track.getTrackMedianSpeed());
126            tablesawRow.setDouble( DIFFUSION_COEFFICIENT,     track.getDiffusionCoefficient());
127            tablesawRow.setDouble( DIFFUSION_COEFFICIENT_EXT, track.getDiffusionCoefficientExt());
128            tablesawRow.setDouble( TOTAL_DISTANCE,            track.getTotalDistance());
129            tablesawRow.setDouble( CONFINEMENT_RATIO,         track.getConfinementRatio());
130            tablesawRow.setInt(    SQUARE_NUMBER,             track.getSquareNumber());
131            tablesawRow.setInt(    LABEL_NUMBER,              track.getLabelNumber());
132        }
133
134        return table;
135    }
136
137    // =====================================================================
138    //  TABLE → ENTITY CONVERSION
139    // =====================================================================
140
141    /**
142     * Converts a validated Tracks {@link Table} into a list of {@link Track} entities.
143     *
144     * @param table a schema-validated table
145     * @return list of Track objects
146     */
147    public List<Track> toEntities(Table table) {
148        List<Track> tracks = new ArrayList<>();
149
150        for (Row row : table) {
151            Track track = new Track();
152
153            track.setUniqueKey(              row.getString( UNIQUE_KEY));
154            track.setExperimentName(         row.getString( EXPERIMENT_NAME));
155            track.setRecordingName(          row.getString( RECORDING_NAME));
156            track.setTrackId(                row.getInt(    TRACK_ID));
157            track.setNumberOfSpots(          row.getInt(    NUMBER_OF_SPOTS));
158            track.setNumberOfGaps(           row.getInt(    NUMBER_OF_GAPS));
159            track.setLongestGap(             row.getInt(    LONGEST_GAP));
160            track.setTrackDuration(          row.getDouble( TRACK_DURATION));
161            track.setTrackXLocation(         row.getDouble( TRACK_X_LOCATION));
162            track.setTrackYLocation(         row.getDouble( TRACK_Y_LOCATION));
163            track.setTrackDisplacement(      row.getDouble( TRACK_DISPLACEMENT));
164            track.setTrackMaxSpeed(          row.getDouble( TRACK_MAX_SPEED));
165            track.setTrackMedianSpeed(       row.getDouble( TRACK_MEDIAN_SPEED));
166            track.setDiffusionCoefficient(   row.getDouble( DIFFUSION_COEFFICIENT));
167            track.setDiffusionCoefficientExt(row.getDouble( DIFFUSION_COEFFICIENT_EXT));
168            track.setTotalDistance(          row.getDouble( TOTAL_DISTANCE));
169            track.setConfinementRatio(       row.getDouble( CONFINEMENT_RATIO));
170            track.setSquareNumber(           row.getInt(    SQUARE_NUMBER));
171            track.setLabelNumber(            row.getInt(    LABEL_NUMBER));
172
173            tracks.add(track);
174        }
175
176        return tracks;
177    }
178
179    // =====================================================================
180    //  APPEND / MERGE
181    // =====================================================================
182
183    /**
184     * Appends all rows from {@code source} into {@code target} while enforcing
185     * the Tracks schema and preserving missing values.
186     *
187     * @param target the destination table
188     * @param source the source table
189     */
190    public void appendInPlace(Table target, Table source) {
191        for (Row row : source) {
192            Row newRow = target.appendRow();
193
194            for (Track.Column colEnum : Track.Column.values()) {
195                String col = colEnum.header;
196                Column<?> targetCol = target.column(col);
197
198                if (targetCol.type() == ColumnType.STRING) {
199                    newRow.setString(col, row.getString(col));
200                } else if (targetCol.type() == ColumnType.INTEGER) {
201                    newRow.setInt(col, row.getInt(col));
202                } else if (targetCol.type() == ColumnType.DOUBLE) {
203                    newRow.setDouble(col, row.getDouble(col));
204                } else if (targetCol.type() == ColumnType.BOOLEAN) {
205                    newRow.setBoolean(col, row.getBoolean(col));
206                }
207            }
208        }
209    }
