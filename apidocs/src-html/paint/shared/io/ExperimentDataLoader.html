036package paint.shared.io;
037
038import paint.shared.objects.Experiment;
039import paint.shared.objects.Recording;
040import paint.shared.objects.Square;
041import paint.shared.utils.PaintLogger;
042import tech.tablesaw.api.Table;
043
044import javax.swing.*;
045import java.nio.file.Path;
046import java.util.List;
047
048import static paint.shared.io.MainIOInterface.readTracksTable;
049import static paint.shared.io.MainIOInterface.readRecordings;
050import static paint.shared.io.MainIOInterface.readSquaresTable;
051import static paint.shared.io.MainIOInterface.trackTableToList;
052import static paint.shared.io.MainIOInterface.squareTableToList;
053
054import static paint.shared.utils.SharedSquareUtils.filterTracksInSquare;
055
056import static paint.shared.constants.PaintFileNames.RECORDINGS_CSV;
057import static paint.shared.constants.PaintFileNames.TRACKS_CSV;
058import static paint.shared.constants.PaintFileNames.SQUARES_CSV;
059
060import static paint.shared.constants.PaintStringConstants.*;
061
062/**
063 * Provides centralized functionality for loading experiment-related data:
064 * recordings, optionally tracks, and optionally squares.
065 */
066public final class ExperimentDataLoader {
067
068    private ExperimentDataLoader() {}
069
070    /**
071     * Loads an experiment with flexible inclusion of tracks and squares.
072     *
073     *
074     * @param projectPath     the root project directory
075     * @param experimentName  the name of the experiment folder
076     * @param loadSquares     whether to include squares (mature project mode)
077     * @param loadTracks      whether to include track data
078     * @return an {@link Experiment} with requested data layers loaded, or {@code null} on failure
079     */
080    public static Experiment loadExperiment(Path    projectPath,
081            String  experimentName,
082            boolean loadSquares,
083            boolean loadTracks) {
084
085        Path experimentPath   = projectPath.resolve(experimentName);
086        Experiment experiment = new Experiment(experimentName);
087
088        // ─── Recordings ───────────────────────────────────────────────────────
089        List<Recording> recordings;
090        try {
091            recordings = readRecordings(experimentPath);
092            for (Recording recording : recordings) {
093                experiment.addRecording(recording);
094            }
095        } catch (Exception e) {
096            PaintLogger.errorf("Failed to read %s in %s", RECORDINGS_CSV, experimentName);
097            return null;
098        }
099
100        // ─── Tracks (optional) ────────────────────────────────────────────────
101        Table         tracksTable = null;
102
103        if (loadTracks) {
104            try {
105                tracksTable = readTracksTable(experimentPath);
106            } catch (Exception e) {
107                PaintLogger.errorf("Failed to read %s in %s", TRACKS_CSV, experimentName);
108                return null;
109            }
110
111            PaintLogger.debugf("Found %d tracks", tracksTable.rowCount());
112
113            for (Recording recording : recordings) {
114                if (!recording.isProcessFlagSet()) {
115                    continue;
116                }
117
118                Table recTracksTable = tracksTable.where(
119                        tracksTable.stringColumn(RECORDING_NAME)
120                                   .isEqualTo(recording.getRecordingName()));
121
122                PaintLogger.debugf("Found %d tracks for recording '%s'",
123                                   recTracksTable.rowCount(), recording.getRecordingName());
124
125                recording.setTracksList(trackTableToList(recTracksTable));
126                recording.setTracksTable(recTracksTable);
127            }
128        }
129
130        // ─── Squares (optional) ───────────────────────────────────────────────
131        if (loadSquares) {
132            Table squaresTable;
133            try {
134                squaresTable = readSquaresTable(experimentPath);
135            } catch (Exception e) {
136                PaintLogger.errorf("Failed to read %s in %s", SQUARES_CSV, experimentName);
137                return null;
138            }
139
140            int numberOfRecordingsToProcess =
141                    (int) recordings.stream()
142                                    .filter(Recording::isProcessFlagSet)
143                                    .count();
144            int numberOfSquares             = squaresTable.rowCount();
145            int numberOfSquaresPerRecording = (numberOfRecordingsToProcess == 0) ? 0 : numberOfSquares / numberOfRecordingsToProcess;
146
147            int numberOfRows = (numberOfSquaresPerRecording > 0)
148                    ? (int) Math.round(Math.sqrt(numberOfSquaresPerRecording))
149                    : 0;
150
151            if (numberOfRows > 0 && numberOfRows * numberOfRows != numberOfSquaresPerRecording) {
152                String msg = "Invalid squares layout in experiment '" + experimentName + "'";
153                PaintLogger.errorf(msg);
154
155                JOptionPane.showMessageDialog(
156                        null,
157                        msg,
158                        "Invalid Configuration",
159                        JOptionPane.ERROR_MESSAGE
160                );
161            }
162
163            for (Recording rec : recordings) {
164                Table recSquares = squaresTable.where(
165                        squaresTable.stringColumn(RECORDING_NAME)
166                                    .matchesRegex("^" + rec.getRecordingName() + "(?:-threshold-\\d{1,3})?$"));
167
168                rec.addSquares(squareTableToList(recSquares));
169
170                // Only map tracks into squares if tracks were loaded
171                if (loadTracks && numberOfRows > 0) {
172                    int   lastRowCol = numberOfRows - 1;
173                    Table recTracks  = rec.getTracksTable();
174
175                    for (Square square : rec.getSquaresOfRecording()) {
176                        Table SquaresTracks = filterTracksInSquare(recTracks, square, lastRowCol);
177                        square.setTracksList(trackTableToList(SquaresTracks));
178                    }
179                }
180            }
181        }
182
183        return experiment;
184    }
185}
