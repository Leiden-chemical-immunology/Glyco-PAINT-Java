003 *  Package:      paint.fiji.trackmate
004 *
005 *  PURPOSE:
006 *    Executes the TrackMate tracking pipeline for a single microscopy recording.
007 *    Handles image loading, brightfield snapshot generation, TrackMate
008 *    configuration setup, execution, visualization, and CSV result export.
009 *
010 *  DESCRIPTION:
011 *    • Loads the ND2 image and optional brightfield reference.
012 *    • Configures and runs TrackMate in deterministic, headless mode.
013 *    • Applies spot and track filtering based on configuration parameters.
014 *    • Exports per-recording images and tracking data to the experiment directory.
015 *    • Supports cancellation and safe cleanup of ImagePlus instances.
016 *
017 *  RESPONSIBILITIES:
018 *    • Manage end-to-end TrackMate processing for a single recording.
019 *    • Integrate Paint configuration and runtime parameters.
020 *    • Handle user cancellation gracefully during processing.
021 *    • Collect and summarize analysis results in {@link TrackMateResults}.
022 *
023 *  USAGE EXAMPLE:
024 *    TrackMateResults result = RunTrackMateOnRecording.runTrackMateOnRecording(
025 *        experimentPath,
026 *        imagesPath,
027 *        trackMateConfig,

029 *        experimentInfo,
030 *        dialog);

083/**
084 * Provides functionality to run TrackMate analysis for a single microscopy
085 * recording. Handles setup, detection, tracking, visualization, and export.
086 * <p>
087 * This class is typically invoked by {@link RunTrackMateOnExperiment} as part
088 * of batch processing within a Paint project.
089 * </p>
090 */
091public class RunTrackMateOnRecording {
092
093    /**
094     * Executes the TrackMate pipeline on a given recording.
095     * Loads the ND2 image, applies analysis parameters, and writes results.
096     *
097     * @param experimentPath       directory where results will be written
098     * @param imagesPath           directory containing the ND2 image
099     * @param trackMateConfig      configuration object with TrackMate parameters
100     * @param threshold            threshold for spot detection
101     * @param experimentInfoRecord metadata describing this recording
102     * @param dialog               optional dialog for user cancellation
103     * @return {@link TrackMateResults} containing analysis statistics,
104     *         or a cancellation result if aborted
105     */
106    public static TrackMateResults runTrackMateOnRecording(Path experimentPath,
107            Path            imagesPath,
108            TrackMateConfig trackMateConfig,
109            double          threshold,
110            ExperimentInfo  experimentInfoRecord,
111            ProjectDialog   dialog) {
112
113        LocalDateTime start = LocalDateTime.now();
114        DebugTools.setRootLevel("OFF");
115
116        ImagePlus imp            = null;
117        ImagePlus impBrightfield = null;
118        ImagePlus capture        = null;
119
120        try {
121            // -----------------------------------------------------------------
122            // Step 1 – Load ND2 image
123            // -----------------------------------------------------------------
124            PaintLogger.raw("                       TrackMate - Image Loading:   ");
125            if (isCancelled(Thread.currentThread(), dialog)) {
126                return cancelEarly(imp, impBrightfield);
127            }
128
129            File nd2File = new File(imagesPath.toFile(), experimentInfoRecord.getRecordingName() + ".nd2");
130            if (!nd2File.exists()) {
131                if (!isCancelled(Thread.currentThread(), dialog)) {
132                    PaintLogger.errorf("Could not open image file: %s", nd2File.getAbsolutePath());
133                }
134                return cancelEarly(imp, impBrightfield);
135            }
136
137            try {
138                imp = IJ.openImage(nd2File.getAbsolutePath());
139            } catch (Exception e) {
140                if (!isCancelled(Thread.currentThread(), dialog)) {
141                    PaintLogger.errorf("Could not load image file: %s", nd2File.getAbsolutePath());
142                }
143                return cancelEarly(imp, impBrightfield);
144            }
145
146            if (isCancelled(Thread.currentThread(), dialog)) {
147                return cancelEarly(imp, impBrightfield);
148            }
149
150            if (imp == null) {
151                if (!isCancelled(Thread.currentThread(), dialog)) {
152                    PaintLogger.errorf("Unsupported format or file not found: %s", nd2File);
153                }
154                return cancelEarly(imp, impBrightfield);
155            }
156
157            imp.show();
158            IJ.wait(100);
159
160            IJ.run(imp, "Enhance Contrast", "saturated=0.35");
161            IJ.run("Grays");
162
163            // -----------------------------------------------------------------
164            // Step 2 – Save Brightfield snapshot
165            // -----------------------------------------------------------------
166            Path jpgPath = experimentPath.resolve("Brightfield Images")
167                                         .resolve(experimentInfoRecord.getRecordingName() + ".jpg");
168            if (Files.notExists(jpgPath.getParent())) {
169                Files.createDirectories(jpgPath.getParent());
170            }
171            if (isCancelled(Thread.currentThread(), dialog)) {
172                return cancelEarly(imp, impBrightfield);
173            }
174
175            Path   brightFieldPath = null;
176            String baseName        = experimentInfoRecord.getRecordingName();
177
178            List<String> candidates = Arrays.asList(
179                    baseName + "-BF.nd2",
180                    baseName + "-BF1.nd2",
181                    baseName + "-BF2.nd2");
182            try {
183                brightFieldPath = candidates.stream()
184                                            .map(imagesPath::resolve)
185                                            .filter(Files::exists)
186                                            .findFirst()
187                                            .orElse(null);
188            } catch (Exception e) {
189                PaintLogger.errorf("Could not locate brightfield file: %s", candidates);
190            }
191
192            if (brightFieldPath == null || Files.notExists(brightFieldPath)) {
193                PaintLogger.warnf("      Could not open brightfield file: %s",
194                                  brightFieldPath == null ? "none found" : brightFieldPath.toString());
195            } else {
196                try {
197                    impBrightfield = IJ.openImage(brightFieldPath.toString());
198                    IJ.run(impBrightfield, "Enhance Contrast", "saturated=0.35");
199                    IJ.saveAs(impBrightfield, "Jpeg", jpgPath.toString());
200                } catch (Exception e) {
201                    PaintLogger.errorf("Error handling brightfield file: %s", e.getMessage());
202                }
203            }
204
205            // -----------------------------------------------------------------
206            // Step 3 – Configure TrackMate
207            // -----------------------------------------------------------------
208            Model model = new Model();
209            model.setLogger(Logger.VOID_LOGGER);
210
211            Settings settings = new Settings(imp);
212            // Detector configuration
213            settings.detectorFactory  = new LogDetectorFactory();
214            settings.detectorSettings = settings.detectorFactory.getDefaultSettings();
215            settings.detectorSettings.put("TARGET_CHANNEL",           trackMateConfig.getTargetChannel());
216            settings.detectorSettings.put("RADIUS",                   trackMateConfig.getRadius());
217            settings.detectorSettings.put("DO_SUBPIXEL_LOCALIZATION", trackMateConfig.isDoSubpixelLocalization());
218            settings.detectorSettings.put("THRESHOLD",                threshold);
219            settings.detectorSettings.put("DO_MEDIAN_FILTERING",      trackMateConfig.isMedianFiltering());
220
221            // Tracker configuration
222            settings.trackerFactory  = new SparseLAPTrackerFactory();
223            settings.trackerSettings = settings.trackerFactory.getDefaultSettings();
224            settings.trackerSettings.put("LINKING_MAX_DISTANCE",            trackMateConfig.getLinkingMaxDistance());
225            settings.trackerSettings.put("ALTERNATIVE_LINKING_COST_FACTOR", trackMateConfig.getAlternativeLinkingCostFactor());
226            settings.trackerSettings.put("ALLOW_GAP_CLOSING",               trackMateConfig.isAllowGapClosing());
227            settings.trackerSettings.put("GAP_CLOSING_MAX_DISTANCE",        trackMateConfig.getGapClosingMaxDistance());
228            settings.trackerSettings.put("MAX_FRAME_GAP",                   trackMateConfig.getMaxFrameGap());
229            settings.trackerSettings.put("ALLOW_TRACK_SPLITTING",           trackMateConfig.isAllowTrackSplitting());
230            settings.trackerSettings.put("SPLITTING_MAX_DISTANCE",          trackMateConfig.getSplittingMaxDistance());
231            settings.trackerSettings.put("ALLOW_TRACK_MERGING",             trackMateConfig.isAllowTrackMerging());
232            settings.trackerSettings.put("MERGING_MAX_DISTANCE",            trackMateConfig.getMergingMaxDistance());
233
234            // Deterministic execution
235            Locale.setDefault(Locale.US);
236            System.setProperty("user.language", "en");
237            System.setProperty("user.country", "US");
238            System.setProperty("java.util.concurrent.ForkJoinPool.common.parallelism", "1");
239            System.setProperty("trackmate.deterministic", "true");
240
241            settings.addSpotFilter(new FeatureFilter("QUALITY", 0, true));
242            settings.addAllAnalyzers();
243            settings.addTrackFilter(new FeatureFilter("NUMBER_SPOTS",
244                                                      trackMateConfig.getMinNumberOfSpotsInTrack(),
245                                                      true));
246
247            // -----------------------------------------------------------------
248            // Step 4 – Execute TrackMate
249            // -----------------------------------------------------------------
250
251            if (isCancelled(Thread.currentThread(), dialog)) {
252                return cancelEarly(imp, impBrightfield);
253            }
254
255            TrackMate trackmate = new TrackMate(model, settings);
256            if (!trackmate.checkInput()) {
257                if (!isCancelled(Thread.currentThread(), dialog)) {
258                    PaintLogger.errorf("TrackMate input check failed: %s", trackmate.getErrorMessage());
259                }
260                return cancelEarly(imp, impBrightfield);
261            }
262
263            // ---- Spot detection ----
264            PaintLogger.raw("\n                       TrackMate - spot detection:  ");
265            try {
266                if (!trackmate.execDetection()) {
267                    // If we reach here with an interrupted thread or external cancel,
268                    // treat it as cancellation, not as a hard error.
269                    if (Thread.currentThread().isInterrupted() || isCancelled(Thread.currentThread(), dialog)) {
270                        Thread.currentThread().interrupt();
271                        PaintLogger.warnf("TrackMate spot detection interrupted (cancellation).");
272                        return cancelEarly(imp, impBrightfield);
273                    }
274
275                    PaintLogger.errorf("TrackMate - execDetection failed: %s",
276                                       trackmate.getErrorMessage());
277                    return cancelEarly(imp, impBrightfield);
278                }
279            } catch (Exception e) {
280                // Special handling: InterruptedException often bubbles out of TrackMate
281                // when the watchdog/worker interrupts the thread.
282                if (e instanceof InterruptedException
283                        || Thread.currentThread().isInterrupted()
284                        || isCancelled(Thread.currentThread(), dialog)) {
285                    Thread.currentThread().interrupt();
286                    PaintLogger.warnf("TrackMate spot detection interrupted (likely cancellation).");
287                    return cancelEarly(imp, impBrightfield);
288                }
289
290                PaintLogger.errorf("Unexpected error during detection: %s", e.getMessage());
291                return cancelEarly(imp, impBrightfield);
292            }
293
294            int numberOfSpots = model.getSpots().getNSpots(false);
295            if (numberOfSpots > trackMateConfig.getMaxNumberOfSpotsInImage()) {
296                PaintLogger.warnf("   TrackMate - Too many spots detected (%d). Limit is %d.",
297                                  numberOfSpots, trackMateConfig.getMaxNumberOfSpotsInImage());
298                return cancelEarly(imp, impBrightfield);
299            } else {
300                String numberOfSpotsString = " (" + numberOfSpots + " spots detected) ";
301                PaintLogger.raw(numberOfSpotsString);
302            }
303
304            if (isCancelled(Thread.currentThread(), dialog)) {
305                return cancelEarly(imp, impBrightfield);
306            }
307
308            // ---- Track detection ----
309            PaintLogger.raw("\n                       TrackMate - track detection: ");
310            try {
311                if (!trackmate.process()) {
312
313                    if (Thread.currentThread().isInterrupted() || isCancelled(Thread.currentThread(), dialog)) {
314                        Thread.currentThread().interrupt();
315                        PaintLogger.warnf("TrackMate track detection interrupted (cancellation).");
316                        return cancelEarly(imp, impBrightfield);
317                    }
318
319                    PaintLogger.errorf("TrackMate process failed: %s", trackmate.getErrorMessage());
320                    return cancelEarly(imp, impBrightfield);
321                }
322            } catch (Exception e) {
323                if (e instanceof InterruptedException
324                        || Thread.currentThread().isInterrupted()
325                        || isCancelled(Thread.currentThread(), dialog)) {
326                    Thread.currentThread().interrupt();
327                    PaintLogger.warnf("TrackMate track detection interrupted (likely cancellation).");
328                    return cancelEarly(imp, impBrightfield);
329                }
330
331                PaintLogger.errorf("Unexpected error during TrackMate process: %s", e.getMessage());
332                return cancelEarly(imp, impBrightfield);
333            }
334
335            int numberOfFilteredTracks = model.getTrackModel().nTracks(true);
336            String numberOfTracksString = " (" + numberOfFilteredTracks + " tracks detected) ";
337            PaintLogger.raw(numberOfTracksString);
338
339            // -----------------------------------------------------------------
340            // Step 5 – Visualization
341            // -----------------------------------------------------------------
342            if (isCancelled(Thread.currentThread(), dialog)) {
343                return cancelEarly(imp, impBrightfield);
344            }
345
346            final SelectionModel  selectionModel = new SelectionModel(model);
347            final DisplaySettings ds             = DisplaySettingsIO.readUserDefault();
348            ds.setSpotVisible(false);
349            ds.setTrackColorBy(DisplaySettings.TrackMateObject.TRACKS,
350                               trackMateConfig.getTrackColoring());
351            final HyperStackDisplayer displayer = new HyperStackDisplayer(model,
352                                                                          selectionModel,
353                                                                          imp,
354                                                                          ds);
355            displayer.render();
356            displayer.refresh();
357
358            capture = CaptureOverlayAction.capture(imp, -1, 1, null);
359
360            Path imagePath = experimentPath.resolve("TrackMate Images")
361                                           .resolve(experimentInfoRecord.getRecordingName() + ".jpg");
362            if (capture != null) {
363                if (!new FileSaver(capture).saveAsTiff(String.valueOf(imagePath))) {
364                    PaintLogger.errorf("Failed to save TIFF to: %s", imagePath);
365                }
366            }
367            IJ.wait(2000);
368
369            // -----------------------------------------------------------------
370            // Step 6 – Write tracks CSV
371            // -----------------------------------------------------------------
372            String tracksName = experimentInfoRecord.getRecordingName() + "-tracks.csv";
373            Path   tracksPath = experimentPath.resolve(tracksName);
374            int    totalSpotsInAllTracks = 0;
375
376            try {
377                totalSpotsInAllTracks = TrackDataExporter.writeTracksCsv(
378                        trackmate,
379                        experimentInfoRecord.getExperimentName(),
380                        experimentInfoRecord.getRecordingName(),
381                        tracksPath,
382                        true);
383            } catch (Exception e) {
384                PaintLogger.errorf("Failed to write tracks to '%s'", tracksPath);
385            }
386
387            // -----------------------------------------------------------------
388            // Step 7 – Summarize and return results
389            // -----------------------------------------------------------------
390            int numberOfSpotsTotal     = model.getSpots().getNSpots(true);
391            int numberOfTracks         = model.getTrackModel().nTracks(false);
392            // int numberOfFilteredTracks = model.getTrackModel().nTracks(true);
393            int numberOfFrames         = imp.getNFrames();
394
395            Duration duration = Duration.between(start, LocalDateTime.now());
396
397            closeImages(imp, impBrightfield, capture);
398
399            return new TrackMateResults(true,
400                                        true,
401                                        numberOfSpotsTotal,
402                                        numberOfTracks,
403                                        numberOfFilteredTracks,
404                                        numberOfFrames,
405                                        duration,
406                                        totalSpotsInAllTracks);
407
408        } catch (Exception e) {
409            // Any unexpected top-level exception: honor cancellation if present
410            if (e instanceof InterruptedException
411                    || Thread.currentThread().isInterrupted()
412                    || isCancelled(Thread.currentThread(), dialog)) {
413                Thread.currentThread().interrupt();
414                PaintLogger.warnf("TrackMate recording interrupted (top-level cancellation).");
415                return cancelEarly(imp, impBrightfield);
416            }
417
418            PaintLogger.errorf("Exception during TrackMate processing: %s", e.getMessage());
419            return cancelEarly(imp, impBrightfield);
420        } finally {
421            closeImages(imp, impBrightfield, capture);
422        }
423    }

426    // Utility methods
427    // -------------------------------------------------------------------------
428
429    /**
430     * Checks for thread interruption or user cancellation.
431     */
432    private static boolean isCancelled(Thread t, ProjectDialog dialog) {
433        return t.isInterrupted() || (dialog != null && dialog.isCancelled());
434    }
435
436    /**
437     * Safely closes visible images.
438     */
439    private static void closeImages(ImagePlus... images) {
440        for (ImagePlus img : images) {
441            try {
442                if (img != null && img.isVisible()) {
443                    img.close();
444                }
445            } catch (Exception e) {
446                PaintLogger.warnf("Error closing image: %s", e.getMessage());
447            }
448        }
449    }
450
451    /**
452     * Cancels processing early and closes images.
453     */
454    private static TrackMateResults cancelEarly(ImagePlus... images) {
455        closeImages(images);
456        PaintLogger.warnf("   Recording cancelled.");
457        return new TrackMateResults(false);
458    }
