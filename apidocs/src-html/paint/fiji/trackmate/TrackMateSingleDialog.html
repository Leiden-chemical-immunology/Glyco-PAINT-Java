014 *      • Save the resulting track and spot data to CSV.

016 *    It provides visual feedback during long-running TrackMate operations
017 *    and prevents concurrent execution through UI-state management.
018 *

020 *    • Interactive recording selection and threshold tuning.
021 *    • Background thread execution for responsive UI.
022 *    • Status feedback with animated dots for active processes.

038package paint.fiji.trackmate;

040import javax.swing.*;
041import javax.swing.border.EmptyBorder;
042import java.awt.*;
043import java.awt.event.WindowAdapter;
044import java.awt.event.WindowEvent;
045import java.util.ArrayList;
046import java.util.Collections;
047import java.util.List;
048import paint.shared.utils.PaintLogger;
049
050public class TrackMateSingleDialog extends JDialog {
051
052    public interface CalculationHandler {
053        void run(String recordingName, int threshold) throws Exception;
054    }
055
056    public interface SaveHandler {
057        void run(String recordingName, int threshold) throws Exception;
058    }
059
060    private final ButtonGroup        recordingGroup   = new ButtonGroup();
061    private final List<JRadioButton> recordingButtons = new ArrayList<>();
062
063    private final JSlider thresholdSlider;
064    private final JLabel  thresholdValueLabel;
065    private final JLabel  statusLabel;
066
067    private volatile Thread  runningThread;
068    private volatile Thread  dotThread;
069    private volatile boolean dotRunning;
070
071    private String selectedRecordingName;
072    private int    selectedThreshold;
073
074    public interface ThresholdProvider {
075        int getThreshold(String recordingName);
076    }
077
078    private ThresholdProvider thresholdProvider;
079
080    public void setThresholdProvider(ThresholdProvider p) {
081        this.thresholdProvider = p;
082        applyThresholdFromProvider(); // set slider for current selection immediately
083    }
084
085    public TrackMateSingleDialog(
086            Window             owner,
087            List<String>       recordingNames,
088            String             initialRecordingName,
089            int                initialThreshold,
090            CalculationHandler onCalculate,
091            SaveHandler        onSave
092    ) {
093        super(owner, "TrackMate Single", ModalityType.MODELESS);
094
095        List<String> names = (recordingNames == null)
096                ? Collections.<String>emptyList()
097                : new ArrayList<>(recordingNames);
098
099        setDefaultCloseOperation(WindowConstants.DO_NOTHING_ON_CLOSE);
100        addWindowListener(new WindowAdapter() {
101            @Override
102            public void windowClosing(WindowEvent e) {
103                handleCancel(); // same behavior as Cancel button
104            }
105        });
106
107        JPanel root = new JPanel(new BorderLayout(10, 10));
108        root.setBorder(new EmptyBorder(12, 12, 12, 12));
109        setContentPane(root);
110
111        // ---------------------------------------------------------------------
112        // Left: recordings
113        // ---------------------------------------------------------------------
114        JPanel recordingsPanel = new JPanel();
115        recordingsPanel.setLayout(new BoxLayout(recordingsPanel, BoxLayout.Y_AXIS));
116        recordingsPanel.setBorder(BorderFactory.createTitledBorder("Recording"));
117
118        boolean anySelected = false;
119        String preferred = safeTrim(initialRecordingName);
120
121        for (String name : names) {
122            if (name == null) continue;
123            final String rec = name.trim();
124            if (rec.isEmpty()) {
125                continue;
126            }
127
128            JRadioButton rb = new JRadioButton(rec);
129            rb.setAlignmentX(Component.LEFT_ALIGNMENT);
130
131            // Select preferred if matches
132            if (!preferred.isEmpty() && preferred.equals(rec)) {
133                rb.setSelected(true);
134                anySelected = true;
135                selectedRecordingName = rec;
136            }
137
138            rb.addActionListener(e -> {
139                selectedRecordingName = rec;
140                applyThresholdFromProvider();
141            });
142
143            recordingGroup.add(rb);
144            recordingButtons.add(rb);
145            recordingsPanel.add(rb);
146        }
147
148        // Fallback: select first if nothing selected
149        if (!anySelected && !recordingButtons.isEmpty()) {
150            JRadioButton first = recordingButtons.get(0);
151            first.setSelected(true);
152            selectedRecordingName = first.getText();
153        }
154
155        JScrollPane recordingsScroll = new JScrollPane(
156                recordingsPanel,
157                JScrollPane.VERTICAL_SCROLLBAR_AS_NEEDED,
158                JScrollPane.HORIZONTAL_SCROLLBAR_NEVER
159        );
160        recordingsScroll.setPreferredSize(new Dimension(420, 320));
161        recordingsScroll.setBorder(BorderFactory.createEmptyBorder());
162
163        // ---------------------------------------------------------------------
164        // Right: threshold slider
165        // ---------------------------------------------------------------------
166        JPanel thresholdPanel = new JPanel();
167        thresholdPanel.setLayout(new BoxLayout(thresholdPanel, BoxLayout.Y_AXIS));
168        thresholdPanel.setBorder(
169                BorderFactory.createCompoundBorder(
170                        BorderFactory.createTitledBorder("Threshold"),
171                        new EmptyBorder(8, 12, 8, 8) // top, left, bottom, right
172                )
173        );
174        // thresholdPanel.add(Box.createVerticalStrut(8));
175
176        int initT = clamp(initialThreshold, 1, 50);
177
178        thresholdValueLabel = new JLabel(String.valueOf(initT));
179        thresholdValueLabel.setFont(thresholdValueLabel.getFont().deriveFont(Font.BOLD, 18f));
180        thresholdValueLabel.setAlignmentX(Component.LEFT_ALIGNMENT);
181
182        thresholdSlider = new JSlider(0, 50, Math.max(1, initT));
183        thresholdSlider.setMajorTickSpacing(10);
184        thresholdSlider.setMinorTickSpacing(0);     // no fine ticks
185        thresholdSlider.setPaintTicks(true);
186        thresholdSlider.setPaintLabels(true);
187        thresholdSlider.setSnapToTicks(false);
188
189        thresholdSlider.setAlignmentX(Component.LEFT_ALIGNMENT);
190
191        thresholdSlider.addChangeListener(e -> {
192            int v = thresholdSlider.getValue();
193
194            // Prevent selecting 0
195            if (v == 0) {
196                thresholdSlider.setValue(1);
197                v = 1;
198            }
199
200            thresholdValueLabel.setText(String.valueOf(v));
201        });
202
203        thresholdPanel.add(Box.createVerticalStrut(6));
204        thresholdPanel.add(thresholdValueLabel);
205        thresholdPanel.add(Box.createVerticalStrut(12));
206        thresholdPanel.add(thresholdSlider);
207        thresholdPanel.add(Box.createVerticalGlue());
208
209        // ---------------------------------------------------------------------
210        // Center layout
211        // ---------------------------------------------------------------------
212        JPanel center = new JPanel(new BorderLayout(10, 10));
213        center.add(recordingsScroll, BorderLayout.CENTER);
214        center.add(thresholdPanel, BorderLayout.EAST);
215        root.add(center, BorderLayout.CENTER);
216
217        // ---------------------------------------------------------------------
218        // Bottom buttons: Calculate / Save / Cancel
219        // ---------------------------------------------------------------------
220        statusLabel = new JLabel(" ");
221        statusLabel.setBorder(new EmptyBorder(0, 0, 6, 0));
222
223        JButton calculateBtn = new JButton("Calculate");
224        JButton saveBtn      = new JButton("Save");
225        JButton cancelBtn    = new JButton("Cancel");
226
227        calculateBtn.addActionListener(e -> {
228            if (!captureSelections()) {
229                Toolkit.getDefaultToolkit().beep();
230                return;
231            }
232            if (runningThread != null) {
233                Toolkit.getDefaultToolkit().beep();
234                return;
235            }
236
237            setUiRunning(true, "Running…");
238
239            runningThread = new Thread(() -> {
240                startDots();
241                try {
242                    if (onCalculate != null) {
243                        onCalculate.run(selectedRecordingName, selectedThreshold);
244                    }
245
246                    if (Thread.currentThread().isInterrupted()) {
247                        SwingUtilities.invokeLater(() -> setUiRunning(false, "Cancelled"));
248                        return;
249                    }
250
251                    SwingUtilities.invokeLater(() -> setUiRunning(false, "Completed"));
252                } catch (Exception ex) {
253                    if (Thread.currentThread().isInterrupted()) {
254                        SwingUtilities.invokeLater(() -> setUiRunning(false, "Cancelled"));
255                    } else {
256                        SwingUtilities.invokeLater(() -> setUiRunning(false, "Failed: " + ex.getMessage()));
257                    }
258                } finally {
259                    stopDots();
260                    runningThread = null;
261                }
262            }, "TrackMateSingle-Worker");
263
264            runningThread.start();
265        });
266
267        saveBtn.addActionListener(e -> {
268            if (!captureSelections()) {
269                Toolkit.getDefaultToolkit().beep();
270                return;
271            }
272            if (runningThread != null) {
273                Toolkit.getDefaultToolkit().beep();
274                return;
275            }
276            try {
277                if (onSave != null) {
278                    onSave.run(selectedRecordingName, selectedThreshold);
279                }
280                statusLabel.setText("Saved");
281            } catch (Exception ex) {
282                statusLabel.setText("Save failed: " + ex.getMessage());
283            }
284        });
285
286        cancelBtn.addActionListener(e -> handleCancel());
287
288        JPanel buttons = new JPanel(new FlowLayout(FlowLayout.RIGHT, 8, 0));
289        buttons.add(calculateBtn);
290        buttons.add(saveBtn);
291        buttons.add(cancelBtn);
292
293        JPanel bottom = new JPanel(new BorderLayout());
294        bottom.add(statusLabel, BorderLayout.WEST);
295        bottom.add(buttons, BorderLayout.EAST);
296
297        root.add(bottom, BorderLayout.SOUTH);
298
299        // store initial
300        selectedThreshold = thresholdSlider.getValue();
301
302        pack();
303        setLocationRelativeTo(owner);
304
305        // local helpers that need button refs
306        this.calculateBtn = calculateBtn;
307        this.saveBtn = saveBtn;
308        this.cancelBtn = cancelBtn;
309    }
310
311    // keep refs for enabling/disabling
312    private final JButton calculateBtn;
313    private final JButton saveBtn;
314    private final JButton cancelBtn;
315
316    private void handleCancel() {
317        Thread t = runningThread;
318        if (t != null) {
319            // first Cancel interrupts running job (dialog stays open)
320            statusLabel.setText("Stopping…");
321            stopDots();
322            t.interrupt();
323            return;
324        }
325        dispose();
326    }
327
328    private void setUiRunning(boolean running, String status) {
329        statusLabel.setText(status);
330        calculateBtn.setEnabled(!running);
331        saveBtn.setEnabled(!running);
332        cancelBtn.setEnabled(true);
333        thresholdSlider.setEnabled(!running);
334        for (JRadioButton rb : recordingButtons) rb.setEnabled(!running);
335    }
336
337    private boolean captureSelections() {
338        String rec = selectedRecordingName;
339        if (rec == null || rec.trim().isEmpty()) {
340            for (JRadioButton rb : recordingButtons) {
341                if (rb.isSelected()) {
342                    rec = rb.getText();
343                    break;
344                }
345            }
346        }
347        if (rec == null || rec.trim().isEmpty()) {
348            return false;
349        }
350
351        selectedRecordingName = rec.trim();
352        selectedThreshold     = thresholdSlider.getValue();
353        return true;
354    }
355
356    private static int clamp(int v, int lo, int hi) {
357        if (v < lo) {
358            return lo;
359        }
360        if (v > hi) {
361            return hi;
362        }
363        return v;
364    }
365
366    private static String safeTrim(String s) {
367        return s == null ? "" : s.trim();
368    }
369
370    private void applyThresholdFromProvider() {
371        if (thresholdProvider == null) {
372            return;
373        }
374        if (selectedRecordingName == null || selectedRecordingName.trim().isEmpty()) {
375            return;
376        }
377
378        int v = thresholdProvider.getThreshold(selectedRecordingName.trim());
379        if (v < 1) {
380            v = 1;
381        }
382        if (v > 50) {
383            v = 50;
384        }
385
386        thresholdSlider.setValue(v); // your changeListener will also update label & enforce 0->1
387    }
388
389    private void startDots() {
390        if (dotThread != null && dotThread.isAlive()) {
391            return;
392        }
393        if (dotRunning) {
394            return;
395        }
396        dotRunning = true;
397        dotThread = new Thread(() -> {
398            int dots = 0;
399            while (dotRunning && !Thread.currentThread().isInterrupted()) {
400                try {
401                    Thread.sleep(1000);
402                } catch (InterruptedException e) {
403                    Thread.currentThread().interrupt();
404                    break;
405                }
406
407                if (!dotRunning) break;
408
409                PaintLogger.raw(".");
410                dots++;
411
412                if (dots >= 80) {
413                    PaintLogger.raw("\n                                                    ");
414                    dots = 0;
415                }
416            }
417        }, "TrackMateSingle-Dots");
418
419        dotThread.setDaemon(true);
420        dotThread.start();
421    }
422
423    private void stopDots() {
424        dotRunning = false;
425        Thread t = dotThread;
426        if (t != null) {
427            t.interrupt();
428        }
429        dotThread = null;
430    }
431
432    @Override
433    public void dispose() {
434        stopDots();
435        super.dispose();
436    }
