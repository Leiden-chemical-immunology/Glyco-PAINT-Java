005 *  PURPOSE:
006 *    Executes the Paint TrackMate analysis pipeline across multiple experiments
007 *    under varying parameter configurations ("sweep mode"). This allows
008 *    systematic exploration of parameter sensitivity by iteratively running
009 *    TrackMate under different settings.
010 *
011 *  DESCRIPTION:
012 *    • Loads the Sweep configuration from Paint's JSON configuration file.
013 *    • Iterates over all active TrackMate parameters marked for sweeping.
014 *    • Creates separate subdirectories for each parameter–value combination.
015 *    • Rewrites Paint configuration files dynamically for each sweep run.
016 *    • Executes the TrackMate workflow for all experiments under the new setup.
017 *    • Optionally flattens the sweep results for summary analysis.
018 *
019 *  RESPONSIBILITIES:
020 *    • Manage configuration cloning and parameter substitution per sweep run.
021 *    • Invoke {@link RunTrackMateOnProject} for each parameter combination.
022 *    • Maintain logging and recovery of the original Paint configuration.
023 *
024 *  USAGE EXAMPLE:
025 *    boolean ok = RunTrackMateOnProjectSweep.runWithSweep(
026 *                     Paths.get("/Paint Project"),
027 *                     Paths.get("/Volumes/Images"),

068/**
069 * Coordinates a "parameter sweep" execution for TrackMate across multiple
070 * experiments. For each active sweep parameter, the Paint configuration
071 * is duplicated and modified, and the entire project is reprocessed.
072 * <p>
073 * This class enables systematic parameter-space exploration for TrackMate
074 * within Paint’s experiment framework.
075 * </p>
076 */
077public class RunTrackMateOnProjectSweep {
078
079    /**
080     * Executes the TrackMate analysis workflow across multiple parameter
081     * configurations defined in the sweep configuration file.
082     * <p>
083     * For each parameter–value combination, this method:
084     * <ul>
085     *   <li>Creates a clean subdirectory under the project’s "Sweep" folder.</li>
086     *   <li>Copies and modifies the Paint configuration file.</li>
087     *   <li>Runs all selected experiments under that parameter setting.</li>
088     *   <li>Logs results and restores the original configuration afterward.</li>
089     * </ul>
090     *
091     * @param projectPath     the root directory of the Paint project
092     * @param imagesPath      the directory containing experiment image data
093     * @param experimentNames list of experiments to include in the sweep
094     * @return {@code true} if all sweep combinations complete successfully;
095     *         {@code false} otherwise
096     * @throws IOException if any file or directory operation fails
097     */
098    public static boolean runWithSweep(Path projectPath,
099                                       Path imagesPath,
100                                       List<String> experimentNames) throws IOException {
101
102        // ---------------------------------------------------------------------
103        // Phase 1 – Load sweep configuration
104        // ---------------------------------------------------------------------
105        Path sweepFile = projectPath.resolve(PAINT_SWEEP_CONFIGURATION_JSON);
106        if (!Files.exists(sweepFile)) {
107            PaintLogger.infof("No sweep configuration found at %s", sweepFile);
108            return false;
109        }
110
111        JsonValidator.Result validation = JsonValidator.validate(sweepFile);
112        if (!validation.valid) {
113            PaintLogger.errorf("Invalid JSON: " + validation.error);
114            return false;
115        }
116
117        SweepConfig sweepConfig = new SweepConfig(sweepFile.toString());
118        Map<String, List<Number>> sweeps = sweepConfig.getActiveSweepValues("TrackMate Sweep");
119
120        if (sweeps.isEmpty()) {
121            PaintLogger.infof("Sweep enabled, but no active sweep parameters defined.");
122            return false;
123        }
124
125        boolean overallStatus = true;
126        List<String[]> summaryRows = new ArrayList<>();
127
128        // ---------------------------------------------------------------------
129        // Phase 2 – Generate sweep summary for logging
130        // ---------------------------------------------------------------------
131        List<String> sweepSummary = new ArrayList<>();
132        sweepSummary.add("");
133        try {
134            for (Map.Entry<String, List<Number>> entry : sweeps.entrySet()) {
135                String       parameter = entry.getKey();       // parameter name
136                List<Number> values    = entry.getValue();  // list of values
137
138                // Convert values to comma-separated string
139                String valueList = values.stream()
140                        .map(Object::toString)
141                        .reduce((a, b) -> a + ", " + b)
142                        .orElse("(none)");
143
144                // Add a header line for the parameter
145                sweepSummary.add(String.format("Parameter: %s", parameter));
146                sweepSummary.add(String.format("Values:    %s", valueList));
147                sweepSummary.add("");
148            }
149
150            // Add the experiments section at the end
151            if (experimentNames != null && !experimentNames.isEmpty()) {
152                String exps = String.join(", ", experimentNames);
153                sweepSummary.add(String.format("Experiments: %s", exps));
154            }
155        } catch (Exception e) {
156            PaintLogger.errorf("Error building sweep summary: %s", e.getMessage());
157        }
158        PaintLogger.doc("Sweep analysis to be performed", sweepSummary);
159
160        // ---------------------------------------------------------------------
161        // Phase 3 – Perform parameter sweep
162        // ---------------------------------------------------------------------
163        try {
164            for (Map.Entry<String, List<Number>> entry : sweeps.entrySet()) {
165                // Cycle through the parameters
166                String parameter    = entry.getKey();       // Holds a parameter that we sweep
167                List<Number> values = entry.getValue();     // Values holds the values
168
169                // Store original parameter value for restoration
170                String originalValue = null;
171                if (!THRESHOLD.equals(parameter)) {
172                    originalValue = PaintConfig.getString("TrackMate", parameter, "undefined");
173                }
174
175                for (Number val : values) {
176                    PaintLogger.blankline();
177                    PaintLogger.blankline();
178                    PaintLogger.infof("Running sweep for %s = %s", parameter, val);
179
180                    // Create clean sweep directory
181                    Path sweepPath = projectPath.resolve("Sweep").resolve("[" + parameter + "]-[" + val + "]");
182                    FileUtils.deleteDirectory(sweepPath.toFile());
183                    Files.createDirectories(sweepPath);
184
185                    // Copy baseline PaintConfig.json
186                    Path baselineConfig = projectPath.resolve(PAINT_CONFIGURATION_JSON);
187                    Path configCopy     = sweepPath.resolve(PAINT_CONFIGURATION_JSON);
188                    try {
189                        Files.copy(baselineConfig, configCopy, StandardCopyOption.REPLACE_EXISTING);
190                        PaintLogger.infof("Copied baseline PaintConfig.json to %s", configCopy);
191                    } catch (IOException e) {
192                        PaintLogger.errorf("Failed to copy PaintConfig.json to %s: %s", configCopy, e.getMessage());
193                    }
194
195                    // Reinitialize PaintConfig for this sweep
196                    PaintConfig.reinitialise(sweepPath);
197
198                    // Apply updated parameter value if it is value that need to be changed in the configuration
199                    if (!THRESHOLD.equals(parameter)) {
200                        if (val.doubleValue() == val.intValue()) {
201                            PaintConfig.setInt("TrackMate", parameter, val.intValue());
202                        } else {
203                            PaintConfig.setDouble("TrackMate", parameter, val.doubleValue());
204                        }
205                    }
206
207                    // Persist modified configuration
208                    PaintConfig.instance().save();
209
210                    // Copy Experiment Info.csv files into sweep directory
211                    for (String expName : experimentNames) {
212                        Path expSrc    = projectPath.resolve(expName).resolve(EXPERIMENT_INFO_CSV);
213                        Path expDstDir = sweepPath.resolve(expName);
214                        Path expDst    = expDstDir.resolve(EXPERIMENT_INFO_CSV);
215                        try {
216                            if (Files.exists(expSrc)) {
217                                Files.createDirectories(expDstDir);
218                                Files.copy(expSrc, expDst, StandardCopyOption.REPLACE_EXISTING);
219
220                                if (THRESHOLD.equals(parameter)) {
221                                    PaintLogger.infof("Updating threshold %f for %s", val.doubleValue(), expDst);
222                                    updateThreshold(expDst, val.doubleValue());
223                                }
224                            } else {
225                                PaintLogger.warnf("Experiment Info.csv not found for %s at %s", expName, expSrc);
226                            }
227                        } catch (IOException e) {
228                            PaintLogger.errorf("Failed to copy Experiment Info.csv for %s: %s", expName, e.getMessage());
229                        }
230                    }
231
232                    // Run TrackMate workflow for all experiments under current sweep
233                    boolean status = RunTrackMateOnProject.runProject(projectPath, imagesPath, experimentNames, null, sweepPath);
234
235                    summaryRows.add(new String[]{
236                            parameter, val.toString(), sweepPath.toString(),
237                            status ? "SUCCESS" : "FAILED"
238                    });
239
240                    if (!status) {
241                        overallStatus = false;
242                    }
243                }
244
245                // Optionally restore original value after each parameter sweep
246                if (!THRESHOLD.equals(parameter)) {
247                    PaintConfig.setString("TrackMate", parameter, originalValue);
248                }
249            }
250
251        } finally {
252            // Always restore PaintConfig context to the project root
253            PaintConfig.reinitialise(projectPath);
254            PaintLogger.infof("Restored PaintConfig to project root: %s", projectPath);
255        }
256
257        // ---------------------------------------------------------------------
258        // Phase 4 – Flatten results (optional)
259        // ---------------------------------------------------------------------
260        if (overallStatus) {
261            flattenSweep(projectPath.resolve("Sweep"), experimentNames, true);
262        }
263
264        return overallStatus;
265    }

277    public static void updateThreshold(Path csvPath, double newValue) throws IOException {
278
279        List<String> lines = Files.readAllLines(csvPath);
280        if (lines.isEmpty()) {
281            return;
282        }
283
284        String   header         = lines.get(0);
285        String[] cols           = header.split(",");
286        int      thresholdIndex = -1;
287
288        for (int i = 0; i < cols.length; i++) {
289            if (THRESHOLD.equalsIgnoreCase(cols[i].trim())) {
290                thresholdIndex = i;
291                break;
292            }
293        }
294
295        if (thresholdIndex < 0) {
296            throw new IOException("CSV does not contain a 'Threshold' column: " + csvPath);
297        }
298
299        List<String> output = new ArrayList<>();
300        output.add(header); // keep header unchanged
301
302        for (int row = 1; row < lines.size(); row++) {
303            String[] parts = lines.get(row).split(",");
304            if (parts.length > thresholdIndex) {
305                parts[thresholdIndex] = String.valueOf(newValue);
306            }
307            output.add(String.join(",", parts));
308        }
309
310        Files.write(csvPath, output, StandardOpenOption.TRUNCATE_EXISTING);
311    }
