040package paint.fiji.trackmate;
041
042import java.time.Duration;
043
044/**
045 * Immutable container for the results of a TrackMate analysis.
046 * <p>
047 * Encapsulates statistics such as the number of detected spots,
048 * total and filtered tracks, number of analyzed frames, and
049 * runtime duration.
050 * </p>
051 */
052public class TrackMateResults {
053    private final boolean  success;
054    private final boolean  calculationPerformed;
055    private final int      numberOfSpots;
056    private final int      numberOfTracks;
057    private final int      numberOfFilteredTracks;
058    private final int      numberOfFrames;
059    private final Duration duration;
060    private final int      numberOfSpotsInAllTracks;
061
062
063    /**
064     * Constructs a {@code TrackMateResults} instance with only a success flag.
065     * Assumes no calculations were performed and sets all numeric fields to zero.
066     *
067     * @param success whether the analysis succeeded
068     */
069    public TrackMateResults(boolean success) {
070        this(success, false, 0, 0, 0, 0, null, 0);
071    }
072
073    /**
074     * Constructs a fully detailed {@code TrackMateResults} instance.
075     *
076     * @param success                true if the analysis succeeded
077     * @param calculationPerformed   true if processing was executed
078     * @param numberOfSpots          number of detected spots
079     * @param numberOfTracks         total number of identified tracks
080     * @param numberOfFilteredTracks number of tracks after filtering
081     * @param numberOfFrames         total number of frames analyzed
082     * @param duration               runtime duration of analysis
083     * @param numberOfSpotsInAllTracks total number of spots in all tracks
084     */
085    public TrackMateResults(boolean success,
086                            boolean calculationPerformed,
087                            int numberOfSpots,
088                            int numberOfTracks,
089                            int numberOfFilteredTracks,
090                            int numberOfFrames,
091                            Duration duration,
092                            int numberOfSpotsInAllTracks) {
093        this.success                  = success;
094        this.calculationPerformed     = calculationPerformed;
095        this.numberOfSpots            = numberOfSpots;
096        this.numberOfTracks           = numberOfTracks;
097        this.numberOfFilteredTracks   = numberOfFilteredTracks;
098        this.numberOfFrames           = numberOfFrames;
099        this.duration                 = duration;
100        this.numberOfSpotsInAllTracks = numberOfSpotsInAllTracks;
101    }
102
103    // -------------------------------------------------------------------------
104    // Getters
105    // -------------------------------------------------------------------------
106
107    /**
108     * @return true if the analysis completed successfully
109     */
110    public boolean isSuccess() {
111        return success;
112    }
113
114    /**
115     * @return true if the calculation was performed
116     */
117    public boolean isCalculationPerformed() {
118        return calculationPerformed;
119    }
120
121    /**
122     * @return total number of detected spots
123     */
124    public int getNumberOfSpots() {
125        return numberOfSpots;
126    }
127
128    /**
129     * @return number of tracks that passed filtering
130     */
131    public int getNumberOfFilteredTracks() {
132        return numberOfFilteredTracks;
133    }
134
135    /**
136     * @return number of frames analyzed
137     */
138    public int getNumberOfFrames() {
139        return numberOfFrames;
140    }
141
142    /**
143     * @return duration of the analysis, or {@code null} if unavailable
144     */
145    public Duration getDuration() {
146        return duration;
147    }
148
149    /**
150     * @return number of spots across all tracks
151     */
152    public int getNumberOfSpotsInAllTracks() {
153        return numberOfSpotsInAllTracks;
154    }
155
156    // -------------------------------------------------------------------------
157    // Representation
158    // -------------------------------------------------------------------------
159
160    /**
161     * Returns a formatted summary string containing all key metrics.
162     *
163     * @return a string summarizing TrackMate run results
164     */
165    @Override
166    public String toString() {
167        return String.format(
168                "Success: %b, Spots: %d, Tracks: %d, Filtered: %d, Frames: %d, Duration(ms): %d",
169                success,
170                numberOfSpots,
171                numberOfTracks,
172                numberOfFilteredTracks,
173                numberOfFrames,
174                duration != null ? duration.toMillis() : 0
175        );
176    }
177}
