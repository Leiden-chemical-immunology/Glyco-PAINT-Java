001/*=============================================================================
002 *  Class:        TrackDataExporter.java
003 *  Package:      paint.fiji.tracks
004 *
005 *  PURPOSE:
006 *    Extracts TrackMate per-track motion statistics, augments them with
007 *    PAINT-calculated features, and exports the resulting dataset as a
008 *    schema-compliant Tracks table (CSV).
009 *
010 *  DESCRIPTION:
011 *    This class bridges Fiji TrackMate with the PAINT shared data layer:
012 *
013 *      • Reads TrackMate model + features for each track
014 *      • Computes extended PAINT attributes via
015 *        {@link paint.fiji.tracks.TrackAttributeCalculations}
016 *      • Builds fully typed {@link paint.shared.objects.Track} entities
017 *      • Produces a schema-validated {@link tech.tablesaw.api.Table}
018 *        (using PAINT’s shared I/O layer)
019 *      • Assigns deterministic track IDs and recording-scoped unique keys
020 *      • Persists the final table via
021 *        {@link paint.shared.io.MainDataInterface#writeSpecificTracksFile}
022 *
023 *  RESPONSIBILITIES:
024 *      • Convert TrackMate track data → PAINT Track objects
025 *      • Compute diffusion, displacement, confinement ratio, and related metrics
026 *      • Ensure deterministic row order for reproducibility
027 *      • Export the Tracks CSV in strict alignment with PAINT’s
028 *        {@code TrackSchema} definition
029 *
030 *  USAGE EXAMPLE:
031 *
032 *      int totalSpots = TrackDataExporter.writeTracksCsv(
033 *          trackmate,
034 *          "ExperimentA",
035 *          "Recording1",
036 *          Paths.get("tracks.csv"),
037 *          true
038 *      );
039 *
040 *  DEPENDENCIES:
041 *      – Fiji TrackMate (fiji.plugin.trackmate.*)
042 *      – paint.shared.objects.Track

086/**
087 * Provides functionality to export TrackMate tracks to a CSV file.
088 * <p>
089 * Combines native TrackMate features with custom PAINT-calculated attributes,
090 * producing a structured dataset suitable for downstream statistical analysis.
091 * </p>
092 */
093public final class TrackDataExporter {
094
095    private TrackDataExporter() {
096        // Utility class; prevent instantiation
097    }
098
099    /**
100     * Writes track information from a TrackMate analysis into a CSV file.
101     * <p>
102     * Includes both TrackMate-calculated and PAINT-derived metrics, such as
103     * total distance, diffusion coefficients, and confinement ratio.
104     * </p>
105     *
106     * @param trackmate          TrackMate instance containing the model and features
107     * @param experimentName     experiment name for metadata tagging
108     * @param recordingName      recording name associated with the tracks
109     * @param csvTracksFilePath  destination CSV file
110     * @param visibleOnly        if true, exports only visible tracks
111     * @return total number of spots across all exported tracks
112     */
113    public static int writeTracksCsv(final TrackMate trackmate,
114                                     final String    experimentName,
115                                     final String    recordingName,
116                                     final Path      csvTracksFilePath,
117                                     final boolean   visibleOnly) {
118
119        // ---------------------------------------------------------------------
120        // Step 1 – Extract TrackMate components
121        // ---------------------------------------------------------------------
122        final Model model               = trackmate.getModel();
123        final TrackModel trackModel     = model.getTrackModel();
124        final FeatureModel featureModel = model.getFeatureModel();
125
126        // Collect and sort track IDs for deterministic output order
127        final Set<Integer>  trackIDsSet = trackModel.trackIDs(visibleOnly);
128        final List<Integer> trackIDs    = new ArrayList<>(trackIDsSet);
129        Collections.sort(trackIDs);
130
131        final List<Track> tracks = new ArrayList<>();
132        int totalSpots = 0;
133
134        // ---------------------------------------------------------------------
135        // Step 2 – Process each track
136        // ---------------------------------------------------------------------
137        for (Integer trackId : trackIDs) {
138            TrackAttributes trackAttributes = calculateTrackAttributes(trackModel, trackId, PaintTiming.TIME_INTERVAL);
139
140            Track track = new Track();
141            track.setExperimentName(experimentName);
142            track.setRecordingName(recordingName);
143
144            // Native TrackMate features
145            track.setNumberOfSpots(     asInt(   featureModel.getTrackFeature(trackId, "NUMBER_SPOTS")));
146            track.setNumberOfGaps(      asInt(   featureModel.getTrackFeature(trackId, "NUMBER_GAPS")));
147            track.setLongestGap(        asInt(   featureModel.getTrackFeature(trackId, "LONGEST_GAP")));
148            track.setTrackDuration(     roundOr( featureModel.getTrackFeature(trackId, "TRACK_DURATION"),     3, -1));
149            track.setTrackXLocation(    roundOr( featureModel.getTrackFeature(trackId, "TRACK_X_LOCATION"),   2, -1));
150            track.setTrackYLocation(    roundOr( featureModel.getTrackFeature(trackId, "TRACK_Y_LOCATION"),   2, -1));
151            track.setTrackDisplacement( roundOr( featureModel.getTrackFeature(trackId, "TRACK_DISPLACEMENT"), 2, -1));
152            track.setTrackMaxSpeed(     roundOr( featureModel.getTrackFeature(trackId, "TRACK_MAX_SPEED"),    2, -1));
153            track.setTrackMedianSpeed(  roundOr( featureModel.getTrackFeature(trackId, "TRACK_MEDIAN_SPEED"), 2, -1));
154
155            // Custom PAINT-calculated attributes
156            track.setDiffusionCoefficient(    roundOr(trackAttributes.diffusionCoeff,    2, -1));
157            track.setDiffusionCoefficientExt( roundOr(trackAttributes.diffusionCoeffExt, 2, -1));
158            track.setTotalDistance(           roundOr(trackAttributes.totalDistance,     2, -1));
159            track.setConfinementRatio(        roundOr(trackAttributes.confinementRatio,  2, -1));
160            track.setSquareNumber(-1);
161            track.setLabelNumber(-1);
162
163            totalSpots += trackAttributes.numberOfSpotsInTracks;
164            tracks.add(track);
165        }
166
167        // ---------------------------------------------------------------------
168        // Step 3 – Build and export table
169        // ---------------------------------------------------------------------
170        try {
171            Table tracksTable = trackListToTable(tracks);
172
173            tracksTable = tracksTable.sortOn(
174                    RECORDING_NAME,
175                    NUMBER_OF_SPOTS,
176                    NUMBER_OF_GAPS,
177                    LONGEST_GAP,
178                    TRACK_DURATION,
179                    TRACK_X_LOCATION,
180                    TRACK_Y_LOCATION,
181                    TRACK_DISPLACEMENT,
182                    TRACK_MAX_SPEED,
183                    TRACK_MEDIAN_SPEED,
184                    DIFFUSION_COEFFICIENT,
185                    DIFFUSION_COEFFICIENT_EXT,
186                    TOTAL_DISTANCE,
187                    CONFINEMENT_RATIO
188            );
189
190            // Replace Track Ids and generate unique keys
191            IntColumn    newIds       = IntColumn.create(TRACK_ID);
192            StringColumn newUniqueKey = StringColumn.create(UNIQUE_KEY);
193
194            for (int i = 0; i < tracksTable.rowCount(); i++) {
195                newIds.append(i);
196                newUniqueKey.append(recordingName + "-" + i);
197            }
198            tracksTable.replaceColumn(TRACK_ID, newIds);
199            tracksTable.replaceColumn(UNIQUE_KEY, newUniqueKey);
200
201            writeSpecificTracksFile(csvTracksFilePath, tracksTable);
202
203        } catch (Exception e) {
204            PaintLogger.errorf("Failed writing track CSV: %s", e.getMessage());
205            e.printStackTrace();
206        }
207
208        return totalSpots;
209    }

215    /**
216     * Rounds a double to the specified number of decimal places.
217     *
218     * @param v      input value (nullable)
219     * @param places number of decimal places
220     * @return rounded value, or {@code Double.NaN} if null
221     */
222    private static double roundTo(Double v, int places) {
223        if (v == null) {
224            return Double.NaN;
225        }
226        double scale = Math.pow(10, places);
227        return Math.round(v * scale) / scale;
228    }
229
230    /**
231     * Rounds a double or returns a fallback if {@code v} is null.
232     *
233     * @param v       input value (nullable)
234     * @param places  decimal precision
235     * @param ifNull  fallback value when input is null
236     * @return rounded value or fallback
237     */
238    @SuppressWarnings("SameParameterValue")
239    private static Double roundOr(Double v, int places, double ifNull) {
240        return v == null ? ifNull : roundTo(v, places);
241    }
242
243    /**
244     * Converts a Double to an int, returning -1 for null or NaN.
245     *
246     * @param v Double value
247     * @return integer value or -1 for invalid input
248     */
249    private static int asInt(Double v) {
250        if (v == null || v.isNaN()) {
251            return -1;
252        }
253        return (int) Math.round(v);
254    }
