011 *    The {@code RecordingLoader} iterates over experiment folders inside a
012 *    {@link paint.shared.objects.Project}, loads each experiment using
013 *    {@link paint.shared.io.ExperimentDataLoader}, and creates a
014 *    {@link RecordingEntry} for each recording that:

019 *
020 *    Recordings missing mandatory resources are skipped with diagnostic logging.
021 *    TrackMate images are expected in:
022 *
023 *      <experiment>/TrackMate Images/<recording>.jpg
024 *
025 *    Brightfield images are expected in:
026 *
027 *      <experiment>/Brightfield Images/<recording>[-BF*].jpg
028 *
029 *  KEY FEATURES:
030 *    • Loads experiments, recordings, squares (no tracks).
031 *    • Validates all required image assets.
032 *    • Returns a complete, ready-to-display list of {@link RecordingEntry}.
033 *    • Uses {@link paint.shared.utils.PaintLogger} for structured diagnostics.
034 *

048package paint.viewer.model;
049
050import paint.shared.io.ExperimentDataLoader;
051import paint.shared.objects.Experiment;
052import paint.shared.objects.Project;
053import paint.shared.objects.Recording;
054import paint.shared.utils.PaintLogger;
055
056import java.nio.file.Files;
057import java.nio.file.Path;
058import java.util.ArrayList;
059import java.util.Iterator;
060import java.util.List;
061
062/**
063 * Loads and validates {@link RecordingEntry} objects for a full PAINT project.
064 *
065 * <p>The loader ensures:</p>
066 * <ul>
067 *   <li>Each recording has its process flag set.</li>
068 *   <li>Required TrackMate and Brightfield images exist.</li>
069 *   <li>Square data is loaded via {@link ExperimentDataLoader}.</li>
070 * </ul>
071 *
072 * <p>Invalid or incomplete recordings are skipped, and diagnostic messages are
073 * logged via {@link PaintLogger}. Returned entries are ready for direct use in
074 * the PAINT viewer.</p>
075 */
076public class RecordingLoader {
077
078    /**
079     * Loads all valid {@link RecordingEntry} objects from a project.
080     *
081     * @param project the project containing experiment directories and metadata
082     * @return list of validated {@link RecordingEntry} objects; empty if none valid
083     */
084    public static List<RecordingEntry> loadFromProject(Project project) {
085
086        List<RecordingEntry> recordingEntries = new ArrayList<>();
087
088        for (String experimentName : project.getExperimentNames()) {
089
090            Path experimentPath = project.getProjectRootPath().resolve(experimentName);
091
092            // ------------------------------------------------------------------
093            // Load full experiment (squares only; tracks intentionally disabled)
094            // ------------------------------------------------------------------
095            Experiment experiment = ExperimentDataLoader.loadExperiment(
096                    project.getProjectRootPath(),
097                    experimentName,
098                    true,   // load squares
099                    false   // do not load tracks
100            );
101
102            if (experiment == null || experiment.getRecordings().isEmpty()) {
103                PaintLogger.warnf("Experiment '%s' contains no valid recordings.", experimentName);
104                continue;
105            }
106
107            for (Recording recording : experiment.getRecordings()) {
108
109                String recordingName = recording.getRecordingName();
110
111                // Skip recordings not flagged for processing
112                if (!recording.isProcessFlagSet()) {
113                    continue;
114                }
115
116                // ------------------------------------------------------------------
117                // TrackMate image (*.jpg)
118                // ------------------------------------------------------------------
119                Path trackmateImagePath = experimentPath
120                        .resolve("TrackMate Images")
121                        .resolve(recordingName + ".jpg");
122
123                if (!Files.exists(trackmateImagePath)) {
124                    PaintLogger.errorf(
125                            "Missing TrackMate image for recording '%s' (%s)",
126                            recordingName, trackmateImagePath
127                    );
128                    continue;
129                }
130
131                // ------------------------------------------------------------------
132                // Brightfield images (recording*.jpg or recording-BF*.jpg)
133                // ------------------------------------------------------------------
134                Path brightfieldDirPath = experimentPath.resolve("Brightfield Images");
135
136                if (!Files.isDirectory(brightfieldDirPath)) {
137                    PaintLogger.errorf(
138                            "Missing Brightfield directory '%s' for recording '%s'",
139                            brightfieldDirPath, recordingName
140                    );
141                    continue;
142                }
143
144                Path brightfieldImagePath = null;
145
146                try (java.util.stream.Stream<Path> stream = Files.list(brightfieldDirPath)) {
147                    for (Iterator<Path> it = stream.iterator(); it.hasNext(); ) {
148                        Path p = it.next();
149                        String fileName = p.getFileName().toString();
150
151                        boolean isMatch =
152                                (fileName.startsWith(recordingName + "-BF") ||
153                                        fileName.startsWith(recordingName))
154                                        && fileName.endsWith(".jpg");
155
156                        if (isMatch) {
157                            brightfieldImagePath = p;
158                            break;
159                        }
160                    }
161                } catch (Exception e) {
162                    PaintLogger.errorf("Error scanning Brightfield directory: %s", e.getMessage());
163                }
164
165                if (brightfieldImagePath == null) {
166                    PaintLogger.warnf("Missing Brightfield image for recording '%s'", recordingName);
167                    continue;
168                }
169
170                // ------------------------------------------------------------------
171                // Construct final entry
172                // ------------------------------------------------------------------
173                RecordingEntry entry = new RecordingEntry(
174                        recording,
175                        trackmateImagePath,
176                        brightfieldImagePath,
177                        experimentName
178                );
179
180                recordingEntries.add(entry);
181            }
182        }
183
184        return recordingEntries;
185    }
186}
