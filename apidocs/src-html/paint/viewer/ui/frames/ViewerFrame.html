085/**
086 * The {@code ViewerFrame} class defines the main window of the PAINT Viewer.
087 * It combines left and right image panels, navigation controls, and metadata panels into
088 * a cohesive interface for browsing, filtering, and analyzing experiment recordings.
089 * <p>
090 * Functionally, the class manages synchronization between user interactions and the
091 * underlying model objects — specifically {@link RecordingEntry}
092 * and {@link paint.shared.objects.Recording}. It enables users to:
093 * <ul>
094 *   <li>Navigate through multiple recordings within an experiment.</li>
095 *   <li>Filter visible squares using configurable thresholds.</li>
096 *   <li>Assign cell IDs interactively via selection dialogs.</li>
097 *   <li>Play TIFF/ND2 recordings directly from disk.</li>
098 *   <li>Preview live recalculations of Tau, R², and density metrics.</li>
099 * </ul>
100 * <p>
101 * This class is instantiated by {@link paint.viewer.app.Viewer} after successful
102 * project initialization. All UI updates occur on the Swing event dispatch thread.

140    /**
141     * Constructs a {@code RecordingViewerFrame} that initializes and displays the complete
142     * recording viewer environment. The frame sets up grid visualization, navigation,
143     * control, and attribute panels while establishing event connections for user actions.
144     *
145     * @param project          the {@link Project} object providing experiment context and paths.
146     * @param recordingEntries list of {@link RecordingEntry} objects representing loaded recordings.
147     */
148    public ViewerFrame(Project project, List<RecordingEntry> recordingEntries) {
149        super("Recording Viewer - " + project.getProjectRootPath().getFileName());
150        this.project                 = project;
151        this.allRecordingEntries     = new ArrayList<>(recordingEntries);                          // This is the unfiltered set of recordings
152        this.recordingEntries        = new ArrayList<>(recordingEntries);                          // This is the filtered set of recordings
153        this.writeRecordingOverride  = new WriteRecordingOverride(project.getProjectRootPath());
154        this.writeSquareOverride     = new WriteSquareOverride(project.getProjectRootPath());
155        this.navigator               = new RecordingNavigator(this::showRecordingEntry);
156
157        setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
158        setLayout(new BorderLayout());
159        setResizable(false);
160
161        // Validate grid configuration: can only be a limited set
162        int     numberOfSquaresInRecording = PaintConfig.getInt(GENERATE_SQUARES, NUMBER_OF_SQUARES_IN_RECORDING, -1);
163        int[]   validSquareLayouts         = {25, 100, 225, 400, 900};
164        boolean isValidSquareLayout        = false;
165
166        for (int valid : validSquareLayouts) {
167            if (numberOfSquaresInRecording == valid) {
168                isValidSquareLayout = true;
169                break;
170            }
171        }
172        if (!isValidSquareLayout) {
173            PaintLogger.errorf("Invalid square layout (d x d)");
174            return;
175        }
176        int numberOfSquareInOneDimension = (int) Math.sqrt(numberOfSquaresInRecording);
177
178        // --- Build UI layout using the layout builder ---
179        ViewerLayoutBuilder builder = new ViewerLayoutBuilder();
180
181        ViewerLayoutBuilder.LayoutComponents ui =
182                builder.build(
183                        numberOfSquareInOneDimension,
184                        this,
185                        this,
186                        this
187                );
188        // Apply content pane
189        setContentPane(ui.rootPanel);
190
191        // Store references
192        this.leftGridPanel           = ui.leftGridPanel;
193        JLabel rightImageLabel       = ui.rightImageLabel;
194        JLabel experimentLabel       = ui.experimentLabel;
195        JLabel recordingLabel        = ui.recordingLabel;
196        this.attributesPanel         = ui.attributesPanel;
197        this.navigationPanel         = ui.navigationPanel;
198        this.controlsPanel           = ui.controlsPanel;
199        this.importOverridesCheckBox = ui.importOverridesCheckBox;
200
201        this.displayUpdater = new RecordingDisplayUpdater(
202                this.leftGridPanel,
203                rightImageLabel,
204                experimentLabel,
205                recordingLabel,
206                this.attributesPanel
207        );
208
209        // Wire behaviour for "Overrides" checkbox
210        initImportOverridesBehaviour();
211
212        setSize(1500, 700);
213        setLocationRelativeTo(null);
214
215        // Load the first entry if available
216        if (!recordingEntries.isEmpty()) {
217            showRecordingEntry(0);
218        }
219    }

221    /**
222     * Initializes startup and runtime behaviour for the "Import Overrides" checkbox:
223     * - If checked at startup → perform override import once.
224     * - If user checks it later → perform override import.
225     * - If user unchecks it → show a message asking to restart the viewer.
226     */
227    private void initImportOverridesBehaviour() {
228        if (importOverridesCheckBox == null) {
229            return;
230        }
231
232        // Set initial label based on initial state
233        importOverridesCheckBox.setText(
234                importOverridesCheckBox.isSelected() ? OVERRIDE_TEXT_ON : OVERRIDE_TEXT_OFF
235        );
236
237        importOverridesCheckBox.addItemListener(e -> {
238            boolean selected = importOverridesCheckBox.isSelected();
239
240            // Update title immediately
241            importOverridesCheckBox.setText(selected ? OVERRIDE_TEXT_ON : OVERRIDE_TEXT_OFF);
242
243            // Your existing behavior
244            if (selected) {
245                performImportOverrides();
246            } else {
247                // Optional: tell user they must restart to fully “unimport”
248                // JOptionPane.showMessageDialog(this, "To remove imported overrides, restart the viewer.", ...);
249            }
250        });
251
252        // Startup behavior: if already selected, import once
253        if (importOverridesCheckBox.isSelected()) {
254            performImportOverrides();
255        }
256
257        importOverridesCheckBox.setToolTipText(
258                "When enabled, reads override CSV files and applies them to the current session."
259        );
260    }

262    /**
263     * Performs the actual "Import Overrides" action by reading override CSV files
264     * and applying them to the current project's recording and square metadata.
265     */
266    private void performImportOverrides() {
267        PaintLogger.debugf("Overrides requested (checkbox is checked).");
268
269        // 1) Apply overrides to all RecordingEntry objects
270        importRecordingOverrides (allRecordingEntries, project.getProjectRootPath());
271        importSquareOverrides    (allRecordingEntries, project.getProjectRootPath());
272        importRecordingExcludes  (allRecordingEntries, project.getProjectRootPath());
273
274        // 2) Re-apply filter for the currently visible recording
275        if (recordingEntries.isEmpty()
276                || currentIndex < 0
277                || currentIndex >= recordingEntries.size()) {
278            leftGridPanel.repaint();
279            return;
280        }
281
282        RecordingEntry current = recordingEntries.get(currentIndex);
283
284        // Build params from the now-updated recording
285        SquareControlParams params = new SquareControlParams(
286                current.getRecording().getMinRequiredDensityRatio(),
287                current.getRecording().getMaxAllowableVariability(),
288                current.getRecording().getMinRequiredRSquared(),
289                current.getRecording().getNeighbourMode()
290        );
291
292        // 3) Re-apply control handler + filter to the grid
293        controlHandler.apply(params, leftGridPanel);
294        leftGridPanel.applyVisibilityFilter();
295
296        // 4) Update the attributes panel preview to reflect the new thresholds
297        int numSquares = PaintConfig.getInt(GENERATE_SQUARES, NUMBER_OF_SQUARES_IN_RECORDING, -1);
298        attributesPanel.updatePreview(
299                current,
300                numSquares,
301                params.minRequiredDensityRatio,
302                params.maxAllowableVariability,
303                params.minRequiredRSquared,
304                params.neighbourMode
305        );
306
307        // 5) Redraw the left grid
308        leftGridPanel.repaint();
309
310        // Ensure UI reflects new exclude state after import
311        if (displayUpdater != null) {
312            displayUpdater.applyExcludedUi(current);
313        }
314        if (controlsPanel != null) {
315            controlsPanel.setExcludeButtonText(current.getRecording().isExcluded());
316        }
317    }

327    /**
328     * Displays a specific recording entry by index, updating both left and right
329     * image panels and all related attribute components.
330     *
331     * @param index the target recording index to display.
332     */
333    private void showRecordingEntry(int index) {
334        // Reset grid interaction deterministically
335        leftGridPanel.setInteractionEnabled(true);
336        leftGridPanel.setSelectionEnabled(false);
337        leftGridPanel.setInfoPopupsEnabled(true);
338
339        if (index < 0 || index >= recordingEntries.size()) {
340            return;
341        }
342
343        assignmentManager.clear();
344
345        // clear grid selection
346        leftGridPanel.clearSelection();
347        leftGridPanel.applyVisibilityFilter();
348
349        currentIndex = index;
350        RecordingEntry entry = recordingEntries.get(index);
351        displayUpdater.show(entry, index, recordingEntries.size());
352        updateNavButtons();
353
354        // --- Ensure Exclude/Include UI always matches the current recording ---
355        if (controlsPanel != null) {
356            controlsPanel.setExcludeButtonText(entry.getRecording().isExcluded());
357        }
358    }

476    /**
477     * Invoked when the user opens the filter dialog.
478     * The dialog allows narrowing the visible recording list by user-defined criteria.
479     * Once the user confirms, the filtered result is applied immediately to the viewer.
480     */
481    @Override
482    public void onFilterRequested() {
483        if (activeDialog != null && activeDialog.isShowing()) {
484            Toolkit.getDefaultToolkit().beep();
485            return;
486        }
487
488        setActionButtonsEnabled(false);
489        leftGridPanel.hideSquareInfoIfVisible();
490        activeDialog = new RecordingFilterDialog(this, recordingEntries, allRecordingEntries, lastFilterCriteria);
491        setGridEnabled(false); // Disable grid interaction
492
493        activeDialog.addWindowListener(new java.awt.event.WindowAdapter() {
494            public void windowClosed(java.awt.event.WindowEvent e) {
495                setGridEnabled(true); // Re-enable grid
496                setActionButtonsEnabled(true);
497                activeDialog = null;
498            }
499        });
500
501        activeDialog.setVisible(true);
502
503        // Now handle user result after closing
504        RecordingFilterDialog fd = (RecordingFilterDialog) activeDialog;
505        if (!fd.isCancelled()) {
506            List<RecordingEntry> filtered = fd.getFilteredRecordings();
507            lastFilterCriteria = fd.getCurrentFilterCriteria();
508            boolean isResetAll = (filtered.size() == allRecordingEntries.size());
509
510            if (isResetAll) {
511                recordingEntries.clear();
512                recordingEntries.addAll(allRecordingEntries);
513                currentIndex = 0;
514                showRecordingEntry(currentIndex);
515                PaintLogger.infof("Filter reset — showing all recordings (%d total).", recordingEntries.size());
516                return;
517            }
518
519            if (!filtered.isEmpty()) {
520                recordingEntries.clear();
521                recordingEntries.addAll(filtered);
522                currentIndex = 0;
523                showRecordingEntry(currentIndex);
524                PaintLogger.infof("Filter applied — showing %d of %d recordings.", filtered.size(), allRecordingEntries.size());
525            } else {
526                JOptionPane.showMessageDialog(
527                        this,
528                        "No recordings match the selected filter criteria.",
529                        "No Results",
530                        JOptionPane.INFORMATION_MESSAGE
531                );
532            }
533        }
534    }

577    /**
578     * Opens a dialog for assigning a cell ID to the user-selected squares.
579     * If previous assignments exist for the current recording, the user is
580     * asked whether to merge the new assignments with the old ones or to
581     * replace all existing assignments entirely.
582     * A CANCEL option aborts the operation without modifying any data.
583     * Includes undo and cancel functions and temporarily disables regular grid
584     * selection until the dialog is closed.
585     */
586    @Override
587    public void onAssignCellsRequested() {
588        if (activeDialog != null && activeDialog.isShowing()) {
589            Toolkit.getDefaultToolkit().beep();
590            return;
591        }
592
593        setActionButtonsEnabled(false);
594        leftGridPanel.hideSquareInfoIfVisible();
595        leftGridPanel.setSelectionEnabled(true);
596        leftGridPanel.setInfoPopupsEnabled(false);
597
598        final JFrame owner = this;
599        activeDialog = new CellAssignmentDialog(owner, new CellAssignmentDialog.Listener() {
600            public void onAssign(int cellId) {
601                handleCellAssignment(cellId);
602            }
603
604            public void onUndo() {
605                assignmentManager.undo(leftGridPanel);
606            }
607
608            public void onCancelSelection() {
609                leftGridPanel.clearSelection();
610                leftGridPanel.applyVisibilityFilter();
611                leftGridPanel.repaint();
612            }
613        });
614
615        activeDialog.addWindowListener(new java.awt.event.WindowAdapter() {
616            public void windowClosed(java.awt.event.WindowEvent e) {
617                leftGridPanel.setSelectionEnabled(false);
618                leftGridPanel.setInfoPopupsEnabled(true);
619                setActionButtonsEnabled(true);
620                activeDialog = null;
621            }
622        });
623
624        activeDialog.setVisible(true);
625    }

627    private void handleCellAssignment(int cellId) {
628        Map<Integer, Integer> userSelectedSquares = assignmentManager.assignUserSelectedSquares(cellId, leftGridPanel);
629
630        if (userSelectedSquares.isEmpty()) {
631            return;
632        }
633
634        RecordingEntry current                = recordingEntries.get(currentIndex);
635        boolean        hasExistingOverrides   = writeSquareOverride.hasOverridesFor(current);
636        boolean        existingOverridesShown = (importOverridesCheckBox != null && importOverridesCheckBox.isSelected());
637        boolean        keepOld                = true;
638
639        // If there are existing overrides and if thgey are not shown ask
640        if (hasExistingOverrides && !existingOverridesShown && assignmentManager.isFirstAssignmentForRecording()) {
641            int choice = JOptionPane.showConfirmDialog(
642                    this,
643                    "There are existing cell assignments for this recording.\n\n" +
644                            "Do you want to keep those and only update the selected squares?\n\n" +
645                            "Choose 'Yes' to merge assignments.\n" +
646                            "Choose 'No' to replace all existing assignments.",
647                    "Existing Assignments Found",
648                    JOptionPane.YES_NO_CANCEL_OPTION,
649                    JOptionPane.QUESTION_MESSAGE
650            );
651
652            if (choice == JOptionPane.CANCEL_OPTION || choice == JOptionPane.CLOSED_OPTION) {
653                return;
654            }
655
656            keepOld = (choice == JOptionPane.YES_OPTION);
657        }
658        assignmentManager.setFirstAssignmentForRecording(false);
659
660        if (keepOld) {
661            writeSquareOverride.mergeSquareOverrides(current, userSelectedSquares);
662        } else {
663            writeSquareOverride.replaceSquareOverrides(current, userSelectedSquares);
664        }
665    }

708    /**
709     * Applies square control parameters from the dialog to the current grid or project scope.
710     * <p>
711     * When invoked in "Preview" mode, recalculates Tau, R², and density values dynamically
712     * without committing them to disk. For full application, thresholds are persisted via
713     * {@link WriteRecordingOverride}.
714     *
715     * @param scope  the operational scope ("Preview" or "Apply").
716     * @param params parameter bundle defining the visibility thresholds and neighbour mode.
717     */
718
719    @Override
720    public void onApplySquareControl(SquareControlDialog.SquareControlListener.Scope scope,
721            SquareControlParams params) {
722
723        if (scope == SquareControlDialog.SquareControlListener.Scope.PREVIEW) {
724            controlHandler.apply(params, leftGridPanel);
725            leftGridPanel.applyVisibilityFilter();
726
727            RecordingEntry currentRecordingEntry = recordingEntries.get(currentIndex);
728            int numSquares = PaintConfig.getInt(GENERATE_SQUARES, NUMBER_OF_SQUARES_IN_RECORDING, -1);
729
730            //  --- Reflect results in attribute preview panel ---
731            attributesPanel.updatePreview(
732                    currentRecordingEntry,
733                    numSquares,
734                    params.minRequiredDensityRatio,
735                    params.maxAllowableVariability,
736                    params.minRequiredRSquared,
737                    params.neighbourMode
738            );
739
740            leftGridPanel.repaint();
741            return;
742        }
743
744        // Full application: persist thresholds and repaint
745        controlHandler.apply(params, leftGridPanel);
746        writeRecordingOverride.writeRecordingOverridesToFile(scope.label(), params, recordingEntries, currentIndex);
747        leftGridPanel.repaint();
748    }

754    /**
755     * Initiates playback of the ND2 or TIFF file corresponding to the current recording.
756     * Performs validation of user selection and file presence before launching playback.
757     * Errors and missing files are reported via dialog boxes.
758     */
759
760    @Override
761    public void onPlayRecordingRequested() {
762        if (activeDialog != null && activeDialog.isShowing()) {
763            Toolkit.getDefaultToolkit().beep();
764            return;
765        }
766        if (playbackController.isPlaying()) {
767            Toolkit.getDefaultToolkit().beep();
768            return;
769        }
770        if (recordingEntries.isEmpty() || currentIndex < 0 || currentIndex >= recordingEntries.size()) {
771            JOptionPane.showMessageDialog(this,
772                                          "No recording selected to play.",
773                                          "No Selection",
774                                          JOptionPane.WARNING_MESSAGE);
775            return;
776        }
777        RecordingEntry entry = recordingEntries.get(currentIndex);
778        playbackController.playRecording(entry);
779    }

841    @Override
842    public void onClose() {
843        if (importOverridesCheckBox != null && importOverridesCheckBox.isSelected()) {
844
845            int choice = JOptionPane.showConfirmDialog(
846                    this,
847                    "You are about to save override data into '-override' files.\n"
848                            + "This may take a while.\n\n"
849                            + "Do you want to save overrides now?",
850                    "Process Overrides?",
851                    JOptionPane.YES_NO_CANCEL_OPTION,
852                    JOptionPane.WARNING_MESSAGE
853            );
854
855            if (choice == JOptionPane.CANCEL_OPTION) {
856                // User aborted closing
857                return;
858            }
859
860            if (choice == JOptionPane.YES_OPTION) {
861                try {
862                    exportOverrides(project.getProjectRootPath(), "-override");
863                } catch (Exception ex) {
864                    ex.printStackTrace();
865                    JOptionPane.showMessageDialog(
866                            this,
867                            "Error while processing overrides:\n" + ex.getMessage(),
868                            "Error",
869                            JOptionPane.ERROR_MESSAGE
870                    );
871                }
872            }
873
874            // NO_OPTION → skip override processing and continue closing
875        }
876
877        // Always close the window afterwards
878        this.dispose();
879    }
