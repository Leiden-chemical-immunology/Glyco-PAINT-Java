022 *
023 *  KEY FEATURES:
024 *    • Fully refreshes the grid, images, labels, and attributes panel.
025 *    • Performs deterministic high-quality scaling of the right-side image.
026 *    • Reads the expected square count from {@link PaintConfig}.

042package paint.viewer.ui;
043
044import paint.shared.config.paintconfig.PaintConfig;
045import paint.viewer.model.RecordingEntry;
046import paint.viewer.ui.panels.SquareGridPanel;
047import paint.viewer.ui.panels.RecordingAttributesPanel;
048
049import javax.swing.*;
050import java.awt.Color;
051
052import static paint.shared.constants.PaintStringConstants.GENERATE_SQUARES;
053import static paint.shared.constants.PaintStringConstants.NUMBER_OF_SQUARES_IN_RECORDING;
054
055import static paint.shared.constants.PaintGeometry.NUMBER_PIXELS_HEIGHT;
056import static paint.shared.constants.PaintGeometry.NUMBER_PIXELS_WIDTH;
057
058/**
059 * Utility class responsible for synchronizing the UI with a newly selected
060 * {@link RecordingEntry}. Called whenever the user navigates between recordings
061 * in the viewer.
062 *
063 * <p>This updater replaces the displayed grid, rescaled images, labels, and
064 * attribute table to reflect the active recording. No state is stored inside
065 * this class—its job is to apply a complete UI refresh against the components
066 * provided to the constructor.</p>
067 */
068public final class RecordingDisplayUpdater {
069
070    private final SquareGridPanel          leftGridPanel;
071    private final JLabel                   rightImageLabel;
072    private final JLabel                   experimentLabel;
073    private final JLabel                   recordingLabel;
074    private final RecordingAttributesPanel attributesPanel;
075    private final Color                    defaultRecordingLabelColor;
076
077    /**
078     * Creates a new updater bound to the UI components it controls.
079     *
080     * @param leftGridPanel     the grid used for displaying square layout and overlays
081     * @param rightImageLabel   the label used to show the right-side image
082     * @param experimentLabel   the label displaying the experiment name
083     * @param recordingLabel    the label displaying the recording name
084     * @param attributesPanel   the table panel showing recording metadata
085     */
086    public RecordingDisplayUpdater(
087            SquareGridPanel leftGridPanel,
088            JLabel rightImageLabel,
089            JLabel experimentLabel,
090            JLabel recordingLabel,
091            RecordingAttributesPanel attributesPanel
092    ) {
093        this.leftGridPanel              = leftGridPanel;
094        this.rightImageLabel            = rightImageLabel;
095        this.experimentLabel            = experimentLabel;
096        this.recordingLabel             = recordingLabel;
097        this.attributesPanel            = attributesPanel;
098        this.defaultRecordingLabelColor = recordingLabel.getForeground();
099    }
100
101    /**
102     * Updates all UI components to show the given recording entry.
103     *
104     * @param entry      the new {@link RecordingEntry} to display
105     * @param index      the zero-based index of the entry in the full list
106     * @param totalSize  the total number of loaded recording entries
107     */
108    public void show(RecordingEntry entry, int index, int totalSize) {
109
110        // --- Left grid panel ---
111        leftGridPanel.setBackgroundImage(entry.getLeftImage());
112        leftGridPanel.setSquares(entry.getRecording().getSquaresOfRecording());
113
114        // --- Right image ---
115        ImageIcon scaled = new ImageIcon(
116                entry.getRightImage().getImage().getScaledInstance(
117                        NUMBER_PIXELS_WIDTH,
118                        NUMBER_PIXELS_HEIGHT,
119                        java.awt.Image.SCALE_SMOOTH
120                )
121        );
122        rightImageLabel.setIcon(scaled);
123
124        // --- Labels ---
125        experimentLabel.setText(
126                "Experiment: " + entry.getExperimentName()
127                        + "   [" + (index + 1) + "/" + totalSize + "]"
128        );
129
130        applyExcludedUi(entry);
131
132        // --- Attribute table ---
133        int numberOfSquares = PaintConfig.getInt(
134                GENERATE_SQUARES,
135                NUMBER_OF_SQUARES_IN_RECORDING,
136                -1
137        );
138
139        attributesPanel.updateFromEntry(entry, numberOfSquares);
140
141        // --- Refresh grid ---
142        leftGridPanel.repaint();
143    }
144
145    /**
146     * Updates the recording label to reflect whether the current recording is excluded.
147     * If excluded, appends "(Excluded)" and sets the label text to red.
148     */
149    public void applyExcludedUi(RecordingEntry entry) {
150        final boolean excluded = isExcluded(entry);
151        if (excluded) {
152            recordingLabel.setText("Recording: " + entry.getRecordingName() + " (Excluded)");
153            recordingLabel.setForeground(Color.RED);
154        } else {
155            recordingLabel.setText("Recording: " + entry.getRecordingName());
156            recordingLabel.setForeground(defaultRecordingLabelColor);
157        }
158    }
159
160    /**
161     * Attempts to read an "excluded" flag from the RecordingEntry / Recording model.
162     */
163    private static boolean isExcluded(RecordingEntry recordingEntry) {
164        if (recordingEntry == null) {
165            return false;
166        }
167
168        return recordingEntry.getRecording().isExcluded();
169    }
170}
