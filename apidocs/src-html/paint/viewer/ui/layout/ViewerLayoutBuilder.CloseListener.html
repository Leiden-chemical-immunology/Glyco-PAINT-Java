003 *  Package:      paint.viewer.ui.layout
004 *
005 *  PURPOSE:
006 *    Constructs the complete Swing user interface layout for the PAINT viewer.
007 *    Centralises creation and assembly of all major UI components used by
008 *    ViewerFrame, providing them in a convenient container object.
009 *
010 *  DESCRIPTION:
011 *    This builder constructs:
012 *      • The square grid panel (left image)
013 *      • The processed / reference image panel (right image)
014 *      • Experiment/recording labels
015 *      • Navigation controls
016 *      • Recording attributes panel
017 *      • Recording controls panel
018 *      • Import-overrides checkbox
019 *      • Close button and bottom panel
020 *
021 *    The method {@code build(...)} returns a {@link LayoutComponents} object
022 *    containing references to all created Swing components required for UI
023 *    updates inside ViewerFrame.
024 *
025 *  KEY FEATURES:
026 *    • Fully encapsulates all layout wiring for the viewer.
027 *    • Ensures stable component references for ViewerFrame.
028 *    • Provides custom layout containers with fixed aspect ratios for images.
029 *    • Offers a clean data container (LayoutComponents) for external use.

077package paint.viewer.ui.layout;

079import paint.viewer.ui.panels.NavigationPanel;
080import paint.viewer.ui.panels.RecordingAttributesPanel;
081import paint.viewer.ui.panels.RecordingControlsPanel;

084import javax.swing.*;
085import javax.swing.border.Border;
086import java.awt.*;

088import static paint.shared.constants.PaintGeometry.NUMBER_PIXELS_HEIGHT;
089import static paint.shared.constants.PaintGeometry.NUMBER_PIXELS_WIDTH;

091/**
092 * Builds the full layout structure for the ViewerFrame. All components required
093 * for UI updates (labels, grid panels, navigation, attributes, controls, etc.)
094 * are created here and returned in a {@link LayoutComponents} container.
095 */
096public class ViewerLayoutBuilder {
097
098    /**
099     * Listener for the viewer close event.
100     */
101    public interface CloseListener {
102        /**
103         * Called when the viewer should be closed.
104         */
105        void onClose();
106    }
107
108    /**
109     * Holds all top-level Swing components created by the layout builder.
110     * {@link paint.viewer.ui.frames.ViewerFrame} uses this object to update the
111     * UI during navigation.
112     */
113    public static class LayoutComponents {
114        /** The main root panel of the viewer. */
115        public final JPanel                   rootPanel;
116        /** Label displaying the right-side (reference) image. */
117        public final JLabel                   rightImageLabel;
118        /** Label displaying the current experiment name. */
119        public final JLabel                   experimentLabel;
120        /** Label displaying the current recording name. */
121        public final JLabel                   recordingLabel;
122        /** Panel containing the grid of squares. */
123        public final SquareGridPanel          leftGridPanel;
124        /** Panel displaying recording metadata attributes. */
125        public final RecordingAttributesPanel attributesPanel;
126        /** Panel containing navigation buttons. */
127        public final NavigationPanel          navigationPanel;
128        /** Panel containing interactive recording controls. */
129        public final RecordingControlsPanel   controlsPanel;
130        /** Checkbox for enabling override import. */
131        public final JCheckBox                importOverridesCheckBox;
132
133        /**
134         * Constructs a new {@code LayoutComponents} container.
135         *
136         * @param rootPanel               the root panel
137         * @param rightImageLabel         right image label
138         * @param experimentLabel         experiment label
139         * @param recordingLabel          recording label
140         * @param leftGridPanel           left grid panel
141         * @param attributesPanel         attributes panel
142         * @param navigationPanel         navigation panel
143         * @param controlsPanel           controls panel
144         * @param importOverridesCheckBox overrides checkbox
145         */
146        LayoutComponents(
147                JPanel                   rootPanel,
148                JLabel                   rightImageLabel,
149                JLabel                   experimentLabel,
150                JLabel                   recordingLabel,
151                SquareGridPanel          leftGridPanel,
152                RecordingAttributesPanel attributesPanel,
153                NavigationPanel          navigationPanel,
154                RecordingControlsPanel   controlsPanel,
155                JCheckBox                importOverridesCheckBox
156        ) {
157            this.rootPanel               = rootPanel;
158            this.rightImageLabel         = rightImageLabel;
159            this.experimentLabel         = experimentLabel;
160            this.recordingLabel          = recordingLabel;
161            this.leftGridPanel           = leftGridPanel;
162            this.attributesPanel         = attributesPanel;
163            this.navigationPanel         = navigationPanel;
164            this.controlsPanel           = controlsPanel;
165            this.importOverridesCheckBox = importOverridesCheckBox;
166        }
167    }
168
169    /**
170     * Creates the full viewer layout, assembles all Swing components into a
171     * top-level root panel, and returns the complete set of components that
172     * ViewerFrame will retain and update during interaction.
173     *
174     * @param gridDim             dimension of the square grid (gridDim × gridDim)
175     * @param navigationListener  listener for previous/next recording navigation
176     * @param recordingsControlListener    listener for recording-specific control actions
177     * @param closeListener       callback for closing the viewer
178     * @return a {@link LayoutComponents} instance containing all UI components
179     */
180    public LayoutComponents build(
181            int                                              gridDim,
182            NavigationPanel.Listener                         navigationListener,
183            RecordingControlsPanel.RecordingsControlListener recordingsControlListener,
184            CloseListener                                    closeListener) {
185
186        // --- Components created here so ViewerFrame never constructs them itself ---
187        JLabel rightImageLabel = new JLabel("", SwingConstants.CENTER);
188        JLabel experimentLabel = new JLabel("", SwingConstants.CENTER);
189        JLabel recordingLabel  = new JLabel("", SwingConstants.CENTER);
190
191        SquareGridPanel          leftGridPanel   = new SquareGridPanel(gridDim, gridDim);
192        RecordingAttributesPanel attributesPanel = new RecordingAttributesPanel();
193        NavigationPanel          navigationPanel = new NavigationPanel(navigationListener);
194        RecordingControlsPanel   controlsPanel   = new RecordingControlsPanel(recordingsControlListener);
195
196        // --- Image panels side-by-side (left grid + right image) ---
197        JPanel imagesInner = new JPanel(new GridLayout(1, 2, 15, 0));
198        imagesInner.add(createSquareImagePanel(leftGridPanel));
199        imagesInner.add(createSquareImagePanel(rightImageLabel));
200
201        // --- Labels panel (Experiment + Recording) ---
202        JPanel labelsPanel = new JPanel(new GridLayout(2, 1));
203        labelsPanel.setBorder(BorderFactory.createEmptyBorder(5, 5, 5, 5));
204
205        experimentLabel.setFont(new Font("SansSerif", Font.BOLD, 13));
206        recordingLabel.setFont(new Font("SansSerif", Font.PLAIN, 12));
207
208        labelsPanel.add(experimentLabel);
209        labelsPanel.add(recordingLabel);
210
211        // --- Main image area with navigation ---
212        JPanel imagesWithNav = new JPanel(new BorderLayout(15, 15));
213        Border outer         = BorderFactory.createLineBorder(Color.DARK_GRAY, 2);
214        Border inner         = BorderFactory.createEmptyBorder(15, 15, 15, 15);
215        imagesWithNav.setBorder(BorderFactory.createCompoundBorder(outer, inner));
216
217        imagesWithNav.add(navigationPanel.getComponent(), BorderLayout.NORTH);
218        imagesWithNav.add(imagesInner,                    BorderLayout.CENTER);
219        imagesWithNav.add(labelsPanel,                    BorderLayout.SOUTH);
220
221        // --- Root layout: left attributes, center images, right controls ---
222        JPanel root = new JPanel(new BorderLayout());
223        root.add(attributesPanel.getComponent(), BorderLayout.WEST);
224        root.add(imagesWithNav,                  BorderLayout.CENTER);
225        root.add(controlsPanel.getComponent(),   BorderLayout.EAST);
226
227        // --- Bottom panel with import overrides checkbox + close button ---
228        JPanel    bottomPanel             = new JPanel(new BorderLayout(10, 10));
229        JCheckBox importOverridesCheckBox = new JCheckBox("Overrides");
230        bottomPanel.add(importOverridesCheckBox, BorderLayout.WEST);
231        JButton    closeButton            = new JButton("Close Viewer");
232        closeButton.addActionListener(e -> closeListener.onClose());
233
234        JPanel rightBox = new JPanel(new FlowLayout(FlowLayout.RIGHT, 10, 10));
235        rightBox.add(closeButton);
236        bottomPanel.add(rightBox, BorderLayout.EAST);
237
238        root.add(bottomPanel, BorderLayout.SOUTH);
239
240        // --- Return all components (ViewerFrame consumes LayoutComponents) ---
241        return new LayoutComponents(
242                root,
243                rightImageLabel,
244                experimentLabel,
245                recordingLabel,
246                leftGridPanel,
247                attributesPanel,
248                navigationPanel,
249                controlsPanel,
250                importOverridesCheckBox
251        );
252    }
253
254    /**
255     * Creates a bordered panel that forces a fixed square aspect ratio based on
256     * NUMBER_PIXELS_WIDTH and NUMBER_PIXELS_HEIGHT. Both the grid and right-image
257     * panels use this container.
258     */
259    private JPanel createSquareImagePanel(JComponent comp) {
260        JPanel panel = new JPanel(new BorderLayout()) {
261            public Dimension getPreferredSize() {
262                return new Dimension(NUMBER_PIXELS_WIDTH, NUMBER_PIXELS_HEIGHT);
263            }
264
265            public void setBounds(int x, int y, int w, int h) {
266                int size = Math.min(w, h);
267                super.setBounds(x, y, size, size);
268            }
269        };
270
271        panel.setBorder(BorderFactory.createLineBorder(Color.BLACK, 1));
272        panel.add(comp, BorderLayout.CENTER);
273        return panel;
274    }
