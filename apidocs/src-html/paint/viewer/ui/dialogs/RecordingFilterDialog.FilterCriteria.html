046/**
047 * Modal dialog to filter a list of {@link RecordingEntry} objects by metadata.
048 * <p>
049 * Supports multi-column filtering with dynamic cross-filtering, allowing users to
050 * refine recording sets based on Cell Type, Probe Name, Probe Type, Adjuvant,
051 * and Concentration.
052 */
053public class RecordingFilterDialog extends JDialog {
054
055    // ----- Data sets -----
056    private final List<RecordingEntry> originalRecordings; // full dataset (for Reset All)
057    private       List<RecordingEntry> filteredRecordings; // current filtered result
058
059    // ----- UI lists -----
060    private final JList<String> cellTypeList;
061    private final JList<String> probeNameList;
062    private final JList<String> probeTypeList;
063    private final JList<String> adjuvantList;
064    private final JList<String> concentrationList;
065
066    // ----- Book-keeping for highlighting and per-column behavior -----
067    private final Map<JList<String>, JPanel> listPanels                                = new HashMap<>();
068    private final Map<JList<String>, JButton> resetButtons                             = new HashMap<>();
069    private final Map<JList<String>, String> titles                                    = new HashMap<>();
070    private final Map<JList<String>, Function<RecordingEntry, String>> valueExtractors = new HashMap<>();
071    private final Map<JList<String>, List<String>> selections                          = new HashMap<>();
072    private final Set<JList<String>> activeFilters                                     = new HashSet<>();
073
074    private       boolean            cancelled                                         = true;
075
076    /**
077     * Constructs a new filter dialog.
078     *
079     * @param owner                    the parent frame
080     * @param currentVisibleRecordings the recordings currently shown in the viewer
081     * @param allRecordings           all available recordings in the project
082     * @param initialCriteria         previously applied filter criteria (optional)
083     */
084    public RecordingFilterDialog(Frame owner,
085                                 List<RecordingEntry> currentVisibleRecordings,
086                                 List<RecordingEntry> allRecordings,
087                                 FilterCriteria initialCriteria) {
088        super(owner, "Filter Recordings", true);
089
090        final FilterCriteria criteria = (initialCriteria == null) ? FilterCriteria.empty() : initialCriteria;
091
092        setLayout(new BorderLayout(10, 10));
093        setResizable(false);
094
095        // Store full dataset
096        this.originalRecordings = new ArrayList<>(allRecordings);
097
098        // Compute the base list (start subset) from the current visible subset and incoming criteria
099        // visible subset when dialog opened
100        List<RecordingEntry> baseList;
101        if (criteria.isNotEmpty()) {
102            baseList = currentVisibleRecordings.stream()
103                                               .filter(e -> criteria.cellTypes.isEmpty()      || criteria.cellTypes.contains(e.getCellType()))
104                                               .filter(e -> criteria.probeNames.isEmpty()     || criteria.probeNames.contains(e.getProbeName()))
105                                               .filter(e -> criteria.probeTypes.isEmpty()     || criteria.probeTypes.contains(e.getProbeType()))
106                                               .filter(e -> criteria.adjuvants.isEmpty()      || criteria.adjuvants.contains(e.getAdjuvant()))
107                                               .filter(e -> criteria.concentrations.isEmpty() || criteria.concentrations.contains(String.valueOf(e.getConcentration())))
108                                               .collect(Collectors.toList());
109        } else {
110            baseList = new ArrayList<>(currentVisibleRecordings);
111        }
112
113        this.filteredRecordings = new ArrayList<>(baseList);
114
115        // Build initial list box contents from baseList (not the full dataset)
116        Set<String> cellTypes      = baseList.stream()
117                                             .map(RecordingEntry::getCellType)
118                                             .collect(Collectors.toCollection(TreeSet::new));
119        Set<String> probeNames     = baseList.stream()
120                                             .map(RecordingEntry::getProbeName)
121                                             .collect(Collectors.toCollection(TreeSet::new));
122        Set<String> probeTypes     = baseList.stream()
123                                             .map(RecordingEntry::getProbeType)
124                                             .collect(Collectors.toCollection(TreeSet::new));
125        Set<String> adjuvants      = baseList.stream()
126                                             .map(RecordingEntry::getAdjuvant)
127                                             .collect(Collectors.toCollection(TreeSet::new));
128        Set<String> concentrations = baseList.stream()
129                                             .map(e -> String.valueOf(e.getConcentration()))
130                                             .collect(Collectors.toCollection(TreeSet::new));
131
132        cellTypeList      = createList(cellTypes);
133        probeNameList     = createList(probeNames);
134        probeTypeList     = createList(probeTypes);
135        adjuvantList      = createList(adjuvants);
136        concentrationList = createList(concentrations);
137
138        // Register metadata for each list box
139        titles.put(cellTypeList,      CELL_TYPE);
140        titles.put(probeNameList,     PROBE_NAME);
141        titles.put(probeTypeList,     PROBE_TYPE);
142        titles.put(adjuvantList,      ADJUVANT);
143        titles.put(concentrationList, CONCENTRATION);
144
145        valueExtractors.put(cellTypeList,      RecordingEntry::getCellType);
146        valueExtractors.put(probeNameList,     RecordingEntry::getProbeName);
147        valueExtractors.put(probeTypeList,     RecordingEntry::getProbeType);
148        valueExtractors.put(adjuvantList,      RecordingEntry::getAdjuvant);
149        valueExtractors.put(concentrationList, e -> String.valueOf(e.getConcentration()));
150
151        // main lists row
152        JPanel listPanel = new JPanel(new GridLayout(1, 6, 10, 0));
153        listPanel.add(createListBoxWithButtons(cellTypeList));
154        listPanel.add(createListBoxWithButtons(probeNameList));
155        listPanel.add(createListBoxWithButtons(probeTypeList));
156        listPanel.add(createListBoxWithButtons(adjuvantList));
157        listPanel.add(createListBoxWithButtons(concentrationList));
158
159        // Right controls
160        JPanel rightButtonPanel = new JPanel();
161        rightButtonPanel.setLayout(new BoxLayout(rightButtonPanel, BoxLayout.Y_AXIS));
162
163        JButton applyButton    = new JButton("Apply");
164        JButton resetAllButton = new JButton("Reset All");
165        JButton cancelButton   = new JButton("Cancel");
166
167        Dimension btnSize = new Dimension(100, 30);
168        applyButton.setMaximumSize(btnSize);
169        resetAllButton.setMaximumSize(btnSize);
170        cancelButton.setMaximumSize(btnSize);
171
172        applyButton.addActionListener(e -> {
173            closeAction();
174        });
175
176        resetAllButton.addActionListener(e -> {
177            // Reset everything to the full dataset, clear all selections and highlights
178            filteredRecordings = new ArrayList<>(originalRecordings);
179            updateAllListsFrom(filteredRecordings);
180
181            // Clear remembered selections and active filter flags
182            for (JList<String> l : selections.keySet()) {
183                selections.put(l, Collections.emptyList());
184                l.clearSelection();
185            }
186            activeFilters.clear();
187            updateResetButtonStates();
188        });
189
190        cancelButton.addActionListener(e -> {
191            cancelled = true;
192            dispose();
193        });
194
195        rightButtonPanel.add(applyButton);
196        rightButtonPanel.add(Box.createVerticalStrut(10));
197        rightButtonPanel.add(resetAllButton);
198        rightButtonPanel.add(Box.createVerticalStrut(10));
199        rightButtonPanel.add(cancelButton);
200        rightButtonPanel.add(Box.createVerticalGlue());
201
202        listPanel.add(rightButtonPanel);
203        add(listPanel, BorderLayout.CENTER);
204        pack();
205        setLocationRelativeTo(owner);
206
207        // Initialize the selection map with empties
208        for (JList<String> l : Arrays.asList(cellTypeList, probeNameList, probeTypeList, adjuvantList, concentrationList)) {
209            selections.put(l, Collections.emptyList());
210        }
211
212        // Restore incoming criteria -> selections/activeFilters, apply, and highlight
213        if (criteria.isNotEmpty()) {
214            restoreSelectionsFromCriteria(criteria);
215            filteredRecordings = applyAllFilters(); // apply all selected lists
216            updateAllListsFrom(filteredRecordings);
217        }
218        updateResetButtonStates();
219
220        // Make sure that pressing the red close button is doung the same as cancel
221        setDefaultCloseOperation(WindowConstants.DO_NOTHING_ON_CLOSE);
222        addWindowListener(new java.awt.event.WindowAdapter() {
223            @Override
224            public void windowClosing(java.awt.event.WindowEvent e) {
225                closeAction();
226            }
227        });
228    }

232    private JList<String> createList(Set<String> values) {
233        JList<String> list = new JList<>(values.toArray(new String[0]));
234        list.setSelectionMode(ListSelectionModel.MULTIPLE_INTERVAL_SELECTION);
235        list.setVisibleRowCount(5);
236        return list;
237    }
238
239    private JPanel createListBoxWithButtons(JList<String> list) {
240        String title = titles.get(list);
241
242        JPanel panel = new JPanel(new BorderLayout());
243        panel.setBorder(BorderFactory.createTitledBorder(title));
244        listPanels.put(list, panel);
245
246        JScrollPane scrollPane = new JScrollPane(list);
247        scrollPane.setPreferredSize(new Dimension(140, 170));
248        panel.add(scrollPane, BorderLayout.CENTER);
249
250        JPanel btnPanel   = new JPanel(new GridLayout(2, 1, 0, 5));
251        JButton filterBtn = new JButton("Filter");
252        JButton resetBtn  = new JButton("Reset");
253        resetBtn.setEnabled(false); // initially disabled
254        resetButtons.put(list, resetBtn);
255
256        filterBtn.addActionListener(e -> onFilterClicked(list));
257        resetBtn.addActionListener(e -> onResetClicked(list));
258
259        btnPanel.add(filterBtn);
260        btnPanel.add(resetBtn);
261        panel.add(btnPanel, BorderLayout.SOUTH);
262        return panel;
263    }

266
267    private void onFilterClicked(JList<String> list) {
268        List<String> selected = list.getSelectedValuesList();
269        if (selected == null || selected.isEmpty()) {
270            return;
271        }
272
273        // Remember this column’s selection and mark it as active
274        selections.put(list, new ArrayList<>(selected));
275        activeFilters.add(list);
276
277        // Recompute from the full dataset (so filters stack properly)
278        filteredRecordings = applyAllFilters();
279
280        // Update visible lists to reflect the new filtered set
281        updateAllListsFrom(filteredRecordings);
282
283        // Restore previous selections for *all* active lists
284        for (Map.Entry<JList<String>, List<String>> entry : selections.entrySet()) {
285            reselect(entry.getKey(), entry.getValue());
286        }
287
288        // Update borders (highlight active lists)
289        updateResetButtonStates();
290    }
291
292    private void onResetClicked(JList<String> list) {
293        // Clear only THIS column’s selection and flag
294        selections.put(list, Collections.emptyList());
295        activeFilters.remove(list);
296
297        // Re-apply the remaining active filters to the base list
298        filteredRecordings = applyAllFilters();
299
300        // Update UI lists from the recomputed working set
301        updateAllListsFrom(filteredRecordings);
302
303        // Clear selection on this list; preserve selections on others
304        list.clearSelection();
305        for (Map.Entry<JList<String>, List<String>> e : selections.entrySet()) {
306            if (!e.getKey().equals(list) && !e.getValue().isEmpty()) {
307                reselect(e.getKey(), e.getValue());
308            }
309        }
310        updateResetButtonStates();
311    }

313    // ===== Core filtering =====
314
315    /**
316     * Apply all currently active filters (as recorded in 'selections') to the baseList.
317     */
318    private List<RecordingEntry> applyAllFilters() {
319        // Always reapply filters to the full dataset, not the baseList.
320        List<RecordingEntry> current = new ArrayList<>(originalRecordings);
321        for (Map.Entry<JList<String>, List<String>> entry : selections.entrySet()) {
322            JList<String> l = entry.getKey();
323            List<String> sel = entry.getValue();
324            if (sel == null || sel.isEmpty()) {
325                continue;
326            }
327
328            Function<RecordingEntry, String> extractor = valueExtractors.get(l);
329            current = current.stream()
330                    .filter(r -> sel.contains(extractor.apply(r)))
331                    .collect(Collectors.toList());
332        }
333        return current;
334    }

336    // ===== List content updates =====
337
338    private void updateAllListsFrom(List<RecordingEntry> source) {
339        updateList(cellTypeList,      collectDistinct(source, valueExtractors.get(cellTypeList)));
340        updateList(probeNameList,     collectDistinct(source, valueExtractors.get(probeNameList)));
341        updateList(probeTypeList,     collectDistinct(source, valueExtractors.get(probeTypeList)));
342        updateList(adjuvantList,      collectDistinct(source, valueExtractors.get(adjuvantList)));
343        updateList(concentrationList, collectDistinct(source, valueExtractors.get(concentrationList)));
344    }
345
346    private Set<String> collectDistinct(List<RecordingEntry> src, Function<RecordingEntry, String> fn) {
347        return src.stream().map(fn).collect(Collectors.toCollection(TreeSet::new));
348    }
349
350    private void updateList(JList<String> list, Set<String> values) {
351        list.setListData(values.toArray(new String[0]));
352    }
353
354    private void reselect(JList<String> list, List<String> values) {
355        if (values == null || values.isEmpty()) {
356            list.clearSelection();
357            return;
358        }
359        ListModel<String> model = list.getModel();
360        java.util.List<Integer> indices = new ArrayList<>();
361        for (int i = 0; i < model.getSize(); i++) {
362            String v = model.getElementAt(i);
363            if (values.contains(v)) {
364                indices.add(i);
365            }
366        }
367        int[] idxArr = indices.stream().mapToInt(Integer::intValue).toArray();
368        list.setSelectedIndices(idxArr);
369    }

371    // ===== Visual highlighting =====
372
373    private void updateResetButtonStates() {
374        for (Map.Entry<JList<String>, JButton> e : resetButtons.entrySet()) {
375            JList<String> list = e.getKey();
376            JButton button = e.getValue();
377            button.setEnabled(activeFilters.contains(list));
378        }
379    }

381    // ===== Criteria restore =====
382
383    private void restoreSelectionsFromCriteria(FilterCriteria filterCriteria) {
384        // Copy criteria into the selection map and mark active filters
385        setSelectionFrom(cellTypeList,      filterCriteria.cellTypes);
386        setSelectionFrom(probeNameList,     filterCriteria.probeNames);
387        setSelectionFrom(probeTypeList,     filterCriteria.probeTypes);
388        setSelectionFrom(adjuvantList,      filterCriteria.adjuvants);
389        setSelectionFrom(concentrationList, filterCriteria.concentrations);
390
391        // Reflect those selections in the JLists
392        reselect(cellTypeList,      selections.get(cellTypeList));
393        reselect(probeNameList,     selections.get(probeNameList));
394        reselect(probeTypeList,     selections.get(probeTypeList));
395        reselect(adjuvantList,      selections.get(adjuvantList));
396        reselect(concentrationList, selections.get(concentrationList));
397    }
398
399    private void setSelectionFrom(JList<String> list, List<String> values) {
400        if (values != null && !values.isEmpty()) {
401            selections.put(list, new ArrayList<>(values));
402            activeFilters.add(list);
403        } else {
404            selections.put(list, Collections.emptyList());
405            activeFilters.remove(list);
406        }
407    }

421    /**
422     * Data class holding selected filter criteria for recordings.
423     */
424    public static class FilterCriteria {
425        /** Selected cell types. */
426        public final List<String> cellTypes;
427        /** Selected probe names. */
428        public final List<String> probeNames;
429        /** Selected probe types. */
430        public final List<String> probeTypes;
431        /** Selected adjuvants. */
432        public final List<String> adjuvants;
433        /** Selected concentrations (stored as strings). */
434        public final List<String> concentrations;
435
436        /**
437         * Constructs a new {@code FilterCriteria} instance.
438         *
439         * @param cellTypes      list of selected cell types
440         * @param probeNames     list of selected probe names
441         * @param probeTypes     list of selected probe types
442         * @param adjuvants      list of selected adjuvants
443         * @param concentrations list of selected concentrations
444         */
445        public FilterCriteria(List<String> cellTypes,
446                              List<String> probeNames,
447                              List<String> probeTypes,
448                              List<String> adjuvants,
449                              List<String> concentrations) {
450            this.cellTypes      = cellTypes;
451            this.probeNames     = probeNames;
452            this.probeTypes     = probeTypes;
453            this.adjuvants      = adjuvants;
454            this.concentrations = concentrations;
455        }
456
457        /** @return a new empty filter criteria object */
458        public static FilterCriteria empty() {
459            return new FilterCriteria(
460                    Collections.emptyList(),
461                    Collections.emptyList(),
462                    Collections.emptyList(),
463                    Collections.emptyList(),
464                    Collections.emptyList()
465            );
466        }
467
468        /** @return true if no filters are active */
469        public boolean isEmpty() {
470            return cellTypes.isEmpty()
471                    && probeNames.isEmpty()
472                    && probeTypes.isEmpty()
473                    && adjuvants.isEmpty()
474                    && concentrations.isEmpty();
475        }
476
477        /** @return true if any filter is active */
478        public boolean isNotEmpty() {
479            return !isEmpty();
480        }
481    }
482
483    public FilterCriteria getCurrentFilterCriteria() {
484        return new FilterCriteria(
485                selections.get(cellTypeList),
486                selections.get(probeNameList),
487                selections.get(probeTypeList),
488                selections.get(adjuvantList),
489                selections.get(concentrationList)
490        );
491    }

493    private void closeAction() {
