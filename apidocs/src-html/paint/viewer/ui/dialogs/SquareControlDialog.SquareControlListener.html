055/**
056 * A dialog for configuring square control parameters in the viewer.
057 * <p>
058 * Provides interactive sliders and radio buttons that allow users to adjust
059 * key square filtering parameters. Changes are previewed live in the main
060 * viewer; pressing an Apply button commits and saves the configuration.
061 */
062public class SquareControlDialog extends JDialog {
063
064    /**
065     * Listener interface for responding to square control parameter changes.
066     */
067    public interface SquareControlListener {
068        /**
069         * Defines the persistence scope for applied parameters.
070         */
071        enum Scope {
072            /** Temporary preview in the UI. */
073            PREVIEW("Preview"),
074            /** Persist only for the current recording. */
075            RECORDING("Recording"),
076            /** Persist for all recordings in the current experiment. */
077            EXPERIMENT("Experiment"),
078            /** Persist for all experiments in the project. */
079            PROJECT("Project");
080
081            private final String label;
082
083            Scope(String label) {
084                this.label = label;
085            }
086
087            /** @return the display label for the scope */
088            public String label() {
089                return label;
090            }
091        }
092
093        /**
094         * Called when parameters are applied (or previewed).
095         *
096         * @param scope  the target persistence scope
097         * @param params the new parameter values
098         */
099        void onApplySquareControl(Scope scope, SquareControlParams params);
100    }
101
102    private final SquareControlListener squareControlListener;
103
104    private final JSlider       densityRatioSlider;
105    private final JSlider       variabilitySlider;
106    private final JSlider       rSquaredSlider;
107    private final JLabel        densityRatioValue;
108    private final JLabel        variabilityValue;
109    private final JLabel        rSquaredValue;
110    private final JRadioButton  neighbourFree;
111    private final JRadioButton  neighbourRelaxed;
112    private final JRadioButton  neighbourStrict;
113    private final double        origDensityRatio;
114    private final double        origVariability;
115    private final double        origRSquared;
116    private final String        origNeighbourMode;
117    /**
118     * Constructs a dialog that enables interactive adjustment of square grid
119     * parameters such as Density Ratio, Variability, and R² thresholds.
120     *
121     * @param owner      the parent frame that owns this dialog
122     * @param gridPanel  the grid panel to update visually during preview
123     * @param squareControlListener   squareControlListener receiving apply and preview callbacks
124     * @param initParams the initial square control parameters
125     */
126    public SquareControlDialog(JFrame                owner,
127                               SquareGridPanel       gridPanel,
128                               SquareControlListener squareControlListener,
129                               SquareControlParams   initParams) {
130        super(owner, "Square Controls", false);
131        this.squareControlListener = squareControlListener;
132
133        setLayout(new BorderLayout(10, 10));
134
135        JPanel content = new JPanel();
136        content.setLayout(new BoxLayout(content, BoxLayout.Y_AXIS));
137
138        // ─────────────────────────────────────────────────────────────────────
139        // Sliders
140        // ─────────────────────────────────────────────────────────────────────
141
142        densityRatioSlider = createSteppedSlider(
143                0,               // min
144                2000,            // max
145                2,               // step
146                initParams.minRequiredDensityRatio
147        );
148        variabilitySlider = createSteppedSlider(
149                0,
150                20,
151                1,
152                initParams.maxAllowableVariability
153        );
154
155        rSquaredSlider = createSteppedSlider(
156                0.0,
157                1.0,
158                0.05,
159                initParams.minRequiredRSquared
160        );
161        densityRatioValue = new JLabel();
162        variabilityValue  = new JLabel();
163        rSquaredValue     = new JLabel();
164
165        JPanel slidersPanel = new JPanel(new GridLayout(1, 3, 15, 0));
166        slidersPanel.setBorder(BorderFactory.createTitledBorder("Square Filters"));
167        slidersPanel.add(wrapSlider(densityRatioSlider, MIN_DENSITY_RATIO, densityRatioValue));
168        slidersPanel.add(wrapSlider(variabilitySlider,  MAX_VARIABILITY,   variabilityValue));
169        slidersPanel.add(wrapSlider(rSquaredSlider, "Min R²", rSquaredValue));
170
171        // ─────────────────────────────────────────────────────────────────────
172        // Neighbour mode radio buttons
173        // ─────────────────────────────────────────────────────────────────────
174
175        neighbourFree    = new JRadioButton("Free");
176        neighbourRelaxed = new JRadioButton("Relaxed");
177        neighbourStrict  = new JRadioButton("Strict");
178
179        ButtonGroup neighbourGroup = new ButtonGroup();
180        neighbourGroup.add(neighbourFree);
181        neighbourGroup.add(neighbourRelaxed);
182        neighbourGroup.add(neighbourStrict);
183
184        switch (initParams.neighbourMode) {
185            case "Relaxed":
186                neighbourRelaxed.setSelected(true);
187                break;
188            case "Strict":
189                neighbourStrict.setSelected(true);
190                break;
191            default:
192                neighbourFree.setSelected(true);
193        }
194
195        JPanel neighbourPanel = new JPanel(new FlowLayout(FlowLayout.LEFT));
196        neighbourPanel.setBorder(BorderFactory.createTitledBorder(NEIGHBOUR_MODE));
197        neighbourPanel.add(neighbourFree);
198        neighbourPanel.add(neighbourRelaxed);
199        neighbourPanel.add(neighbourStrict);
200        neighbourPanel.setAlignmentX(Component.CENTER_ALIGNMENT);
201
202        // ─────────────────────────────────────────────────────────────────────
203        // Apply / Cancel controls
204        // ─────────────────────────────────────────────────────────────────────
205
206        JPanel applyPanel       = new JPanel(new FlowLayout(FlowLayout.LEFT));
207        applyPanel.setBorder(BorderFactory.createTitledBorder("Apply Changes"));
208        JButton applyRecording  = new JButton("Apply to Recording");
209        JButton applyExperiment = new JButton("Apply to Experiment");
210        JButton applyProject    = new JButton("Apply to Project");
211        JButton cancelButton    = new JButton("Cancel");
212
213        for (JButton b : new JButton[]{applyRecording, applyExperiment, applyProject, cancelButton}) {
214            b.setPreferredSize(new Dimension(180, 28));
215        }
216
217        applyPanel.add(applyRecording);
218        applyPanel.add(applyExperiment);
219        applyPanel.add(applyProject);
220        applyPanel.add(cancelButton);
221        applyPanel.setAlignmentX(Component.CENTER_ALIGNMENT);
222
223        // ─────────────────────────────────────────────────────────────────────
224        // Layout assembly
225        // ─────────────────────────────────────────────────────────────────────
226        content.add(slidersPanel);
227        content.add(Box.createVerticalStrut(10));
228        content.add(neighbourPanel);
229        content.add(Box.createVerticalStrut(10));
230        content.add(applyPanel);
231        add(content, BorderLayout.CENTER);
232
233        // ─────────────────────────────────────────────────────────────────────
234        // Listeners for live preview
235        // ─────────────────────────────────────────────────────────────────────
236        ChangeListener sliderListener = new ChangeListener() {
237            @Override
238            public void stateChanged(ChangeEvent e) {
239                updateValueLabels();
240                propagatePreview();
241                SwingUtilities.invokeLater(gridPanel::repaint);
242            }
243        };
244        densityRatioSlider.addChangeListener(sliderListener);
245        variabilitySlider.addChangeListener(sliderListener);
246        rSquaredSlider.addChangeListener(sliderListener);
247
248        neighbourFree.addActionListener(   e -> propagatePreview());
249        neighbourRelaxed.addActionListener(e -> propagatePreview());
250        neighbourStrict.addActionListener( e -> propagatePreview());
251
252        // Apply button actions (commit and write to the file)
253        applyRecording.addActionListener(e -> {
254            squareControlListener.onApplySquareControl(SquareControlListener.Scope.RECORDING, collectParams());
255            dispose();
256        });
257        applyExperiment.addActionListener(e -> {
258            squareControlListener.onApplySquareControl(SquareControlListener.Scope.EXPERIMENT, collectParams());
259            dispose();
260        });
261        applyProject.addActionListener(e -> {
262            squareControlListener.onApplySquareControl(SquareControlListener.Scope.PROJECT, collectParams());
263            dispose();
264        });
265
266        cancelButton.addActionListener(e -> closeAction());
267
268        // ─────────────────────────────────────────────────────────────────────
269        // Preserve original values for cancel/restore
270        // ─────────────────────────────────────────────────────────────────────
271
272        origDensityRatio  = initParams.minRequiredDensityRatio;
273        origVariability   = initParams.maxAllowableVariability;
274        origRSquared      = initParams.minRequiredRSquared;
275        origNeighbourMode = initParams.neighbourMode;
276
277        updateValueLabels();
278        propagatePreview();
279
280        pack();
281        setLocationRelativeTo(owner);
282
283        // Make sure that pressing the red close button is doung the same as cancel
284        setDefaultCloseOperation(WindowConstants.DO_NOTHING_ON_CLOSE);
285        addWindowListener(new java.awt.event.WindowAdapter() {
286            @Override
287            public void windowClosing(java.awt.event.WindowEvent e) {
288                closeAction();
289            }
290        });
291    }

293    /**
294     * Wraps a slider and value label inside a titled panel.
295     */
296    private JPanel wrapSlider(JSlider slider, String title, JLabel valueLabel) {
297        JPanel panel = new JPanel(new BorderLayout());
298        panel.setBorder(BorderFactory.createTitledBorder(title));
299        panel.add(slider, BorderLayout.CENTER);
300        valueLabel.setHorizontalAlignment(SwingConstants.CENTER);
301        panel.add(valueLabel, BorderLayout.SOUTH);
302        return panel;
303    }
304
305    /**
306     * Updates the numeric labels to reflect current slider values.
307     */
308    private void updateValueLabels() {
309        densityRatioValue.setText(
310                formatSliderValue(getSliderDouble(densityRatioSlider, 0, 2), 2)
311        );
312        variabilityValue.setText(
313                formatSliderValue(getSliderDouble(variabilitySlider, 0, 1), 1)
314        );
315        rSquaredValue.setText(
316                formatSliderValue(getSliderDouble(rSquaredSlider, 0.0, 0.05), 0.05)
317        );
318    }
319
320    /**
321     * Sends live parameter updates for real-time preview without persistence.
322     */
323    private void propagatePreview() {
324        squareControlListener.onApplySquareControl(SquareControlListener.Scope.PREVIEW, collectParams());
325    }
326
327    /**
328     * Returns the currently selected neighbour mode string.
329     */
330    private String getNeighbourMode() {
331        if (neighbourFree.isSelected()) {
332            return "Free";
333        }
334        if (neighbourRelaxed.isSelected()) {
335            return "Relaxed";
336        }
337        return "Strict";
338    }
339
340    /**
341     * Restores original slider and neighbour mode values, updating the preview.
342     */
343    private void restoreOriginals() {
344        densityRatioSlider.setValue((int) Math.round(origDensityRatio / 2.0));
345        variabilitySlider.setValue((int) Math.round(origVariability / 1.0));
346        rSquaredSlider.setValue((int) Math.round(origRSquared / 0.05));
347
348        switch (origNeighbourMode) {
349            case "Relaxed":
350                neighbourRelaxed.setSelected(true);
351                break;
352            case "Strict":
353                neighbourStrict.setSelected(true);
354                break;
355            default:
356                neighbourFree.setSelected(true);
357        }
358
359        updateValueLabels();
360        propagatePreview();
361    }
362
363    /**
364     * Collects the current slider and neighbour mode state into a parameter object.
365     */
366    private SquareControlParams collectParams() {
367        return new SquareControlParams(
368                getSliderDouble(densityRatioSlider, 0, 2),
369                getSliderDouble(variabilitySlider, 0, 1),
370                getSliderDouble(rSquaredSlider, 0.0, 0.05),
371                getNeighbourMode()
372        );
373    }

375    private JSlider createSteppedSlider(double min, double max, double step, double initialValue) {
376
377        int steps = (int) Math.round((max - min) / step);
378        int initial = (int) Math.round((initialValue - min) / step);
379
380        JSlider slider = new JSlider(JSlider.VERTICAL, 0, steps,
381                                     Math.max(0, Math.min(steps, initial)));
382
383        // We want major ticks ONLY
384        slider.setPaintTicks(true);
385        slider.setSnapToTicks(false);  // IMPORTANT: manual snapping below
386        slider.setMinorTickSpacing(0); // no minor ticks
387
388        // Major tick spacing (labelled)
389        int major = Math.max(1, steps / 5);
390        slider.setMajorTickSpacing(major);
391
392        // Label table for major ticks only
393        java.util.Hashtable<Integer, JLabel> table = new java.util.Hashtable<>();
394        for (int i = 0; i <= steps; i += major) {
395            double v = min + (i * step);
396            table.put(i, new JLabel(formatSliderValue(v, step)));
397        }
398        slider.setLabelTable(table);
399        slider.setPaintLabels(true);
400
401        return slider;
402    }
403
404    private String formatSliderValue(double v, double step) {
405
406        // Integer step → format as integer
407        if (step >= 1.0) {
408            return Integer.toString((int) Math.round(v));
409        }
410
411        // Step of 0.1 → one decimal
412        if (Math.abs(step - 0.1) < 1e-9) {
413            return String.format("%.1f", v);
414        }
415
416        // Step of 0.05 → two decimals
417        if (Math.abs(step - 0.05) < 1e-9) {
418            return String.format("%.2f", v);
419        }
420
421        // Default fallback
422        return String.valueOf(v);
423    }
424
425    private double getSliderDouble(JSlider slider, double min, double step) {
426        return min + slider.getValue() * step;
427    }
428
429    private void closeAction() {
430        restoreOriginals();
431        dispose();
432    }
