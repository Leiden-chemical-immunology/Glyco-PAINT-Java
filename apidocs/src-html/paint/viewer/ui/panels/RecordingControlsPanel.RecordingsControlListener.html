002 *  Class:        RecordingControlsPanel.java
003 *  Package:      paint.viewer.ui.panels
004 *
005 *  PURPOSE:
006 *    Provides a control panel for managing recording-related actions and
007 *    grid display options in the PAINT viewer. Supports filtering, square
008 *    selection, cell assignment, playback, data export, and visualization
009 *    toggling.
010 *
011 *  DESCRIPTION:
012 *    The panel organizes a structured set of user controls into three groups:
013 *      1) Recording and grid action buttons (Filter, Select, Assign, Play,
014 *         Export, Show Squares)
015 *      2) Visualization toggles for borders and shading
016 *      3) Number display mode options (None, Label, Square)
017 *
018 *    A {@link RecordingsControlListener} interface defines callbacks for all interactive
019 *    controls, allowing the parent viewer frame to handle the actions.
020 *    Layout is vertical for readability, with consistent sizing and spacing
021 *    across buttons and toggle groups.
022 *
023 *  KEY FEATURES:
024 *    • Centralized control panel for recording actions and display settings.
025 *    • Live toggles for borders, shading, and number-display modes.
026 *    • Listener interface for flexible integration with viewer logic.

041package paint.viewer.ui.panels;
042
043import javax.swing.*;
044import java.awt.*;
045
046/**
047 * A user interface panel containing interactive controls for managing recordings
048 * and grid visualization settings within the PAINT viewer.
049 */
050public class RecordingControlsPanel {
051
052    /**
053     * Defines callback methods for handling user actions triggered
054     * by buttons or toggle components on this panel.
055     */
056    public interface RecordingsControlListener {
057        void onFilterRequested();                                             // Called when the user requests to filter the recording list
058        void onSelectSquaresRequested();                                      // Called when the user opens the square-selection dialog
059        void onAssignCellsRequested();                                        // Called when the user opens the cell-assignment dialog
060        void onPlayRecordingRequested();                                      // Called when the user requests to play the current recording
061        void onExportLeftImageRequested();                                    // Called when the user requests to export the left grid image
062        void onShowSquaresRequested();                                        // Called when the user requests to view the squares CSV
063        void onBordersToggled(boolean showBorders);                           // Triggered when borders are toggled on or off
064        void onShadingToggled(boolean showShading);                           // Triggered when shading is toggled on or off
065        void onNumberModeChanged(SquareGridPanel.NumberMode mode);            // Called when the numeric display mode (None, Label, Square) is changed
066        void onExcludeToggleRequested();                                      // Called when the Exclude/Include button is pressed
067    }
068
069    private final JPanel  root;
070    private final JButton filterRecordingsButton;
071    private final JButton selectSquaresButton;
072    private final JButton assignCellsButton;
073    private final JButton playRecordingButton;
074    private final JButton exportImageButton;
075    private final JButton showSquaresButton;
076    private final JButton excludeButton;
077
078    /**
079     * Constructs a {@code RecordingControlsPanel} containing action buttons,
080     * visualization toggles, and numbering mode options. All actions are
081     * delegated to the provided {@link RecordingsControlListener}.
082     *
083     * @param recordingsControlListener the recordingsControlListener that receives callbacks when user actions occur
084     */
085    public RecordingControlsPanel(final RecordingsControlListener recordingsControlListener) {
086
087        root = new JPanel(new BorderLayout());
088        root.setBorder(BorderFactory.createCompoundBorder(
089                BorderFactory.createLineBorder(Color.DARK_GRAY, 2),
090                BorderFactory.createEmptyBorder(10, 10, 10, 10)
091        ));
092        root.setPreferredSize(new Dimension(240, 0));
093
094        JPanel content = new JPanel();
095        content.setLayout(new BoxLayout(content, BoxLayout.Y_AXIS));
096
097        // === Action buttons ===
098        filterRecordingsButton = new JButton("Filter Recordings");
099        selectSquaresButton    = new JButton("Select Squares");
100        assignCellsButton      = new JButton("Assign Cells");
101        playRecordingButton    = new JButton("Play Recording");
102        exportImageButton      = new JButton("Export Image");
103        showSquaresButton      = new JButton("Show Squares Data");
104        excludeButton          = new JButton("Exclude Recording");
105
106        for (JButton button : new JButton[]{
107                filterRecordingsButton,
108                selectSquaresButton,
109                assignCellsButton,
110                playRecordingButton,
111                exportImageButton,
112                showSquaresButton,
113                excludeButton
114        }) {
115            button.setAlignmentX(Component.CENTER_ALIGNMENT);
116            button.setMaximumSize(new Dimension(Integer.MAX_VALUE, 28));
117        }
118
119        JPanel controls = new JPanel();
120        controls.setLayout(new BoxLayout(controls, BoxLayout.Y_AXIS));
121        controls.setBorder(BorderFactory.createTitledBorder("Controls"));
122        controls.setAlignmentX(Component.LEFT_ALIGNMENT);
123        controls.setMaximumSize(new Dimension(Integer.MAX_VALUE, controls.getPreferredSize().height));
124
125        controls.add(filterRecordingsButton);
126        controls.add(Box.createVerticalStrut(10));
127        controls.add(selectSquaresButton);
128        controls.add(Box.createVerticalStrut(10));
129        controls.add(assignCellsButton);
130        controls.add(Box.createVerticalStrut(10));
131        controls.add(playRecordingButton);
132        controls.add(Box.createVerticalStrut(10));
133        controls.add(exportImageButton);
134        controls.add(Box.createVerticalStrut(10));
135        controls.add(showSquaresButton);
136        controls.add(Box.createVerticalStrut(10));
137        controls.add(excludeButton);
138
139        // === Borders and shading toggles ===
140        JCheckBox showBorders = new JCheckBox("Show borders", true);
141        JCheckBox showShading = new JCheckBox("Show shading", true);
142
143        JPanel borders = new JPanel();
144        borders.setLayout(new BoxLayout(borders, BoxLayout.Y_AXIS));
145        borders.setBorder(BorderFactory.createTitledBorder("Borders and Shading"));
146        borders.setAlignmentX(Component.LEFT_ALIGNMENT);
147        borders.setMaximumSize(new Dimension(Integer.MAX_VALUE, borders.getPreferredSize().height));
148        borders.add(showBorders);
149        borders.add(Box.createVerticalStrut(5));
150        borders.add(showShading);
151
152        // === Number display options ===
153        JRadioButton none   = new JRadioButton("None", true);
154        JRadioButton label  = new JRadioButton("Label");
155        JRadioButton square = new JRadioButton("Square");
156
157        ButtonGroup buttonGroup = new ButtonGroup();
158        buttonGroup.add(none);
159        buttonGroup.add(label);
160        buttonGroup.add(square);
161
162        JPanel numbers = new JPanel();
163        numbers.setLayout(new BoxLayout(numbers, BoxLayout.Y_AXIS));
164        numbers.setBorder(BorderFactory.createTitledBorder("Numbers"));
165        numbers.setAlignmentX(Component.LEFT_ALIGNMENT);
166        numbers.setMaximumSize(new Dimension(Integer.MAX_VALUE, numbers.getPreferredSize().height));
167        numbers.add(none);
168        numbers.add(Box.createVerticalStrut(5));
169        numbers.add(label);
170        numbers.add(Box.createVerticalStrut(5));
171        numbers.add(square);
172
173        // === Action listeners ===
174        filterRecordingsButton.addActionListener(e -> recordingsControlListener.onFilterRequested());
175        selectSquaresButton.addActionListener(   e -> recordingsControlListener.onSelectSquaresRequested());
176        assignCellsButton.addActionListener(     e -> recordingsControlListener.onAssignCellsRequested());
177        playRecordingButton.addActionListener(   e -> recordingsControlListener.onPlayRecordingRequested());
178        exportImageButton.addActionListener(     e -> recordingsControlListener.onExportLeftImageRequested());
179        showSquaresButton.addActionListener(     e -> recordingsControlListener.onShowSquaresRequested());
180        excludeButton.addActionListener(         e -> recordingsControlListener.onExcludeToggleRequested());
181
182        showBorders.addActionListener(           e -> recordingsControlListener.onBordersToggled(showBorders.isSelected()));
183        showShading.addActionListener(           e -> recordingsControlListener.onShadingToggled(showShading.isSelected()));
184        none.addActionListener(                  e -> recordingsControlListener.onNumberModeChanged(SquareGridPanel.NumberMode.NONE));
185        label.addActionListener(                 e -> recordingsControlListener.onNumberModeChanged(SquareGridPanel.NumberMode.LABEL));
186        square.addActionListener(                e -> recordingsControlListener.onNumberModeChanged(SquareGridPanel.NumberMode.SQUARE));
187
188        // === Layout order ===
189        content.add(controls);
190        content.add(Box.createVerticalStrut(10));
191        content.add(borders);
192        content.add(Box.createVerticalStrut(10));
193        content.add(numbers);
194        content.add(Box.createVerticalGlue());
195
196        root.add(content, BorderLayout.NORTH);
197    }
198
199    /**
200     * Returns the root Swing component representing this panel.
201     *
202     * @return the root {@link JComponent}
203     */
204    public JComponent getComponent() {
205        return root;
206    }
207
208    /**
209     * Enables or disables all action buttons on the panel.
210     * <p>
211     * This is used to disable user interaction while modal dialogs
212     * are open or while long-running operations are in progress.
213     *
214     * @param enabled {@code true} to enable all action buttons,
215     *                {@code false} to disable them
216     */
217    public void setButtonsEnabled(boolean enabled) {
218        filterRecordingsButton.setEnabled(enabled);
219        selectSquaresButton.setEnabled(enabled);
220        assignCellsButton.setEnabled(enabled);
221        playRecordingButton.setEnabled(enabled);
222        exportImageButton.setEnabled(enabled);
223        showSquaresButton.setEnabled(enabled);
224        excludeButton.setEnabled(enabled);
225    }
226
227    /**
228     * Updates the text and appearance of the exclusion toggle button
229     * based on the current recording's exclusion status.
230     *
231     * @param excluded {@code true} if the recording is currently excluded,
232     *                 {@code false} otherwise
233     */
234    public void setExcludeButtonText(boolean excluded) {
235        if (excludeButton == null) {
236            return;
237        }
238
239        excludeButton.setText(excluded ? "Include Recording" : "Exclude Recording");
240        Color defaultFg = UIManager.getColor("Button.foreground");
241        if (defaultFg == null) {
242            defaultFg = Color.BLACK;
243        }
244
245        excludeButton.setForeground(excluded ? Color.RED : defaultFg);
246    }
247}
