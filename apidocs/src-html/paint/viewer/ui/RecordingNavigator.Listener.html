031package paint.viewer.ui;
032
033import paint.viewer.model.RecordingEntry;
034
035import java.util.List;
036
037/**
038 * Handles safe navigation through a list of recordings.
039 * {@link paint.viewer.ui.frames.ViewerFrame} registers as the listener and
040 * updates the UI whenever a new index is selected.
041 */
042public final class RecordingNavigator {
043
044    /**
045     * Listener interface for responding to navigation events.
046     */
047    public interface Listener {
048        /**
049         * Called when a navigation action requires shifting to a new recording.
050         *
051         * @param newIndex the target index in the recording list
052         */
053        void onNavigateTo(int newIndex);
054    }
055
056    private final Listener listener;
057
058    /**
059     * Constructs a {@code RecordingNavigator} with the specified listener.
060     *
061     * @param listener the object to notify when navigation occurs
062     */
063    public RecordingNavigator(Listener listener) {
064        this.listener = listener;
065    }
066
067    /**
068     * Navigates to the first recording in the list.
069     *
070     * @param entries the list of available recordings
071     */
072    public void first(List<RecordingEntry> entries) {
073        if (entries.isEmpty()) return;
074        listener.onNavigateTo(0);
075    }
076
077    /**
078     * Navigates to the last recording in the list.
079     *
080     * @param entries the list of available recordings
081     */
082    public void last(List<RecordingEntry> entries) {
083        if (entries.isEmpty()) return;
084        listener.onNavigateTo(entries.size() - 1);
085    }
086
087    /**
088     * Navigates to the next recording in the list.
089     *
090     * @param entries      the list of available recordings
091     * @param currentIndex the index of the currently displayed recording
092     */
093    public void next(List<RecordingEntry> entries, int currentIndex) {
094        if (entries.isEmpty()) return;
095        int next = Math.min(entries.size() - 1, currentIndex + 1);
096        listener.onNavigateTo(next);
097    }
098
099    /**
100     * Navigates to the previous recording in the list.
101     *
102     * @param entries      the list of available recordings
103     * @param currentIndex the index of the currently displayed recording
104     */
105    public void prev(List<RecordingEntry> entries, int currentIndex) {
106        if (entries.isEmpty()) return;
107        int prev = Math.max(0, currentIndex - 1);
108        listener.onNavigateTo(prev);
109    }
110}
