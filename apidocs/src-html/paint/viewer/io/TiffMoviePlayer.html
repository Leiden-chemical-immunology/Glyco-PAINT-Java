014 *      • Frame navigation via slider
015 *      • Playback speed adjustment (0.5× – 4×)

019 *    background thread. All UI rendering occurs on the Swing EDT as required.
020 *
021 *  KEY FEATURES:
022 *    • Plays TIFF stacks as movies using ImageJ processors.
023 *    • Uses calibration metadata (frame interval) when available.
024 *    • Thread-safe interplay between worker threads and Swing.
025 *    • Graceful cleanup when the movie window is closed.
026 *    • Suppresses ImageJ console output for clean UI integration.

059/**
060 * A lightweight GUI-based TIFF stack movie player built on ImageJ and Swing.
061 * <p>
062 * Loads a TIFF file, extracts frame timing, and provides a movie player with:
063 * play/pause, playback speed control, and frame navigation. Rendering is
064 * performed on Swing components, while TIFF handling is delegated to ImageJ.
065 * </p>
066 */
067public class TiffMoviePlayer {
068
069    /**
070     * Loads and plays a multi-frame TIFF stack in a dedicated Swing window.
071     *
072     * @param tiffPath absolute or relative path to the TIFF file
073     */
074    public void playMovie(String tiffPath) {
075
076        final String fileName = new File(tiffPath).getName();
077
078        // ---------------------------------------------------------------------
079        // LOADING DIALOG (lightweight splash while ImageJ loads the TIFF)
080        // ---------------------------------------------------------------------
081        final JDialog loadingDialog = new JDialog((Frame) null, "Loading Recording", false);
082        loadingDialog.setDefaultCloseOperation(JDialog.DO_NOTHING_ON_CLOSE);
083        loadingDialog.setLayout(new BorderLayout());
084
085        JLabel label = new JLabel("Loading " + fileName + "…", SwingConstants.CENTER);
086        label.setFont(new Font("SansSerif", Font.PLAIN, 13));
087        label.setBorder(new EmptyBorder(25, 20, 25, 20));
088        loadingDialog.add(label, BorderLayout.CENTER);
089
090        loadingDialog.setSize(300, 120);
091        loadingDialog.setResizable(false);
092        loadingDialog.setLocationRelativeTo(null);
093
094        SwingUtilities.invokeLater(() -> loadingDialog.setVisible(true));
095
096        // ==========================================================================
097        // BACKGROUND THREAD: load TIFF, apply contrast enhancement, open UI window
098        // ==========================================================================
099        new Thread(() -> {
100
101            System.setProperty("apple.awt.UIElement", "true");
102            IJ.redirectErrorMessages();
103            IJ.showStatus("");
104
105            // Silence ImageJ's console
106            PrintStream originalOut = System.out;
107            System.setOut(new PrintStream(new OutputStream() {
108                @Override
109                public void write(int b) { /* ignore console output */ }
110            }));
111
112            ImagePlus imp = IJ.openImage(tiffPath);
113
114            if (imp == null) {
115                try {
116                    ImporterOptions opts = new ImporterOptions();
117                    opts.setId(tiffPath);
118                    opts.setAutoscale(true);
119                    opts.setStackOrder(ImporterOptions.ORDER_XYCZT);
120
121                    ImagePlus[] imps = BF.openImagePlus(opts);
122                    if (imps != null && imps.length > 0) {
123                        imp = imps[0];
124                    }
125                } catch (Exception bfErr) {
126                    bfErr.printStackTrace();
127                }
128            }
129            // Restore normal stdout
130            System.setOut(originalOut);
131
132            // Close loading dialog
133            SwingUtilities.invokeLater(loadingDialog::dispose);
134
135            if (imp == null) {
136                SwingUtilities.invokeLater(() -> JOptionPane.showMessageDialog(
137                        null,
138                        "Failed to open image file:\n" + tiffPath,
139                        "Error",
140                        JOptionPane.ERROR_MESSAGE
141                ));
142                return;
143            }
144
145            // Optional contrast enhancement
146            IJ.run(imp, "Enhance Contrast", "saturated=0.35");
147
148            // Determine playback speed using calibration metadata
149            int delay = 50; // fallback
150            Calibration cal = imp.getCalibration();
151            if (cal != null && cal.frameInterval > 0) {
152                delay = (int) Math.round(cal.frameInterval * 1000);
153            }
154            final int baseDelayMs = delay;
155
156            // ------------------------------------------------------------
157            // UI CONSTRUCTION — must occur on the EDT
158            // ------------------------------------------------------------
159
160            final ImagePlus impFinal       = imp;
161
162            SwingUtilities.invokeLater(() -> buildAndRunMovieWindow(impFinal, fileName, baseDelayMs));
163
164        }, "TiffLoaderThread").start();
165    }
166
167    // =============================================================================================
168    // INTERNAL: Build movie UI window + player loop
169    // =============================================================================================
170    private void buildAndRunMovieWindow(ImagePlus imp, String fileName, int baseDelayMs) {
171
172        final JFrame frame = new JFrame("Movie Player - " + fileName);
173        frame.setDefaultCloseOperation(JFrame.DISPOSE_ON_CLOSE);
174        frame.setLayout(new BorderLayout());
175
176        // ----- Image display area -----
177        final JLabel imageLabel = new JLabel("", SwingConstants.CENTER);
178        imageLabel.setOpaque(true);
179        imageLabel.setBackground(Color.DARK_GRAY);
180
181        JPanel imagePanel = new JPanel(new BorderLayout());
182        imagePanel.setBackground(Color.DARK_GRAY);
183        imagePanel.setBorder(new EmptyBorder(4, 4, 4, 4));
184        imagePanel.add(imageLabel, BorderLayout.CENTER);
185        frame.add(imagePanel, BorderLayout.CENTER);
186
187        // ----- Movie metadata -----
188        final int totalFrames         = imp.getStackSize();
189        final JSlider frameSlider     = new JSlider(1, totalFrames, 1);
190        final JLabel frameLabel       = new JLabel("Frame: 1");
191        final JButton playPauseButton = new JButton("⏸ Pause");
192        final JSlider speedSlider     = new JSlider(50, 400, 100); // 0.5× – 4.0×
193        final JLabel speedLabel       = new JLabel("Speed: 1.0×");
194
195        // ----- Layout for controls -----
196        JPanel controls = new JPanel(new GridLayout(2, 1, 0, 3));
197        controls.setBorder(new EmptyBorder(6, 6, 6, 6));
198        controls.setBackground(new Color(245, 245, 245));
199
200        JPanel topRow = new JPanel(new FlowLayout(FlowLayout.CENTER, 8, 0));
201        topRow.setBackground(controls.getBackground());
202        frameSlider.setPreferredSize(new Dimension(260, 25));
203        topRow.add(playPauseButton);
204        topRow.add(frameSlider);
205        topRow.add(frameLabel);
206
207        JPanel bottomRow = new JPanel(new FlowLayout(FlowLayout.CENTER, 8, 0));
208        bottomRow.setBackground(controls.getBackground());
209        speedSlider.setPreferredSize(new Dimension(100, 25));
210        speedSlider.setMajorTickSpacing(50);
211        speedSlider.setPaintTicks(true);
212        bottomRow.add(speedLabel);
213        bottomRow.add(speedSlider);
214
215        controls.add(topRow);
216        controls.add(bottomRow);
217
218        // ----- Close button -----
219        JButton closeButton = new JButton("Close");
220        closeButton.addActionListener(e -> frame.dispose());
221
222        JPanel closePanel = new JPanel(new FlowLayout(FlowLayout.RIGHT, 8, 4));
223        closePanel.setBackground(controls.getBackground());
224        closePanel.add(closeButton);
225
226        JPanel bottomPanel = new JPanel(new BorderLayout());
227        bottomPanel.add(controls, BorderLayout.CENTER);
228        bottomPanel.add(closePanel, BorderLayout.SOUTH);
229
230        frame.add(bottomPanel, BorderLayout.SOUTH);
231
232        // ----- Initial layout -----
233        frame.setSize(
234                Math.min(imp.getWidth() + 40, 1000),
235                Math.min(imp.getHeight() + 180, 900)
236        );
237        frame.setLocationRelativeTo(null);
238        frame.setVisible(true);
239
240        // =======================================================================================
241        // Player state + listeners
242        // =======================================================================================
243        final boolean[] playing = {true};
244        final int[] currentFrame = {1};
245
246        // Toggle play/pause
247        playPauseButton.addActionListener(e -> {
248            playing[0] = !playing[0];
249            playPauseButton.setText(playing[0] ? "⏸ Pause" : "▶️ Play");
250        });
251
252        // Playback speed label
253        speedSlider.addChangeListener(e -> {
254            double raw = speedSlider.getValue() / 100.0;
255            double speed = Math.round(raw * 2) / 2.0;  // step of 0.5×
256            speedLabel.setText(String.format("Speed: %.1fx", speed));
257        });
258
259        // Manual frame navigation (only when paused)
260        frameSlider.addChangeListener(e -> {
261            if (!frameSlider.getValueIsAdjusting() && !playing[0]) {
262                final int frameIndex = frameSlider.getValue();
263                ImageProcessor ip2 = imp.getStack().getProcessor(frameIndex);
264                final BufferedImage img2 = ip2.getBufferedImage();
265                SwingUtilities.invokeLater(() -> {
266                    imageLabel.setIcon(new ImageIcon(img2));
267                    frameLabel.setText("Frame: " + frameIndex);
268                });
269                currentFrame[0] = frameIndex;
270            }
271        });
272
273        // Show frame 1 immediately
274        ImageProcessor ip0 = imp.getStack().getProcessor(1);
275        imageLabel.setIcon(new ImageIcon(ip0.getBufferedImage()));
276
277        // =======================================================================================
278        // MOVIE PLAYBACK THREAD
279        // =======================================================================================
280        new Thread(() -> {
281            while (frame.isVisible()) {
282
283                if (playing[0]) {
284                    int frameIdx = currentFrame[0];
285                    ImageProcessor ip = imp.getStack().getProcessor(frameIdx);
286                    final BufferedImage img = ip.getBufferedImage();
287
288                    SwingUtilities.invokeLater(() -> {
289                        imageLabel.setIcon(new ImageIcon(img));
290                        frameLabel.setText("Frame: " + frameIdx);
291                        frameSlider.setValue(frameIdx);
292                    });
293
294                    // Speed multiplier
295                    double raw = speedSlider.getValue() / 100.0;
296                    double speed = Math.round(raw * 2) / 2.0;
297                    if (speed <= 0.0) {
298                        speed = 0.5;
299                    }
300
301                    long sleepTime = (long) (baseDelayMs / speed);
302                    if (sleepTime < 5) {
303                        sleepTime = 5;
304                    }
305
306                    try {
307                        Thread.sleep(sleepTime);
308                    } catch (InterruptedException ignored) { }
309
310                    // Next frame
311                    currentFrame[0]++;
312                    if (currentFrame[0] > totalFrames) {
313                        currentFrame[0] = 1;
314                    }
315
316                } else {
317                    // Paused: reduce CPU usage
318                    try { Thread.sleep(100); }
319                    catch (InterruptedException ignored) { }
320                }
321            }
322        }, "TiffMoviePlaybackThread").start();
323    }
