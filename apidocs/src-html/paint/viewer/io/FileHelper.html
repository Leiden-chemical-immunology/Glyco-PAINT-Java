038package paint.viewer.io;
039
040import java.awt.*;
041import java.io.BufferedReader;
042import java.io.BufferedWriter;
043import java.io.IOException;
044import java.nio.file.Files;
045import java.nio.file.Path;
046
047/**
048 * Static utility class providing file operations for the PAINT Viewer, including:
049 * <ul>
050 *     <li>Exporting Swing components as high-resolution PNG images</li>
051 *     <li>Filtering a project's {@code squares.csv} to a temporary CSV</li>
052 *     <li>Automatically opening filtered results via Desktop integration</li>
053 * </ul>
054 *
055 * <p>Instantiation is prevented via a private constructor.</p>
056 */
057public final class FileHelper {
058
059    /** Prevent instantiation. */
060    private FileHelper() {
061    }
062
063
064    // =========================================================================
065    // CSV FILTERING
066    // =========================================================================
067
068    /**
069     * Extracts only the rows from an experiment's {@code squares.csv} matching the
070     * given recording name, writes them to a temporary CSV file, marks the file
071     * read-only, and opens it using the desktop's default CSV viewer.
072     *
073     * <p>If no rows match, an empty CSV (with header) is still produced.</p>
074     *
075     * @param projectRoot   the root directory of the PAINT project
076     * @param experimentName the experiment folder containing {@code squares.csv}
077     * @param recordingName  the recording name to filter for
078     * @throws IOException if reading or writing fails, or Desktop integration is unavailable
079     */
080    public static void filterAndOpenSquaresCsv(Path projectRoot,
081            String experimentName,
082            String recordingName)
083            throws IOException {
084
085        Path origCsv = projectRoot
086                .resolve(experimentName)
087                .resolve("squares.csv");
088
089        if (!Files.exists(origCsv)) {
090            throw new IOException("Squares.csv not found: " + origCsv);
091        }
092
093        // Create temporary filtered file
094        Path tempFile = Files.createTempFile("Squares " + recordingName, ".csv");
095
096        try (BufferedReader r = Files.newBufferedReader(origCsv);
097             BufferedWriter w = Files.newBufferedWriter(tempFile)) {
098
099            String header = r.readLine();
100            if (header != null) {
101                w.write(header);
102                w.newLine();
103
104                String line;
105                while ((line = r.readLine()) != null) {
106                    if (line.contains(recordingName)) {
107                        w.write(line);
108                        w.newLine();
109                    }
110                }
111            }
112        }
113
114        tempFile.toFile().setReadOnly();
115
116        if (Desktop.isDesktopSupported()) {
117            Desktop.getDesktop().open(tempFile.toFile());
118        } else {
119            throw new IOException("Desktop integration not supported.");
120        }
121    }
122}
