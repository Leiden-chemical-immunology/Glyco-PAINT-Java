037package paint.viewer.io;
038
039import javax.swing.*;
040import java.awt.*;
041import java.awt.image.BufferedImage;
042import java.nio.file.Files;
043import java.nio.file.Path;
044
045/**
046 * Static utility for exporting Swing panels as high-resolution PNG images.
047 *
048 * <p>This class cannot be instantiated.</p>
049 */
050public final class PanelExporter {
051
052    /** Prevent instantiation. */
053    private PanelExporter() { }
054
055    /**
056     * Exports the given {@link JPanel} to a PNG image at a specified scaling factor.
057     * All visual overlays painted on the panel are included.
058     *
059     * <p><strong>Example:</strong><br>
060     * {@code export(leftGridPanel, Paths.get("output.png"), 2.0);} produces a
061     * double-resolution export suitable for publication-quality figures.</p>
062     *
063     * @param panel  the panel to render
064     * @param output the destination PNG file
065     * @param scale  scale factor (1.0 = current on-screen size)
066     *
067     * @throws Exception if rendering or writing fails
068     */
069    public static void export(JPanel panel, Path output, double scale) throws Exception {
070
071        // Compute scaled output dimensions
072        int width  = (int) (panel.getWidth()  * scale);
073        int height = (int) (panel.getHeight() * scale);
074
075        // Create image buffer
076        BufferedImage img = new BufferedImage(width, height, BufferedImage.TYPE_INT_RGB);
077
078        Graphics2D g2 = img.createGraphics();
079
080        // High-quality rendering hints
081        g2.setRenderingHint(RenderingHints.KEY_ANTIALIASING,
082                            RenderingHints.VALUE_ANTIALIAS_ON);
083        g2.setRenderingHint(RenderingHints.KEY_RENDERING,
084                            RenderingHints.VALUE_RENDER_QUALITY);
085        g2.setRenderingHint(RenderingHints.KEY_INTERPOLATION,
086                            RenderingHints.VALUE_INTERPOLATION_BICUBIC);
087
088        // Apply scaling and render the panel
089        g2.scale(scale, scale);
090        panel.paintAll(g2);
091        g2.dispose();
092
093        // Ensure directory exists
094        if (output.getParent() != null) {
095            Files.createDirectories(output.getParent());
096        }
097
098        // Write PNG
099        javax.imageio.ImageIO.write(img, "png", output.toFile());
100    }
101}
