042package paint.viewer.app;
043
044import paint.shared.config.paintconfig.PaintConfig;
045import paint.shared.dialogs.ProjectDialog;
046import paint.shared.dialogs.ProjectDialog.DialogMode;
047import paint.shared.utils.PaintLogger;
048import paint.shared.utils.PaintPrefs;
049import paint.viewer.model.RecordingEntry;
050import paint.viewer.model.RecordingLoader;
051import paint.viewer.ui.frames.ViewerFrame;
052
053import javax.swing.*;
054import java.nio.file.Path;
055import java.nio.file.Paths;
056import java.util.List;
057
058/**
059 * Main entry point for the PAINT Viewer application.
060 * <p>
061 * This class sets up preferences, configuration, and logging, and then opens
062 * the project selection dialog. Once the user confirms their selection, it
063 * loads the corresponding recordings and launches the main viewer interface.
064 * </p>
065 */
066public class Viewer {
067
068    /**
069     * Application entry point.
070     *
071     * @param args command-line arguments (unused)
072     */
073    public static void main(String[] args) {
074        SwingUtilities.invokeLater(() -> {
075
076            // --- Step 1: Load last-used project from preferences ---
077            String lastProject = PaintPrefs.getString("Path", "Project Root", System.getProperty("user.home"));
078            Path   projectPath = Paths.get(lastProject);
079
080            // --- Step 2: Initialise logging and configuration ---
081            PaintConfig.initialise(projectPath);
082            PaintLogger.initialise(projectPath, "Viewer");
083            PaintLogger.setLevel(PaintPrefs.getString("Runtime", "Log Level", "INFO"));
084
085            // --- Step 3: Show the project specification dialog ---
086            ProjectDialog specificationDialog =
087                    new ProjectDialog(null, projectPath, DialogMode.VIEWER);
088
089            // Callback when user presses OK
090            specificationDialog.setCalculationCallback(project -> {
091                try {
092                    List<RecordingEntry> recordingEntries =
093                            RecordingLoader.loadFromProject(project);
094
095                    if (recordingEntries.isEmpty()) {
096                        JOptionPane.showMessageDialog(
097                                null,
098                                "No valid recordings found in selected experiments.",
099                                "No Recordings",
100                                JOptionPane.WARNING_MESSAGE
101                        );
102                        return false;
103                    }
104
105                    ViewerFrame viewer = new ViewerFrame(project, recordingEntries);
106                    viewer.setVisible(true);
107                    return true;
108
109                } catch (Exception ex) {
110                    PaintLogger.errorf("Viewer launch failed");
111                    return false;
112                }
113            });
114
115            // --- Step 4: Open dialog (blocking) ---
116            specificationDialog.showDialog();
117        });
118    }
119}
