012 *
013 *        <project>/Viewer/Square Override.csv

018 *

020 *
021 *  KEY FEATURES:
022 *    • Plain data container (POJO) for square overrides.

039package paint.viewer.override.square_override;
040
041/**
042 * Data container representing a single row from {@code Square Override.csv}.
043 * <p>
044 * Each instance identifies one square via:
045 * <ul>
046 *   <li>{@code experimentName}</li>
047 *   <li>{@code recordingName}</li>
048 *   <li>{@code squareNumber}</li>
049 * </ul>
050 * and provides the overridden {@code cellId} (plus an informational timestamp).
051 * <p>
052 * This class contains no business logic; it is intended for CSV parsing and
053 * in-memory override application.
054 */
055public class SquareOverride {
056
057    /** Experiment identifier used for matching. */
058    private String experimentName;
059
060    /** Recording identifier used for matching. */
061    private String recordingName;
062
063    /** Square number within the recording used for matching. */
064    private int    squareNumber;
065
066    /** Overridden cell ID to apply to the matched square. */
067    private int    cellId;
068
069    /** Timestamp captured when the override was written (informational). */
070    private String timestamp;
071
072    // ───────────────────────────────────────────────────────────────────────────────
073    // CONSTRUCTORS
074    // ───────────────────────────────────────────────────────────────────────────────
075
076    /** Creates an empty SquareOverride (typically populated via CSV parsing). */
077    public SquareOverride() {
078    }
079
080    // ───────────────────────────────────────────────────────────────────────────────
081    // ACCESSORS
082    // ───────────────────────────────────────────────────────────────────────────────
083
084    /** @return experiment name used for matching */
085    public String getExperimentName() {
086        return experimentName;
087    }
088
089    /**
090     * Sets the experiment name used for matching.
091     *
092     * @param experimentName experiment identifier
093     */
094    public void setExperimentName(String experimentName) {
095        this.experimentName = experimentName;
096    }
097
098    /** @return recording name used for matching */
099    public String getRecordingName() {
100        return recordingName;
101    }
102
103    /**
104     * Sets the recording name used for matching.
105     *
106     * @param recordingName recording identifier
107     */
108    public void setRecordingName(String recordingName) {
109        this.recordingName = recordingName;
110    }
111
112    /** @return square number used for matching */
113    public int getSquareNumber() {
114        return squareNumber;
115    }
116
117    /**
118     * Sets the square number used for matching.
119     *
120     * @param squareNumber square index/number within the recording
121     */
122    public void setSquareNumber(int squareNumber) {
123        this.squareNumber = squareNumber;
124    }
125
126    /** @return overridden cell ID */
127    public int getCellId() {
128        return cellId;
129    }
130
131    /**
132     * Sets the overridden cell ID to apply to the matched square.
133     *
134     * @param cellId new cell assignment
135     */
136    public void setCellId(int cellId) {
137        this.cellId = cellId;
138    }
139
140    /**
141     * Sets the informational timestamp associated with this override.
142     *
143     * @param timestamp timestamp string as written in the CSV
144     */
145    public void setTimestamp(String timestamp) {
146        this.timestamp = timestamp;
147    }
148
149    // ───────────────────────────────────────────────────────────────────────────────
150    // DEBUG
151    // ───────────────────────────────────────────────────────────────────────────────
152
153    /**
154     * Debug-friendly representation of this override row.
155     *
156     * @return string representation
157     */
158    @Override
159    public String toString() {
160        return "SquareOverride{" +
161                "experimentName='" + experimentName + '\'' +
162                ", recordingName='" + recordingName + '\'' +
163                ", squareNumber=" + squareNumber +
164                ", cellId=" + cellId +
165                ", timestamp='" + timestamp + '\'' +
166                '}';
167    }
168}
