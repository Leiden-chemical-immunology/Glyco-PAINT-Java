003 *  Package:      paint.viewer.override.square_override
004 *
005 *  PURPOSE:
006 *    Persists per-square cell assignment overrides (cellId) for the PAINT Viewer.
007 *    This allows user-selected square→cellId mappings to be saved to disk and
008 *    restored across Viewer sessions.
009 *
010 *  DESCRIPTION:
011 *    Maintains the CSV file:
012 *
013 *        <project>/Viewer/Square Override.csv
014 *
015 *    Each row stores one square-level override:
016 *
017 *        experimentName, recordingName, squareNumber, cellId, timestamp
018 *
019 *    The writer supports:
020 *      • Merging (add/update/remove) overrides for a recording.
021 *      • Replacing all overrides for a recording in one operation.
022 *      • Querying whether a recording already has overrides.
023 *      • Atomic writes via temporary file replacement.
024 *
025 *    Header validation is performed; malformed headers cause the writer to
026 *    rebuild the file contents from scratch (non-destructive to in-memory state).
027 *
028 *  KEY FEATURES:
029 *    • Per-square persistent cell assignments (squareNumber → cellId).
030 *    • Non-destructive merge semantics (remove when cellId == 0).
031 *    • Full replace mode for a recording’s overrides.
032 *    • Atomic on-disk writes to reduce risk of corruption.
033 *    • Automatic header validation and recovery.
034 *

048package paint.viewer.override.square_override;
049
050import paint.shared.utils.PaintLogger;
051import paint.viewer.model.RecordingEntry;
052
053import java.io.IOException;
054import java.nio.file.*;
055import java.time.LocalDateTime;
056import java.util.*;
057
058import static paint.shared.constants.PaintStringConstants.SQUARE_NUMBER;
059import static paint.shared.constants.PaintStringConstants.CELL_ID;
060import static paint.shared.constants.PaintStringConstants.TIME_STAMP;
061import static paint.shared.constants.PaintStringConstants.EXPERIMENT_NAME;
062import static paint.shared.constants.PaintStringConstants.RECORDING_NAME;
063
064
065/**
066 * Writes and maintains per-square cell assignment overrides for the PAINT Viewer.
067 * <p>
068 * Overrides are stored in {@code <project>/Viewer/Square Override.csv}. Each row identifies
069 * a square using {@code (experimentName, recordingName, squareNumber)} and provides an
070 * overridden {@code cellId} plus a timestamp. This class performs disk I/O only; it does
071 * not apply overrides to in-memory {@link paint.shared.objects.Square} instances.
072 */
073public class WriteSquareOverride {
074
075    /** Path to: Viewer/Square Override.csv */
076    private final Path csvFilePath;
077
078    /** CSV header columns (must match the persisted file exactly) */
079    private static final String[] HEADER = {
080            EXPERIMENT_NAME,
081            RECORDING_NAME,
082            SQUARE_NUMBER,
083            CELL_ID,
084            TIME_STAMP
085    };
086
087    /**
088     * Builds a composite key for a (experiment, recording, square) combination.
089     */
090    private static String key(String exp, String rec, int square) {
091        return exp + "§" + rec + "§" + square;
092    }
093
094    /**
095     * Constructs a new writer for the given project and ensures the {@code Viewer/} directory exists.
096     *
097     * @param projectPath root path of the project
098     */
099    public WriteSquareOverride(Path projectPath) {
100        Path viewerPath = projectPath.resolve("Viewer");
101        try {
102            Files.createDirectories(viewerPath);
103        } catch (IOException ignored) {}
104
105        this.csvFilePath = viewerPath.resolve("Square Override.csv");
106    }
107
108    /**
109     * Writes or updates the cell assignment overrides for selected squares.
110     * <p>
111     * The provided {@code assignments} are merged into any existing overrides:
112     * <ul>
113     *     <li>If {@code cellId == 0}, the override row for that square is removed.</li>
114     *     <li>Otherwise, the square’s row is added or replaced with the new cellId and timestamp.</li>
115     * </ul>
116     *
117     * @param recordingEntry the recording entry to update
118     * @param assignments    map of squareNumber → cellId
119     */
120    public void writeSquareOverridesToFile(RecordingEntry recordingEntry, Map<Integer, Integer> assignments) {
121        String experiment = recordingEntry.getExperimentName();
122        String recording  = recordingEntry.getRecordingName();
123        String timestamp  = LocalDateTime.now().toString();
124
125        // Load existing overrides into a map
126        Map<String, String> map = loadExistingRows();
127
128        // Update or remove entries
129        for (Map.Entry<Integer, Integer> e : assignments.entrySet()) {
130            int square = e.getKey();
131            int cellId = e.getValue();
132
133            String k = key(experiment, recording, square);
134
135            if (cellId == 0) {
136                // Remove override for this square
137                map.remove(k);
138            } else {
139                // Update override (replace existing row for this square)
140                String line = experiment + "," +
141                        recording  + "," +
142                        square     + "," +
143                        cellId     + "," +
144                        timestamp;
145
146                map.put(k, line);
147            }
148        }
149
150        writeAll(map);
151    }
152
153    // ========================================================================
154    //   LOAD EXISTING ROWS FROM CSV
155    // ========================================================================
156
157    /**
158     * Reads the CSV file into a map keyed by (experiment, recording, square).
159     * <p>
160     * If the file is missing, an empty map is returned. If the header is malformed,
161     * a warning is logged and an empty map is returned (caller will rewrite a clean file).
162     *
163     * @return a map of compositeKey → CSV row
164     */
165    private Map<String, String> loadExistingRows() {
166        Map<String, String> map = new LinkedHashMap<>();
167
168        if (!Files.exists(csvFilePath)) {
169            return map;
170        }
171
172        try {
173            List<String> lines = Files.readAllLines(csvFilePath);
174            if (lines.isEmpty()) {
175                return map;
176            }
177
178            String expectedHeader = String.join(",", HEADER);
179
180            // Validate header
181            if (!lines.get(0).equalsIgnoreCase(expectedHeader)) {
182                PaintLogger.warnf("Invalid Square Override.csv header. Rebuilding.");
183                return map;  // Start fresh
184            }
185
186            // Parse each row
187            for (int i = 1; i < lines.size(); i++) {
188                String line = lines.get(i).trim();
189                if (line.isEmpty()) continue;
190
191                String[] p = line.split(",", 5);
192                if (p.length < 4) continue;
193
194                String exp = p[0];
195                String rec = p[1];
196                int square = Integer.parseInt(p[2]);
197
198                map.put(key(exp, rec, square), line);
199            }
200
201        } catch (Exception e) {
202            PaintLogger.errorf("Failed reading Square Override.csv: %s", e.getMessage());
203        }
204
205        return map;
206    }
207
208    // ========================================================================
209    //   SAVE ALL ROWS BACK TO DISK
210    // ========================================================================
211
212    /**
213     * Writes all overrides back to the CSV file using an atomic file replace.
214     *
215     * @param map compositeKey → CSV row
216     */
217    private void writeAll(Map<String, String> map) {
218
219        String header = String.join(",", HEADER);
220
221        List<String> out = new ArrayList<>();
222        out.add(header);
223        out.addAll(map.values());
224
225        Path tmp = csvFilePath.resolveSibling(csvFilePath.getFileName() + ".tmp");
226
227        try {
228            Files.write(tmp, out);
229            Files.move(tmp, csvFilePath, StandardCopyOption.REPLACE_EXISTING);
230        } catch (IOException e) {
231            PaintLogger.errorf("Failed writing Square Override.csv: %s", e.getMessage());
232        }
233    }
234
235    // ========================================================================
236    //   QUERY FOR EXISTING OVERRIDES
237    // ========================================================================
238
239    /**
240     * Checks whether this recording already has override entries in {@code Square Override.csv}.
241     *
242     * @param re recording entry
243     * @return true if at least one override exists for this recording
244     */
245    public boolean hasOverridesFor(RecordingEntry re) {
246        String prefix = re.getExperimentName() + "," + re.getRecordingName() + ",";
247        try {
248            if (!Files.exists(csvFilePath)) {
249                return false;
250            }
251            return Files.readAllLines(csvFilePath)
252                        .stream()
253                        .anyMatch(line -> line.startsWith(prefix));
254        } catch (IOException e) {
255            PaintLogger.warnf("Error checking square overrides: %s", e.getMessage());
256            return false;
257        }
258    }
259
260    // ========================================================================
261    //   MERGE / REPLACE LOGIC
262    // ========================================================================
263
264    /**
265     * Merges new assignments into existing ones without removing other squares
266     * (except where {@code cellId == 0}, which removes that square’s override row).
267     */
268    public void mergeSquareOverrides(RecordingEntry re, Map<Integer, Integer> newAssignments) {
269        writeSquareOverridesToFile(re, newAssignments);
270    }
271
272    /**
273     * Replaces all existing assignments for the given recording with only the
274     * provided assignments.
275     *
276     * @param re             the recording entry whose overrides should be replaced
277     * @param newAssignments map of squareNumber → cellId
278     */
279    public void replaceSquareOverrides(
280            RecordingEntry re,
281            Map<Integer, Integer> newAssignments
282    ) {
283        String prefix = re.getExperimentName() + "," + re.getRecordingName() + ",";
284
285        try {
286            List<String> lines = Files.exists(csvFilePath)
287                    ? Files.readAllLines(csvFilePath)
288                    : new ArrayList<>();
289
290            String header = String.join(",", HEADER);
291
292            // Validate header
293            if (lines.isEmpty() || !lines.get(0).equals(header)) {
294                lines.clear();
295                lines.add(header);
296            }
297
298            // Remove all old rows for this recording
299            lines.removeIf(line -> line.startsWith(prefix) && !line.equals(header));
300
301            // Add new entries
302            String experimentName = re.getExperimentName();
303            String recordingName  = re.getRecordingName();
304            String timestamp      = LocalDateTime.now().toString();
305
306            for (Map.Entry<Integer,Integer> entry : newAssignments.entrySet()) {
307                int cellId = entry.getValue();
308                if (cellId == 0) {
309                    continue;
310                }
311
312                int squareNumber = entry.getKey();
313
314                String newLine =
315                        experimentName + "," +
316                                recordingName  + "," +
317                                squareNumber   + "," +
318                                cellId         + "," +
319                                timestamp;
320
321                lines.add(newLine);
322            }
323
324            // Atomic write
325            Path tmp = csvFilePath.resolveSibling("Square Override.tmp");
326            Files.write(tmp, lines);
327            Files.move(tmp, csvFilePath, StandardCopyOption.REPLACE_EXISTING);
328        } catch (IOException ex) {
329            PaintLogger.errorf("Error writing square overrides: %s", ex.getMessage());
330        }
331    }
332}
