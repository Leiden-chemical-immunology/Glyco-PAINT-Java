001/*==============================================================================
002 *  Class:        WriteRecordingOverride.java
003 *  Package:      paint.viewer.override.recording_override
004 *
005 *  PURPOSE:
006 *    Persist per-recording threshold override settings created in the PAINT
007 *    Viewer, so user-modified filtering parameters can be restored in later
008 *    Viewer sessions or applied in batch across an experiment or project.
009 *
010 *  DESCRIPTION:
011 *    This class manages the CSV file:
012 *
013 *        <project>/Viewer/Recording Override.csv
014 *
015 *    Each CSV row represents one recording-level override identified by the
016 *    composite key (experimentName, recordingName), storing:
017 *
018 *        experimentName,
019 *        recordingName,
020 *        timestamp,
021 *        minRequiredDensityRatio,
022 *        maxAllowableVariability,
023 *        minRequiredRSquared,
024 *        neighbourMode
025 *
026 *    The writer supports three scopes:
027 *      • Recording  – update only the currently selected recording
028 *      • Experiment – update all recordings in the same experiment
029 *      • Project    – update every recording in the project
030 *
031 *    Updates are non-destructive: existing entries for the same composite key
032 *    are replaced; all other rows are preserved. Writes are performed via a
033 *    temporary file and a replace/move operation to reduce the chance of
034 *    leaving a partially-written CSV.
035 *
036 *  KEY FEATURES:
037 *    • Persistent storage of viewer threshold overrides.
038 *    • Validates CSV header; rebuilds file if malformed.
039 *    • Non-destructive updates via composite-key map.
040 *    • Safe write strategy using temporary file replacement.
041 *
042 *  AUTHOR:
043 *    Hans Bakker
044 *
045 *  MODULE:
046 *    paint-viewer
047 *
048 *  UPDATED:
049 *    2025-12-31

055package paint.viewer.override.recording_override;
056
057import paint.shared.utils.PaintLogger;
058import paint.viewer.model.SquareControlParams;
059import paint.viewer.model.RecordingEntry;
060
061import java.io.IOException;
062import java.nio.file.*;
063import java.time.LocalDateTime;
064import java.util.*;
065
066import static paint.shared.constants.PaintStringConstants.*;
067
068/**
069 * Handles persistence of per-recording filter overrides in the PAINT Viewer.
070 * <p>
071 * Overrides are stored in:
072 * <pre>
073 *     Viewer/Recording Override.csv
074 * </pre>
075 * Each entry maps a {@code (experimentName, recordingName)} pair to a set of
076 * filter threshold values chosen by the user. This enables restoring these
077 * thresholds between Viewer sessions and applying batch updates across scopes.
078 */
079public class WriteRecordingOverride {
080
081    /** Path to: Viewer/Recording Override.csv */
082    private final Path csvFilePath;
083
084    /**
085     * Column headers for the CSV (must match the reader expectations elsewhere).
086     * The file is considered invalid if the header does not match, and will be
087     * rebuilt from scratch on the next write.
088     */
089    private static final String[] HEADER = {
090            EXPERIMENT_NAME,
091            RECORDING_NAME,
092            TIME_STAMP,
093            MIN_REQUIRED_DENSITY_RATIO,
094            MAX_ALLOWABLE_VARIABILITY,
095            MIN_REQUIRED_R_SQUARED,
096            NEIGHBOUR_MODE
097    };
098
099    /**
100     * Builds the composite key used for identifying override rows.
101     *
102     * @param exp experiment name
103     * @param rec recording name
104     * @return composite key: {@code exp + "§" + rec}
105     */
106    private static String key(String exp, String rec) {
107        return exp + "§" + rec;
108    }
109
110    /**
111     * Creates a new writer and ensures the {@code Viewer/} folder exists.
112     *
113     * @param projectPath root of the PAINT project
114     */
115    public WriteRecordingOverride(Path projectPath) {
116        Path viewerPath = projectPath.resolve("Viewer");
117
118        try {
119            Files.createDirectories(viewerPath);
120        } catch (IOException e) {
121            PaintLogger.warnf("Failed to create Viewer directory: %s", e.getMessage());
122        }
123
124        this.csvFilePath = viewerPath.resolve("Recording Override.csv");
125    }
126
127    /**
128     * Applies and writes overrides for the selected scope:
129     *
130     * <ul>
131     *   <li>{@code "Recording"}  – update only the current recording</li>
132     *   <li>{@code "Experiment"} – update all recordings in the same experiment</li>
133     *   <li>{@code "Project"}    – update every recording in the viewer</li>
134     * </ul>
135     *
136     * Overrides are written non-destructively: existing rows for the same
137     * {@code (experimentName, recordingName)} pair are replaced; all other entries
138     * are preserved.
139     *
140     * @param scope            one of: "Recording", "Experiment", "Project"
141     * @param params           current UI threshold parameters to persist
142     * @param recordingEntries all loaded recording entries in the viewer
143     * @param currentIndex     index of the currently selected recording entry
144     */
145    public void writeRecordingOverridesToFile(
146            String scope,
147            SquareControlParams params,
148            List<RecordingEntry> recordingEntries,
149            int currentIndex
150    ) {
151        String timestamp = LocalDateTime.now().toString();
152
153        if ("Recording".equals(scope)) {
154            RecordingEntry re = recordingEntries.get(currentIndex);
155            writeOrUpdate(re.getExperimentName(), re.getRecordingName(), params, timestamp);
156            updateMemory(re, params);
157            return;
158        }
159
160        if ("Experiment".equals(scope)) {
161            String experiment = recordingEntries.get(currentIndex).getExperimentName();
162
163            for (RecordingEntry re : recordingEntries) {
164                if (re.getExperimentName().equals(experiment)) {
165                    writeOrUpdate(re.getExperimentName(), re.getRecordingName(), params, timestamp);
166                    updateMemory(re, params);
167                }
168            }
169            return;
170        }
171
172        if ("Project".equals(scope)) {
173            for (RecordingEntry re : recordingEntries) {
174                writeOrUpdate(re.getExperimentName(), re.getRecordingName(), params, timestamp);
175                updateMemory(re, params);
176            }
177        }
178    }
179
180    // ====================================================================================
181    // WRITE / UPDATE LOGIC
182    // ====================================================================================
183
184    /**
185     * Writes a new override entry or updates an existing one for the given recording.
186     * All other entries are preserved unchanged.
187     *
188     * @param experimentName experiment identifier
189     * @param recordingName  recording identifier
190     * @param params         threshold values to store
191     * @param timestamp      ISO timestamp representing when the override was written
192     */
193    private void writeOrUpdate(
194            String experimentName,
195            String recordingName,
196            SquareControlParams params,
197            String timestamp
198    ) {
199        Map<String, String> map = loadExistingRows();
200
201        String line =
202                experimentName + "," +
203                        recordingName  + "," +
204                        timestamp      + "," +
205                        params.minRequiredDensityRatio + "," +
206                        params.maxAllowableVariability + "," +
207                        params.minRequiredRSquared     + "," +
208                        params.neighbourMode;
209
210        map.put(key(experimentName, recordingName), line);
211        writeAll(map);
212    }
213
214    // ====================================================================================
215    // READ CSV INTO MAP
216    // ====================================================================================
217
218    /**
219     * Loads the existing override rows from disk and returns them as a map keyed by
220     * {@code experimentName§recordingName}.
221     * <p>
222     * If the CSV does not exist, an empty map is returned.
223     * If the header is not exactly as expected, the file is treated as invalid and
224     * ignored (empty map returned), so the next write recreates it.
225     *
226     * @return map of {@code compositeKey → CSV row}
227     */
228    private Map<String, String> loadExistingRows() {
229        Map<String, String> map = new LinkedHashMap<>();
230
231        if (!Files.exists(csvFilePath)) {
232            return map;
233        }
234
235        try {
236            List<String> lines = Files.readAllLines(csvFilePath);
237            if (lines.isEmpty()) return map;
238
239            String expectedHeader = String.join(",", HEADER);
240
241            // Header is invalid → ignore file and rebuild on next write.
242            if (!lines.get(0).equalsIgnoreCase(expectedHeader)) {
243                PaintLogger.warnf("Invalid Recording Override.csv header. Rebuilding.");
244                return map;
245            }
246
247            // Parse entries (store full line, indexed by composite key).
248            for (int i = 1; i < lines.size(); i++) {
249                String line = lines.get(i).trim();
250                if (line.isEmpty()) continue;
251
252                String[] parts = line.split(",", 3);
253                if (parts.length < 2) continue;
254
255                String exp = parts[0];
256                String rec = parts[1];
257
258                map.put(key(exp, rec), line);
259            }
260
261        } catch (IOException ex) {
262            PaintLogger.errorf("Failed reading Recording Override.csv: %s", ex.getMessage());
263        }
264
265        return map;
266    }
267
268    // ====================================================================================
269    // WRITE ALL ROWS (ATOMIC)
270    // ====================================================================================
271
272    /**
273     * Writes all override entries back to disk using a temporary file and replace/move.
274     * <p>
275     * This reduces the risk of leaving a truncated CSV if the process is interrupted.
276     *
277     * @param map map of {@code compositeKey → CSV row}
278     */
279    private void writeAll(Map<String, String> map) {
280        String header = String.join(",", HEADER);
281
282        List<String> out = new ArrayList<>();
283        out.add(header);
284        out.addAll(map.values());
285
286        Path tmp = csvFilePath.resolveSibling(csvFilePath.getFileName() + ".tmp");
287
288        try {
289            Files.write(tmp, out);
290            Files.move(tmp, csvFilePath, StandardCopyOption.REPLACE_EXISTING);
291        } catch (IOException e) {
292            PaintLogger.errorf("Failed writing Recording Override.csv: %s", e.getMessage());
293        }
294    }
295
296    // ====================================================================================
297    // UPDATE IN-MEMORY RECORDING ENTRY
298    // ====================================================================================
299
300    /**
301     * Updates the in-memory {@link paint.shared.objects.Recording} referenced by the
302     * provided {@link RecordingEntry} so the UI and filtering logic immediately reflect
303     * the current override values.
304     *
305     * @param re     recording entry to update
306     * @param params source threshold values (typically from the UI)
307     */
308    private void updateMemory(RecordingEntry re, SquareControlParams params) {
309        re.getRecording().setMinRequiredDensityRatio(params.minRequiredDensityRatio);
310        re.getRecording().setMaxAllowableVariability(params.maxAllowableVariability);
311        re.getRecording().setMinRequiredRSquared(params.minRequiredRSquared);
312        re.getRecording().setNeighbourMode(params.neighbourMode);
313    }
