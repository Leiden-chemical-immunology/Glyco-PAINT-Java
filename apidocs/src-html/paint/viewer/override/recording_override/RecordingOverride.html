001/*==============================================================================
002 *  Class:        RecordingOverride.java
003 *  Package:      paint.viewer.override.recording_override
004 *
005 *  PURPOSE:
006 *    Represents a single recording-level override entry. Each instance defines
007 *    replacement threshold parameters for one recording, identified by the
008 *    (experimentName, recordingName) pair.
009 *
010 *  DESCRIPTION:
011 *    This class models one row from the "Recording Override.csv" file located in
012 *    the Viewer directory of a PAINT project. It is a simple data container
013 *    designed for direct CSV parsing and in-memory consumption by
014 *    ImportRecordingOverride and related override/export utilities.
015 *
016 *    The values stored here replace the corresponding threshold parameters
017 *    (density ratio, R², variability, neighbour mode) on the target recording.
018 *
019 *  KEY FEATURES:
020 *    • Plain data container (POJO) with no behavior.
021 *    • Field layout mirrors the Recording Override CSV columns.
022 *    • Used exclusively for override import/export and in-memory mutation.
023 *
024 *  AUTHOR:
025 *    Hans Bakker
026 *
027 *  MODULE:
028 *    paint-viewer
029 *
030 *  UPDATED:
031 *    2025-12-31
032 *

035==============================================================================*/

037package paint.viewer.override.recording_override;
038
039public class RecordingOverride {
040
041    private String experimentName;
042    private String recordingName;
043
044    private double minRequiredDensityRatio;
045    private double minRequiredRSquared;
046    private double maxAllowableVariability;
047    private String neighbourMode;
048
049    // ───────────────────────────────────────────────────────────────────────────────
050    // CONSTRUCTORS
051    // ───────────────────────────────────────────────────────────────────────────────
052
053    /**
054     * Default constructor required for CSV parsing and reflective instantiation.
055     */
056    public RecordingOverride() {
057    }
058
059    // ───────────────────────────────────────────────────────────────────────────────
060    // ACCESSORS
061    // ───────────────────────────────────────────────────────────────────────────────
062
063    public String getExperimentName() {
064        return experimentName;
065    }
066
067    public void setExperimentName(String experimentName) {
068        this.experimentName = experimentName;
069    }
070
071    public String getRecordingName() {
072        return recordingName;
073    }
074
075    public void setRecordingName(String recordingName) {
076        this.recordingName = recordingName;
077    }
078
079    public double getMinRequiredDensityRatio() {
080        return minRequiredDensityRatio;
081    }
082
083    public void setMinRequiredDensityRatio(double minRequiredDensityRatio) {
084        this.minRequiredDensityRatio = minRequiredDensityRatio;
085    }
086
087    public double getMinRequiredRSquared() {
088        return minRequiredRSquared;
089    }
090
091    public void setMinRequiredRSquared(double minRequiredRSquared) {
092        this.minRequiredRSquared = minRequiredRSquared;
093    }
094
095    public double getMaxAllowableVariability() {
096        return maxAllowableVariability;
097    }
098
099    public void setMaxAllowableVariability(double maxAllowableVariability) {
100        this.maxAllowableVariability = maxAllowableVariability;
101    }
102
103    public String getNeighbourMode() {
104        return neighbourMode;
105    }
106
107    public void setNeighbourMode(String neighbourMode) {
108        this.neighbourMode = neighbourMode;
109    }
110
111    // ───────────────────────────────────────────────────────────────────────────────
112    // DEBUG
113    // ───────────────────────────────────────────────────────────────────────────────
114
115    /**
116     * Returns a human-readable representation of this override entry,
117     * primarily intended for logging and debugging.
118     */
119    @Override
120    public String toString() {
121        return "RecordingOverride{" +
122                "experimentName='" + experimentName + '\'' +
123                ", recordingName='" + recordingName + '\'' +
124                ", minRequiredDensityRatio=" + minRequiredDensityRatio +
125                ", minRequiredRSquared=" + minRequiredRSquared +
126                ", maxAllowableVariability=" + maxAllowableVariability +
127                ", neighbourMode='" + neighbourMode + '\'' +
128                '}';
129    }
130}
