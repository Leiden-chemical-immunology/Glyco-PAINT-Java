041package paint.viewer.override.recording_override;
042
043import static paint.shared.constants.PaintStringConstants.*;
044
045import paint.shared.utils.PaintLogger;
046import paint.viewer.model.RecordingEntry;
047import tech.tablesaw.api.Table;
048
049import java.nio.file.Files;
050import java.nio.file.Path;
051import java.util.ArrayList;
052import java.util.HashMap;
053import java.util.List;
054import java.util.Map;
055
056/**
057 * Imports and applies recording-level threshold overrides to a collection of
058 * {@link RecordingEntry} objects.
059 * <p>
060 * Overrides are loaded from <project>/Viewer/Recording Override.csv. Each row
061 * specifies the updated threshold values for a single recording. Rows are matched
062 * to {@link RecordingEntry} objects using a composite key:
063 * {@code experimentName + "§" + recordingName}.
064 * <p>
065 * This class performs in-memory mutation only (no file writes).
066 */
067public final class ImportRecordingOverride {
068
069    // ────────────────────────────────────────────────────────────
070    // PUBLIC ENTRY POINT
071    // ────────────────────────────────────────────────────────────
072
073    /**
074     * Loads and applies all recording overrides to the given list of
075     * {@link RecordingEntry} objects.
076     * <p>
077     * If <project>/Viewer/Recording Override.csv does not exist, this method
078     * returns immediately without modifying anything.
079     *
080     * @param recordingEntries list of in-memory {@link RecordingEntry} objects to update
081     * @param projectPath      project root folder containing the {@code Viewer} directory
082     */
083    public static void importRecordingOverrides(List<RecordingEntry> recordingEntries, Path projectPath) {
084
085        Path csvPath = projectPath.resolve("Viewer").resolve("Recording Override.csv");
086
087        if (!Files.exists(csvPath)) {
088            PaintLogger.infof("No Recording Override.csv present - no overrides to apply.");
089            return;
090        }
091
092        List<RecordingOverride> overrides = loadRecordingOverride(csvPath);
093        applyInternal(recordingEntries, overrides);
094    }
095
096    // ────────────────────────────────────────────────────────────
097    // INTERNAL APPLY
098    // ────────────────────────────────────────────────────────────
099
100    /**
101     * Applies overrides that were already loaded from CSV.
102     * <p>
103     * Implementation notes:
104     * <ul>
105     *   <li>Builds a composite-key map for O(1) override lookup.</li>
106     *   <li>Mutates {@link RecordingEntry} recording threshold fields in place.</li>
107     *   <li>Logs only fields whose values changed.</li>
108     * </ul>
109     *
110     * @param entries   recording entries to update
111     * @param overrides overrides parsed from CSV
112     */
113    private static void applyInternal(List<RecordingEntry> entries, List<RecordingOverride> overrides) {
114
115        Map<String, RecordingOverride> map = new HashMap<>();
116
117        for (RecordingOverride override : overrides) {
118            map.put(key(override.getExperimentName(), override.getRecordingName()), override);
119        }
120
121
122        for (RecordingEntry entry : entries) {
123
124            String experimentName = entry.getExperimentName();
125            String recordingName  = entry.getRecordingName();
126
127            RecordingOverride override = map.get(key(experimentName, recordingName));
128
129            if (override != null) {
130
131                // Capture current values for change-only logging
132                double oldDensityRatio   = entry.getRecording().getMinRequiredDensityRatio();
133                double oldRSquared       = entry.getRecording().getMinRequiredRSquared();
134                double oldVariability    = entry.getRecording().getMaxAllowableVariability();
135                String oldNeighbourMode  = entry.getRecording().getNeighbourMode();
136
137                // Apply overrides (in-memory only)
138                entry.getRecording().setMinRequiredDensityRatio(override.getMinRequiredDensityRatio());
139                entry.getRecording().setMinRequiredRSquared(override.getMinRequiredRSquared());
140                entry.getRecording().setMaxAllowableVariability(override.getMaxAllowableVariability());
141                entry.getRecording().setNeighbourMode(override.getNeighbourMode());
142
143                // Log what changed for this recording
144                PaintLogger.infof(
145                        "Applied Recording override on %s",
146                        entry.getRecordingName());
147
148                if (oldDensityRatio != override.getMinRequiredDensityRatio()) {
149                    PaintLogger.infof("                    DensityRatio:   %-6.2f → %.2f",
150                                      oldDensityRatio, override.getMinRequiredDensityRatio());
151                }
152
153                if (oldRSquared != override.getMinRequiredRSquared()) {
154                    PaintLogger.infof("                    R²:             %-6.2f → %.2f",
155                                      oldRSquared,
156                                      override.getMinRequiredRSquared());
157                }
158
159                if (oldVariability != override.getMaxAllowableVariability()) {
160                    PaintLogger.infof("                    Variability:    %-6.2f → %.2f",
161                                      oldVariability,
162                                      override.getMaxAllowableVariability());
163                }
164
165                if (!oldNeighbourMode.equals(override.getNeighbourMode())) {
166                    PaintLogger.infof("                    Neighbour Mode: %6s → %s",
167                                      oldNeighbourMode,
168                                      override.getNeighbourMode());
169                }
170                PaintLogger.blankline();
171            }
172        }
173    }
174
175    // ────────────────────────────────────────────────────────────
176    // CSV LOADING
177    // ────────────────────────────────────────────────────────────
178
179    /**
180     * Loads all rows from {@code Recording Override.csv} into a list of
181     * {@link RecordingOverride} objects.
182     *
183     * @param csvFile path to {@code Recording Override.csv}
184     * @return list of parsed {@link RecordingOverride} objects (possibly empty)
185     */
186    public static List<RecordingOverride> loadRecordingOverride(Path csvFile) {
187
188        List<RecordingOverride> list = new ArrayList<>();
189
190        try {
191            Table table = Table.read().csv(csvFile.toString());
192
193            for (int i = 0; i < table.rowCount(); i++) {
194                RecordingOverride recordingOverride = new RecordingOverride();
195
196                recordingOverride.setExperimentName(          table.column(EXPERIMENT_NAME).get(i).toString());
197                recordingOverride.setRecordingName(           table.column(RECORDING_NAME).get(i).toString());
198                recordingOverride.setMinRequiredDensityRatio( Double.parseDouble(table.column(MIN_REQUIRED_DENSITY_RATIO).get(i).toString()));
199                recordingOverride.setMinRequiredRSquared(     Double.parseDouble(table.column(MIN_REQUIRED_R_SQUARED).get(i).toString()));
200                recordingOverride.setMaxAllowableVariability( Double.parseDouble(table.column(MAX_ALLOWABLE_VARIABILITY).get(i).toString()));
201                recordingOverride.setNeighbourMode(           table.column(NEIGHBOUR_MODE).get(i).toString());
202
203                list.add(recordingOverride);
204            }
205        } catch (Exception ex) {
206            PaintLogger.errorf( "Error reading Recording Override.csv → " + ex.getMessage());
207        }
208
209        return list;
210    }
211
212    /**
213     * Builds the composite key used for override lookups.
214     *
215     * @param experimentName experiment name
216     * @param recordingName  recording name
217     * @return composite key in the format {@code experimentName + "§" + recordingName}
218     */
219    private static String key(String experimentName, String recordingName) {
220        return experimentName + "§" + recordingName;
221    }
222
223    /** Utility class; not instantiable. */
224    private ImportRecordingOverride() {}
225}
