011 *    ("Recording Name") listing recordings that should be excluded from
012 *    downstream processing.
013 *

015 *    semantics (how and where exclusions are applied) are handled elsewhere

036package paint.viewer.override.recording_exclude;
037
038public class RecordingExclude {
039
040    /**
041     * Name of the recording to be excluded.
042     * Must exactly match the "Recording Name" used in Recordings.csv.
043     */
044    private String recordingName;
045
046    // ───────────────────────────────────────────────────────────────────────────────
047    // CONSTRUCTORS
048    // ───────────────────────────────────────────────────────────────────────────────
049
050    /**
051     * Default constructor required for CSV deserialization.
052     */
053    public RecordingExclude() {
054    }
055
056    // ───────────────────────────────────────────────────────────────────────────────
057    // ACCESSORS
058    // ───────────────────────────────────────────────────────────────────────────────
059
060    /**
061     * Returns the recording name associated with this exclusion.
062     *
063     * @return recording name
064     */
065    public String getRecordingName() {
066        return recordingName;
067    }
068
069    /**
070     * Sets the recording name associated with this exclusion.
071     *
072     * @param recordingName recording name to exclude
073     */
074    public void setRecordingName(String recordingName) {
075        this.recordingName = recordingName;
076    }
077
078    // ───────────────────────────────────────────────────────────────────────────────
079    // DEBUG
080    // ───────────────────────────────────────────────────────────────────────────────
081
082    @Override
083    public String toString() {
084        return "RecordingExclude{" +
085                "recordingName='" + recordingName + '\'' +
086                '}';
087    }
088}
