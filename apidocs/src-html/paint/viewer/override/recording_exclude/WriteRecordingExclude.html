001/*==============================================================================
002 *  Class:        WriteRecordingExclude.java
003 *  Package:      paint.viewer.override.recording_exclude
004 *
005 *  PURPOSE:
006 *    Utility methods for persisting "recording excluded" state from the Viewer.
007 *    Supports:
008 *      (1) patching the per-experiment Recordings.csv Exclude flag, and
009 *      (2) maintaining the project-level Viewer/Recording Exclude.csv list.
010 *
011 *  DESCRIPTION:
012 *    The Viewer can mark recordings as excluded. This class provides two small
013 *    persistence helpers:
014 *
015 *      • patchRecordingExcluded(...)
016 *          Reads an experiment's Recordings.csv, updates the "Exclude" boolean
017 *          for one recording, and writes the file back to disk.
018 *
019 *      • updateExcludeRecordingsCsv(...)
020 *          Updates the project-level Viewer/Recording Exclude.csv file, which
021 *          contains a single column ("Recording Name") listing all excluded
022 *          recordings. When excluded=true the recording is added (if not
023 *          already present). When excluded=false it is removed (if present).
024 *
025 *    These methods are intended to be called directly from Viewer UI actions
026 *    (e.g., toggling the Exclude/Include button).
027 *
028 *  KEY FEATURES:
029 *    • Updates the per-experiment Recordings.csv Exclude column for one recording.
030 *    • Maintains the Viewer/Recording Exclude.csv list (create/read/update).
031 *    • Avoids duplicates when adding; removes all matching rows when removing.
032 *
033 *  AUTHOR:
034 *    Hans Bakker

037 *    paint-viewer
038 *
039 *  UPDATED:
040 *    2025-12-31

046package paint.viewer.override.recording_exclude;
047
048import paint.shared.io.MainIOInterface;
049import paint.shared.objects.Recording;
050import paint.shared.utils.PaintLogger;
051import tech.tablesaw.api.BooleanColumn;
052import tech.tablesaw.api.StringColumn;
053import tech.tablesaw.api.Table;
054
055import java.nio.file.Files;
056import java.nio.file.Path;
057
058import static paint.shared.constants.PaintFileNames.RECORDINGS_CSV;
059
060public class WriteRecordingExclude {
061
062    /**
063     * Updates the per-experiment Recordings.csv by setting the {@code Exclude}
064     * flag for a single recording.
065     * <p>
066     * This method reads the experiment's recordings table, finds the row whose
067     * recording name matches {@code recordingName}, updates the boolean exclude
068     * value, and writes the table back to disk.
069     *
070     * @param experimentPath path to the experiment folder that contains Recordings.csv
071     * @param recordingName  the recording name to update
072     * @param excluded       the new exclude state to store
073     */
074    public static void patchRecordingExcluded(
075            Path experimentPath,
076            String recordingName,
077            boolean excluded
078    ) {
079        Table table = MainIOInterface.readRecordingsTable(experimentPath);
080        if (table == null) {
081            return;
082        }
083
084        StringColumn nameCol     = table.stringColumn(Recording.Column.RECORDING_NAME.header);
085        BooleanColumn excludedCol = table.booleanColumn(Recording.Column.EXCLUDE.header);
086
087        for (int row = 0; row < table.rowCount(); row++) {
088            if (nameCol.get(row).equals(recordingName)) {
089                excludedCol.set(row, excluded);
090                break;
091            }
092        }
093
094        MainIOInterface.writeSpecificRecordingsFile(
095                experimentPath.resolve(RECORDINGS_CSV),
096                table
097        );
098    }
099
100    /**
101     * Updates the project-level {@code Viewer/Recording Exclude.csv} file.
102     * <p>
103     * The file is expected to contain exactly one column named {@code "Recording Name"}
104     * listing recordings that should be treated as excluded. If the Viewer folder
105     * or CSV does not exist, they are created as needed.
106     * <ul>
107     *   <li>If {@code excluded == true}: add {@code recordingName} if not present.</li>
108     *   <li>If {@code excluded == false}: remove all matching rows for {@code recordingName}.</li>
109     * </ul>
110     *
111     * @param projectPath    project root folder containing the {@code Viewer} directory
112     * @param recordingName  recording name to add/remove
113     * @param excluded       whether the recording should be present in the exclude list
114     */
115    public static void updateExcludeRecordingsCsv(Path projectPath, String recordingName, boolean excluded) {
116        try {
117            Path viewerDir = projectPath.resolve("Viewer");
118            if (Files.notExists(viewerDir)) {
119                Files.createDirectories(viewerDir);
120            }
121
122            Path file = viewerDir.resolve("Recording Exclude.csv");
123
124            final String colName = "Recording Name";
125            Table table;
126
127            if (Files.exists(file)) {
128                table = Table.read().csv(file.toString());
129                if (!table.columnNames().contains(colName)) {
130                    // If the file exists but is malformed (missing expected column), rebuild a minimal table.
131                    table = Table.create("Exclude Recordings", StringColumn.create(colName));
132                }
133            } else {
134                // Create a new minimal exclude table with the expected single column.
135                table = Table.create("Exclude Recordings", StringColumn.create(colName));
136            }
137
138            StringColumn col = table.stringColumn(colName);
139
140            if (excluded) {
141                // Append only if it is not already present.
142                boolean already = false;
143                for (int i = 0; i < table.rowCount(); i++) {
144                    if (recordingName.equals(col.get(i))) {
145                        already = true;
146                        break;
147                    }
148                }
149                if (!already) {
150                    col.append(recordingName);
151                }
152            } else {
153                // Remove all occurrences (iterate backwards to keep indices valid).
154                for (int i = table.rowCount() - 1; i >= 0; i--) {
155                    if (recordingName.equals(col.get(i))) {
156                        table = table.dropRows(i);
157                    }
158                }
159            }
160
161            table.write().csv(file.toString());
162
163        } catch (Exception e) {
164            PaintLogger.errorf("Failed to update Exclude Recordings.csv: %s", e.getMessage());
165        }
166    }
167}
