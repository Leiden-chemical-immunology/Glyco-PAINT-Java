001/*==============================================================================
002 *  Class:        ImportRecordingExclude.java
003 *  Package:      paint.viewer.override.recording_exclude
004 *
005 *  PURPOSE:
006 *    Loads "Recording Exclude.csv" from the project's /Viewer folder and applies
007 *    the exclude list to either:
008 *      • in-memory Viewer RecordingEntry objects, or
009 *      • an in-memory Tablesaw Recordings table.
010 *
011 *  DESCRIPTION:
012 *    The exclude file is intentionally minimal and currently contains only:
013 *
014 *        Recording Name
015 *
016 *    Applying excludes is a two-step operation:
017 *      1) Reset all recordings to Exclude = false.
018 *      2) For each Recording Name listed in the exclude file, set Exclude = true
019 *         on the matching recording(s).
020 *
021 *    This importer performs in-memory mutation only. Persistence (writing updated
022 *    Recordings CSVs) is handled elsewhere.
023 *
024 *  KEY FEATURES:
025 *    • Safe no-op when the CSV file does not exist.
026 *    • Applies excludes to both RecordingEntry lists and Tablesaw tables.
027 *    • Resets Exclude flags before applying new excludes for deterministic results.
028 *
029 *  AUTHOR:
030 *    Hans Bakker
031 *
032 *  MODULE:
033 *    paint-viewer
034 *
035 *  UPDATED:
036 *    2025-12-31

042package paint.viewer.override.recording_exclude;
043
044import paint.shared.utils.PaintLogger;
045import paint.viewer.model.RecordingEntry;
046import tech.tablesaw.api.BooleanColumn;
047import tech.tablesaw.api.StringColumn;
048import tech.tablesaw.api.Table;
049
050import java.nio.file.Files;
051import java.nio.file.Path;
052import java.util.ArrayList;
053import java.util.List;
054
055import static paint.shared.constants.PaintStringConstants.*;
056
057public class ImportRecordingExclude {
058
059    /**
060     * Loads and applies recording excludes to the provided list of {@link RecordingEntry}
061     * objects. If "Recording Exclude.csv" does not exist, this method returns without
062     * modifying anything.
063     *
064     * @param recordingEntries recordings to update in memory
065     * @param projectPath      project root folder containing /Viewer/Recording Exclude.csv
066     */
067    public static void importRecordingExcludes(List<RecordingEntry> recordingEntries, Path projectPath) {
068        Path csvPath = projectPath.resolve("Viewer").resolve("Recording Exclude.csv");
069
070        if (!Files.exists(csvPath)) {
071            PaintLogger.infof("No Recording Exclude.csv present - no overrides to apply.");
072            return;
073        }
074
075        List<RecordingExclude> excludes = loadRecordingExclude(csvPath);
076        applyInternal(projectPath, recordingEntries, excludes);
077    }
078
079    /**
080     * Loads and applies recording excludes to the provided in-memory Recordings table.
081     * If "Recording Exclude.csv" does not exist, this method returns without modifying
082     * anything.
083     *
084     * @param recordingsTable Recordings table to update in memory
085     * @param projectPath     project root folder containing /Viewer/Recording Exclude.csv
086     */
087    public static void importRecordingExcludes(Table recordingsTable, Path projectPath) {
088
089        Path csvPath = projectPath.resolve("Viewer").resolve("Recording Exclude.csv");
090
091        if (!Files.exists(csvPath)) {
092            PaintLogger.infof("No Recording Exclude.csv present - no overrides to apply.");
093            return;
094        }
095
096        List<RecordingExclude> excludes = loadRecordingExclude(csvPath);
097        applyInternal(projectPath, recordingsTable, excludes);
098    }
099
100    // ────────────────────────────────────────────────────────────
101    // INTERNAL APPLY
102    // ────────────────────────────────────────────────────────────
103
104    /**
105     * Applies excludes already loaded from CSV to the provided Recordings table.
106     * This method always resets the full Exclude column to {@code false} first,
107     * and then sets {@code true} only for Recording Names listed in the exclude file.
108     *
109     * @param projectPath      project root folder (currently only used for optional debug output)
110     * @param recordingsTable  target Recordings table to update
111     * @param excludes         list of excludes parsed from CSV
112     */
113    private static void applyInternal(Path projectPath, Table recordingsTable,
114            List<RecordingExclude> excludes) {
115
116        BooleanColumn excludeCol = recordingsTable.booleanColumn("Exclude");
117
118        // Reset all to false first (deterministic)
119        for (int i = 0; i < excludeCol.size(); i++) {
120            excludeCol.set(i, false);
121        }
122
123        // Apply excludes by Recording Name
124        boolean first = true;
125        for (RecordingExclude exclude : excludes) {
126            StringColumn  nameCol       = recordingsTable.stringColumn("Recording Name");
127            String        recordingName = exclude.getRecordingName();
128
129            for (int row = 0; row < recordingsTable.rowCount(); row++) {
130                if (recordingName.equals(nameCol.get(row))) {
131                    excludeCol.set(row, true);
132                    if (first) {
133                        PaintLogger.blankline();
134                        first = false;
135                    }
136                    PaintLogger.infof("Recording excluded: " + recordingName);
137                    break;
138                }
139            }
140        }
141    }
142
143    /**
144     * Applies excludes already loaded from CSV to the provided list of {@link RecordingEntry}
145     * objects. This method always resets all entries to {@code excluded=false} first,
146     * and then sets {@code true} only for Recording Names listed in the exclude file.
147     *
148     * @param projectPath      project root folder (currently only used for optional debug output)
149     * @param recordingEntries target entries to update in memory
150     * @param excludes         list of excludes parsed from CSV
151     */
152    private static void applyInternal(Path projectPath, List<RecordingEntry> recordingEntries,
153            List<RecordingExclude> excludes) {
154
155        // Reset all to false first (deterministic)
156        for (RecordingEntry entry : recordingEntries) {
157            entry.getRecording().setExcluded(false);
158        }
159
160        // Apply excludes by Recording Name
161        for (RecordingExclude exclude : excludes) {
162            String recordingName = exclude.getRecordingName();
163
164            for (RecordingEntry entry : recordingEntries) {
165                if (recordingName.equals(entry.getRecordingName())) {
166                    entry.getRecording().setExcluded(true);
167                    PaintLogger.infof("Recording excluded: " + recordingName);
168                    break;
169                }
170            }
171        }
172    }
173
174    // ────────────────────────────────────────────────────────────
175    // CSV LOADING
176    // ────────────────────────────────────────────────────────────
177
178    /**
179     * Loads all rows from "Recording Exclude.csv" into a list of {@link RecordingExclude}
180     * objects. The file is expected to contain a column named {@code Recording Name}.
181     *
182     * @param csvFile path to Recording Exclude.csv
183     * @return list of parsed {@link RecordingExclude} objects (possibly empty)
184     */
185    public static List<RecordingExclude> loadRecordingExclude(Path csvFile) {
186
187        List<RecordingExclude> recordingExcludeList = new ArrayList<>();
188
189        try {
190            Table table = Table.read().csv(csvFile.toString());
191
192            for (int i = 0; i < table.rowCount(); i++) {
193                RecordingExclude recordingExclude = new RecordingExclude();
194                recordingExclude.setRecordingName(table.column(RECORDING_NAME).get(i).toString());
195                recordingExcludeList.add(recordingExclude);
196            }
197        } catch (Exception ex) {
198            PaintLogger.errorf("Error reading Recording Exclude.csv → " + ex.getMessage());
199        }
200
201        return recordingExcludeList;
202    }

204}
